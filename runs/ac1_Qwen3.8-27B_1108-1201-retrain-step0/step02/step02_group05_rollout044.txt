"""
Optimized peak-autocorrelation minimization using Nesterov-accelerated subgradient
descent with momentum, coordinate descent refinement, diverse multi-start exploration,
and symmetric sequence exploitation. Targets score < 1.5027 for n=256 sequences.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a)) for a normalized sequence (sum=1)."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def compute_subgradient(a, top_m=10, temperature=0.01):
    """
    Compute a smoothed subgradient of max(conv(a,a)) w.r.t. a.
    Uses softmax weighting over top_m lags for stability.
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    # Softmax weights over top lags
    shifted = (c[:n] - c_max) / temperature  # Only first n lags (by symmetry of non-neg constraint, max is often near center)
    # Actually use all lags for generality
    shifted = (c - c_max) / temperature
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_exp if False else exp_shifted)
    weights = exp_shifted / sum_exp

    # Compute weighted subgradient
    # d(a*a)_k / da_i = 2 * a_{k-i} if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    # Vectorized: for each k, grad contribution is 2 * weights[k] * (a reversed, shifted by k)
    # a[k-i] for i = 0..n-1 is the sequence a[k], a[k-1], ..., clipped to valid range
    # This is equivalent to: for each k, take a[max(0,k-n+1):k+1] reversed, padded
    # More efficiently: convolve weights with reversed a
    # grad[i] = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
    # = 2 * (weights * a_rev)[i] where a_rev is a reversed and appropriately handled
    # Actually: grad[i] = 2 * sum_k weights[k] * a[k-i] for 0<=k-i<n
    # Let j = k-i, so k = i+j: grad[i] = 2 * sum_j weights[i+j] * a[j] for valid i+j
    # This is a cross-correlation of weights and a:
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, weights)
    # cross[k] = sum_j a_rev[j] * weights[k-j] = sum_j a[n-1-j] * weights[k-j]
    # We want: grad[i] = 2 * sum_k weights[k] * a[k-i]
    # For k from 0 to 2n-2, a[k-i] is valid when 0<=k-i<n
    # Let's just do it directly with the convolution result
    # cross[m] = sum_j a[n-1-j] * weights[m-j]
    # Set m = 2n-2-i: cross[2n-2-i] = sum_j a[n-1-j] * weights[2n-2-i-j]
    # Set j' = n-1-j: cross[2n-2-i] = sum_j' a[j'] * weights[2n-2-i-(n-1-j')] = sum_j' a[j'] * weights[n-1-i+j']
    # Hmm, this is getting complicated. Let me just use the direct formula.
    # grad[i] = 2 * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
    #          = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    grad = np.zeros(n)
    for j in range(n):
        # weights[i+j] for i=0..n-1, j fixed
        # This is weights[j:j+n]
        grad += 2.0 * a[j] * weights[j:j+n]

    return grad, c, weights


def nesterov_subgrad_descent(a0, max_time=120.0, top_m=10, tau=0.01,
                              lr_init=0.005, momentum_init=0.9,
                              decay_lr=0.9999, decay_mom=1.0 + 1e-6):
    """
    Nesterov-accelerated subgradient descent on the probability simplex.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    v = np.zeros(n)
    y = a.copy()
    lr = lr_init
    mu = momentum_init
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall_count = 0

    while time.monotonic() - t0 < max_time:
        # Nesterov lookahead
        y = a + mu * v
        y = np.maximum(y, 1e-12)
        y = y / np.sum(y)

        # Compute subgradient
        grad, c, weights = compute_subgradient(y, top_m=top_m, temperature=tau)

        # Project gradient onto tangent space of simplex (subtract mean)
        grad_tangent = grad - np.mean(grad)

        # Momentum update
        v = mu * v - lr * grad_tangent
        a_new = a + v

        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

        # Check improvement
        score_new = 2.0 * n * np.max(fft_convolve(a_new, a_new))

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                # Restart momentum
                v *= 0.5
                lr *= 0.7
                stall_count = 0

        a = a_new
        lr *= decay_lr
        mu = min(mu * decay_mom, 0.999)

    return best_a, best_score


def coordinate_descent_refine(a0, max_time=60.0, top_lags=5):
    """
    Coordinate descent: for each index i, find optimal a_i that minimizes
    the maximum convolution value, optimizing over the top lags.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < max_time:
        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            # Fix all except a[i], optimize a[i]
            # (a*a)_k = a[i]^2 * delta(2i,k) + a[i] * b_k + c_k
            # For k != 2i: (a*a)_k is linear in a[i] with slope 2*a[k-i] (if valid)
            # For k = 2i: quadratic a[i]^2
            # We want min_{a[i]>=0} max_k (a*a)_k subject to sum=1
            # Since we're changing a[i], the sum constraint means we need to adjust
            # Actually, let's just do a 1D search: try a few values of a[i]
            old_val = a[i]
            best_val = old_val
            best_max = np.inf

            c_current = fft_convolve(a, a)
            max_c = np.max(c_current)

            # Try perturbation values
            for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                trial = a.copy()
                trial[i] = old_val * scale
                s = np.sum(trial)
                if s < 1e-10:
                    continue
                trial = trial / s
                c_trial = fft_convolve(trial, trial)
                max_trial = np.max(c_trial)
                if max_trial < best_max:
                    best_max = max_trial
                    best_val = old_val * scale

            if best_val != old_val:
                a[i] = best_val
                s = np.sum(a)
                a = a / s
                new_score = 2.0 * n * np.max(fft_convolve(a, a))
                if new_score < best_score:
                    best_score = new_score
                    improved = True

        if not improved:
            break

    return a, best_score


def symmetric_optimize(a0, max_time=60.0):
    """
    Optimize assuming symmetry: a[i] = a[n-1-i].
    Halves the search space.
    """
    n = len(a0)
    half = (n + 1) // 2
    # Extract half
    a_half = a0[:half].copy()
    # If n is even, the middle two should be equal; if odd, middle is free
    if n % 2 == 0:
        a_half[-1] = (a0[half - 1] + a0[half]) / 2.0

    def expand(h):
        if n % 2 == 0:
            return np.concatenate([h, h[::-1]])
        else:
            return np.concatenate([h, h[-2::-1]])

    a_full = expand(a_half)
    a_full = a_full / np.sum(a_full)
    best_score = compute_score(a_full)
    t0 = time.monotonic()
    lr = 0.01
    rng = np.random.default_rng(456)

    while time.monotonic() - t0 < max_time:
        a_full = expand(a_half)
        a_full = np.maximum(a_full, 1e-10)
        a_full = a_full / np.sum(a_full)

        # Compute gradient on full sequence, then project to half
        c = fft_convolve(a_full, a_full)
        # Simple subgradient
        peak_idx = int(np.argmax(c))
        grad_full = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n, peak_idx + 1)
        for i in range(lo, hi):
            grad_full[i] += 2.0 * a_full[peak_idx - i]

        # Average gradient for symmetric pairs
        grad_half = np.zeros(half)
        for i in range(half):
            partner = n - 1 - i
            grad_half[i] = 0.5 * (grad_full[i] + grad_full[partner])

        grad_half -= np.mean(grad_half)
        a_half = a_half - lr * grad_half
        a_half = np.maximum(a_half, 1e-10)

        # Check score
        a_full_new = expand(a_half)
        a_full_new = a_full_new / np.sum(a_full_new)
        s = compute_score(a_full_new)
        if s < best_score:
            best_score = s

        lr *= 0.999
        if lr < 1e-5:
            lr = 0.005  # restart

    a_best = expand(a_half)
    a_best = a_best / np.sum(a_best)
    return a_best, compute_score(a_best)


def diverse_init(n, rng, method='random'):
    """Generate diverse initial sequences."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif method == 'chirp':
        t = np.arange(n, dtype=np.float64)
        freq = 0.5 + rng.random() * 2.0
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t / n)
        a = np.clip(a, 0.01, None)
    elif method == 'bell':
        t = np.arange(n, dtype=np.float64)
        center = n / 2.0
        width = n * (0.1 + 0.3 * rng.random())
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a = np.clip(a, 0.005, None)
    elif method == 'sinc':
        t = np.arange(n, dtype=np.float64) - n / 2.0
        bw = 0.3 + 0.4 * rng.random()
        a = np.sinc(bw * t / n)
        a = np.abs(a)
        a = np.clip(a, 0.005, None)
    elif method == 'two_val':
        # Random two-valued sequence
        p = 0.3 + 0.4 * rng.random()
        a = np.where(rng.random(n) < p, 1.0, 0.1)
    elif method == 'symmetric_random':
        half = (n + 1) // 2
        a_half = 1.0 + 0.3 * rng.standard_normal(half)
        a_half = np.clip(a_half, 0.01, None)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, a_half[-2::-1]])
    else:
        a = np.ones(n)

    a = np.clip(a, 1e-4, None)
    return a / np.sum(a)


def multi_top_subgrad(a0, max_time=180.0, top_m_schedule=None, tau_schedule=None):
    """
    Subgradient descent with annealed top_m and temperature.
    More robust than pure Nesterov for non-smooth objectives.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if top_m_schedule is None:
        top_m_schedule = [1, 2, 3, 5, 8, 10, 15, 20]
    if tau_schedule is None:
        tau_schedule = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]

    lr = 0.003
    mu = 0.9
    stage_time = max_time / len(top_m_schedule)

    for stage in range(len(top_m_schedule)):
        if time.monotonic() - t0 > max_time:
            break
        stage_start = time.monotonic()
        top_m = top_m_schedule[stage]
        tau = tau_schedule[stage] if stage < len(tau_schedule) else 0.001

        while time.monotonic() - stage_start < stage_time:
            # Lookahead
            y = a + mu * v
            y = np.maximum(y, 1e-12)
            y = y / np.sum(y)

            # Subgradient with softmax over top-m
            c = fft_convolve(y, y)
            c_max = np.max(c)
            shifted = (c - c_max) / tau
            exp_s = np.exp(shifted)
            weights = exp_s / np.sum(exp_s)

            # Compute gradient: grad[i] = 2 * sum_j weights[i+j] * a[j]
            grad = np.zeros(n)
            for j in range(n):
                grad += 2.0 * y[j] * weights[j:j+n]

            # Project to tangent
            grad -= np.mean(grad)

            # Update
            v = mu * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s < 1e-10:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s

            score = 2.0 * n * np.max(fft_convolve(a_new, a_new))
            a = a_new

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Adaptive learning rate
            lr = max(lr * 0.9999, 1e-5)

    return best_a, best_score


def annealed_pnorm_lbfgs(theta, n, max_time=120.0):

            return obj, np.zeros(n)

            return obj, np.zeros(n)

    best_theta = theta.copy()

            if s_cd < best_score:
                best_score = s_cd
                best_seq = a_cd.copy()
                print(f"  *** NEW BEST (CD): {best_score:.6f} ***")
            else:
                print(f"  CD best: {s_cd:.6f} (no improvement)")

    # Phase 4: Symmetric optimization
    print(f"\n=== Phase 4: Symmetric Optimization (score={best_score:.6f}) ===")
    if best_seq is not None:
        n = len(best_seq)
        # Make symmetric and optimize
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = a_sym / np.sum(a_sym)
        t_alloc = min(90.0, deadline - time.monotonic() - 120)
        if t_alloc > 15:
            a_sym_best, s_sym = symmetric_optimize(a_sym, max_time=t_alloc)
            if s_sym < best_score:
                best_score = s_sym
                best_seq = a_sym_best.copy()
                print(f"  *** NEW BEST (Sym): {best_score:.6f} ***")
            else:
                print(f"  Sym best: {s_sym:.6f} (no improvement)")

    # Phase 5: Try different lengths
    print(f"\n=== Phase 5: Length Exploration (score={best_score:.6f}) ===")
    for n_try in [192, 224, 288, 320, 384]:
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(90.0, deadline - time.monotonic() - 60)
        if t_alloc < 20:
            break

        # Interpolate best sequence to new length
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            a_interp = a_interp / np.sum(a_interp)
        else:
            a_interp = diverse_init(n_try, rng, 'chirp')

        a_best_local, s_local = nesterov_subgrad_descent(a_interp, max_time=t_alloc * 0.6)
        if s_local < best_score:
            best_score = s_local
            best_seq = a_best_local.copy()
            print(f"  n={n_try}: *** NEW BEST: {best_score:.6f} ***")
        else:
            print(f"  n={n_try}: best={s_local:.6f}")

    # Phase 6: Final polish - combined approach on best found
    print(f"\n=== Phase 6: Final Polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            # Run multi-top subgradient
            t1 = min(remaining * 0.4, 120.0)
            a_mts, s_mts = multi_top_subgrad(best_seq, max_time=t1)
            if s_mts < best_score:
                best_score = s_mts
                best_seq = a_mts.copy()
                print(f"  After multi-top: {best_score:.6f}")

            remaining = deadline - time.monotonic() - 30
            if remaining > 20:
                # Coordinate descent again
                t2 = min(remaining * 0.4, 60.0)
                a_cd2, s_cd2 = coordinate_descent_refine(best_seq, max_time=t2)
                if s_cd2 < best_score:
                    best_score = s_cd2
                    best_seq = a_cd2.copy()
                    print(f"  After CD2: {best_score:.6f}")

            remaining = deadline - time.monotonic() - 30
            if remaining > 20:
                # One more L-BFGS pass
                n = len(best_seq)
                theta = np.log(best_seq + 1e-30)
                t3 = min(remaining * 0.5, 60.0)
                a_final, s_final = annealed_pnorm_lbfgs(theta, n, max_time=t3)
                if s_final < best_score:
                    best_score = s_final
                    best_seq = a_final.copy()
                    print(f"  After final L-BFGS: {best_score:.6f}")

    # Final: return the sequence (scale to reasonable values)
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: n={len(final_list)}, evaluate_sequence={final_score:.6f} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
