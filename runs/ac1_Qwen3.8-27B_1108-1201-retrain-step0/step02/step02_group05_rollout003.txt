"""
LP-based peak reduction for minimizing max autocorrelation of non-negative sequences.
Combines LP-guided iterative refinement, L-BFGS with p-norm annealing, and local search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score for sequence: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (s ** 2)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p^p / (sum)^2."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    s = np.sum(a)
    
    # We want: 2n * (sum(c^p))^(1/p) / s^2
    # But since softmax gives sum=1, s=1 and we just minimize 2n * (sum(c^p))^(1/p)
    
    cp = c ** (p - 1)
    sum_c_pad = np.sum(cp * c)
    if sum_c_pad < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_c_pad ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_c_pad ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def softmax_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def lp_peak_step(f, M, max_iter_lp=100):
    """
    Given current f (sum=1) and peak bound M, solve LP to find g that
    maximizes sum(g) subject to conv(f,g) <= M and g >= 0.
    Returns rescaled g (sum=1) or None if LP fails.
    """
    n = len(f)
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = f[j_min:j_max+1]
    
    c_obj = -np.ones(n)
    b_vec = np.full(2*n-1, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_vec, 
                     bounds=[(0, None)]*n, method='highs',
                     options={'maxiter': max_iter_lp})
    
    if result.success:
        g = result.x
        g_sum = np.sum(g)
        if g_sum > 1e-10:
            return g / g_sum
    return None

"""
LP-guided peak reduction for minimizing max autocorrelation of non-negative sequences.
Combines LP-based direction finding, L-BFGS with p-norm/soft-max annealing,
and targeted local search. Designed to beat 1.5027 on length-256 sequences.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fft_convolve(a, a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (s ** 2)


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def pnorm_obj_grad(theta, p):
    """Objective and gradient for 2n * (sum(c^p))^(1/p) where c=conv(softmax(theta),softmax(theta))."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    cp = c ** (p - 1)
    sum_c_pad = np.sum(cp * c)
    if sum_c_pad < 1e-30:
        return 0.0, np.zeros(n)
    pnorm = sum_c_pad ** (1.0 / p)
    score = 2.0 * n * pnorm
    factor = 4.0 * n * (sum_c_pad ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp)
    grad_a = factor * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta


def softmax_obj_grad(theta, tau):
    """Objective and gradient for 2n * soft_max(conv(softmax(theta),softmax(theta)), tau)."""
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    score = 2.0 * n * soft_max_val
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return score, grad_theta

    s, _ = pnorm_obj_grad(x, p)
    return s


def _pnorm_grad(x, p):
    _, g = pnorm_obj_grad(x, p)
    return g


def _smoothax_obj(x, tau):
    s, _ = softmax_obj_grad(x, tau)
    return s


def _smoothax_grad(x, tau):
    _, g = softmax_obj_grad(x, tau)
    return g


def build_lp_matrix(f):
    """Build the (2n-1) x n constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max + 1] = f[j_min:j_max + 1]
    return A


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g/sum(g) or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'maxiter': 5000})
        if result.success:
            g = result.x
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def find_best_blend(f, g, n_trials=25):
    """Find best t for (1-t)f + tg minimizing peak."""
    best_t = 0.0
    best_peak = np.max(fft_convolve(f, f))

    for t_val in ts:
        f_new = (1 - t_val) * f + t_val * g
        peak = np.max(fft_convolve(f_new, f_new))
        if peak < best_peak:
            best_peak = peak
            best_t = t_val

            peak = np.max(fft_convolve(f_new, f_new))

def subgradient_direction(f, M, top_k=3):
    """
    Subgradient-based direction: reduce contributions to top peaks.
    Returns a direction vector (not normalized).
    """
    n = len(f)
    c = fft_convolve(f, f)
    # Get top k peaks
    peak_idxs = np.argsort(c)[-top_k:]
    # Compute negative subgradient (direction to decrease peaks)
    grad = np.zeros(n)
    for pk in peak_idxs:
        # d/d a_j of conv(f,f)_pk = 2 * sum_i f[i] * delta(pk - i - j)
        # = 2 * f_rev_conv kicked... simpler: for each j, contribution is 2*f[clamp(pk-j, 0, n-1)] if valid
        for j in range(n):
            i = pk - j
            if 0 <= i < n:
                grad[j] -= 2.0 * f[i] / top_k
    return grad

                    grad[j] += 2.0 * f[i] / len(peak_idxs)
        
        # Momentum
        momentum = beta * momentum - step * grad
        f_new = f + momentum
        f_new = np.maximum(f_new, 0)
        f_new = f_new / np.sum(f_new)
        
        new_score = local_score(f_new)
        if new_score < current_score:
            f = f_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = f.copy()
            step = min(step * 1.01, 0.1)
        else:
            step *= 0.95
            f = f_new  # keep going anyway with smaller step
    
    return best, best_score


def polish_local_search(a_init, time_limit, rng):
    """Simulated annealing with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.0008
    cooling = 0.99995
    last_print = t0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.30:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.012, size=n_inc))
        elif r < 0.50:
            # Multi-peak reduction
            c = fft_convolve(current, current)
            top_peaks = np.argsort(c)[-4:]
            for pk in top_peaks:
                contribs = np.zeros(n)
                for i in range(min(n, pk + 1)):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(2, n // 25)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.012, 0.008, size=len(top_idx)))
            low_idx = rng.choice(np.argsort(current)[:n // 8], size=min(6, n // 8), replace=False)
            candidate[low_idx] *= np.exp(rng.normal(0.012, 0.008, size=len(low_idx)))
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.80:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.maximum(candidate, 1e-12)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 5e-7)
        
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            print(f"    SA: score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: LP-guided peak reduction + L-BFGS p-norm/soft-max + SA polish.
    Targets non-negative sequences of length ~256 to minimize 2n*max(conv)/(sum)^2.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-10, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()
    
    # Phase 1: LP-based iteration on provided sequence
    print("\n=== Phase 1: LP peak reduction (primary) ===")
    if start_seq is not None:
        n = len(start_seq)
        lp_time = min(500.0, max(60.0, (deadline - t_global) * 0.22))
        print(f"  Time budget: {lp_time:.0f}s, n={n}")
        f_lp, lp_score = run_lp_phase(start_seq, lp_time)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = f_lp.copy()
            print(f"  LP improved to: {best_score:.6f} *** NEW BEST ***")
    
    # Phase 2: L-BFGS p-norm refinement
    print(f"\n=== Phase 2: L-BFGS p-norm (best={best_score:.6f}) ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        p_schedule = [4, 6, 8, 12, 16, 24, 32, 48, 64]
        lbgfs_time = min(250.0, max(30.0, (deadline - time.monotonic()) * 0.12))
        lbgfs_deadline = time.monotonic() + lbgfs_time
        maxiter = max(50, int(lbgfs_time / 2.0))
        theta_best, s = run_lbgfs_pnorm(theta, p_schedule, maxiter_per_p=maxiter // len(p_schedule),
                                        deadline=lbgfs_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  L-BFGS improved to: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  L-BFGS best: {s:.6f}")
    
    # Phase 3: Soft-max refinement
    print(f"\n=== Phase 3: Soft-max L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        theta = np.log(best_seq + 1e-30)
        tau_schedule = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
        sm_time = min(200.0, max(20.0, (deadline - time.monotonic()) * 0.10))
        sm_deadline = time.monotonic() + sm_time
        maxiter = max(40, int(sm_time / 3.0))
        theta_best, s = run_lbgfs_softmax(theta, tau_schedule, maxiter_per_tau=maxiter // len(tau_schedule),
                                          deadline=sm_deadline)
        if s < best_score:
            best_score = s
            best_seq = softmax(theta_best).copy()
            print(f"  Soft-max improved to: {best_score:.6f} *** NEW BEST ***")
        else:
            print(f"  Soft-max best: {s:.6f}")
    
    # Phase 4: Second LP round
    print(f"\n=== Phase 4: LP round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        lp_time2 = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.15))
        f_lp2, lp_score2 = run_lp_phase(best_seq, lp_time2)
        if lp_score2 < best_score:
            best_score = lp_score2
            best_seq = f_lp2.copy()
            print(f"  LP round 2: {best_score:.6f} *** NEW BEST ***")
    
    # Phase 5: Subgradient polish
    print(f"\n=== Phase 5: Subgradient polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        sg_time = min(100.0, max(10.0, (deadline - time.monotonic()) * 0.05))
        f_sg, sg_score = subgrad_polish(best_seq, sg_time, rng)
        if sg_score < best_score:
            best_score = sg_score
            best_seq = f_sg.copy()
            print(f"  Subgrad: {best_score:.6f} *** NEW BEST ***")
    
    # Phase 6: Final LP micro-round
    print(f"\n=== Phase 6: LP micro-round (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        lp_time3 = min(100.0, max(10.0, (deadline - time.monotonic()) * 0.10))
        f_lp3, lp_score3 = run_lp_phase(best_seq, lp_time3, verbose=False)
        if lp_score3 < best_score:
            best_score = lp_score3
            best_seq = f_lp3.copy()
            print(f"  LP micro: {best_score:.6f} *** NEW BEST ***")
    
    # Phase 7: SA local search
    print(f"\n=== Phase 7: SA polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = max(20.0, deadline - time.monotonic() - 5.0)
        print(f"  SA time: {polish_time:.0f}s")
        polished, p_score = polish_local_search(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  SA: {best_score:.6f} *** NEW BEST ***")
    
    # Final: one more tiny L-BFGS if time
    if best_seq is not None and time.monotonic() < deadline - 15:
        theta = np.log(best_seq + 1e-30)
        try:
            result = minimize(_smoothax_obj, theta, method='L-BFGS-B',
                              jac=_smoothax_grad, args=(0.0001,),
                              options={'maxiter': 100, 'ftol': 1e-15})
            s = local_score(softmax(result.x))
            if s < best_score:
                best_score = s
                best_seq = softmax(result.x).copy()
                print(f"  Final L-BFGS: {best_score:.6f} *** NEW BEST ***")
        except Exception:
            pass
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
