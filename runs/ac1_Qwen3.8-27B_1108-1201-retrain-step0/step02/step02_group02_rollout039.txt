   ```python
   import numpy as np
   import time
   from scipy.optimize import linprog
   from scipy.fft import rfft, irfft

   def fft_convolve(a, b):
       n, m = len(a), len(b)
       size = 1
       while size < n + m - 1:
           size <<= 1
       return irfft(rfft(a, size) * rfft(b, size), size)[:n+m-1]

   def score(a):
       n = len(a)
       S = np.sum(a)
       if S < 0.01:
           return np.inf
       c = fft_convolve(a, a)
       return 2.0 * n * np.max(c) / (S * S)

   def lp_direction(a, M):
       """Solve LP to find direction g that maximizes sum(g) subject to
       conv(a,g) <= M and g >= 0."""
       N = len(a)
       # Build constraint matrix
       # A_ub[k, j] = a[k-j] if 0 <= k-j < N, else 0
       K = 2*N - 1  # number of constraints
       A = np.zeros((K, N))
       for k in range(K):
           # g[j] appears with coefficient a[k-j] when 0 <= k-j < N
           # j ranges from max(0, k-N+1) to min(N-1, k)
           j_start = max(0, k - N + 1)
           j_end = min(N - 1, k)
           if j_start <= j_end:
               A[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
       
       # Solve LP
       res = linprog(-np.ones(N), A_ub=A, b_ub=M*np.ones(K),
                     bounds=[(0, None)]*N, method='highs')
       if res.success:
           return res.x
       return None

   def spectral_flatten(a, alpha=0.5):
       """Apply spectral flattening iteration."""
       N = len(a)
       L = 1
       while L < 2*N - 1:
           L <<= 1
       a_padded = np.zeros(L)
       a_padded[:N] = a
       A = np.fft.fft(a_padded)
       mag = np.abs(A)
       mag[mag < 1e-15] = 1e-15
       target = np.sqrt(np.sum(mag**2) / L)
       new_mag = mag ** (1 - alpha) * target ** alpha
       A_new = new_mag * np.exp(1j * np.angle(A))
       x = np.fft.ifft(A_new).real
       a_new = x[:N]
       a_new = np.maximum(0, a_new)
       s = np.sum(a_new)
       if s < 1e-10:
           return a.copy()
       return a_new / s
   ```

```python
"""
Hybrid LP + spectral flattening + subgradient optimization for minimizing
the peak of self-convolution of non-negative sequences.
Uses: (1) LP-based Frank-Wolfe-like directions, (2) Fourier-domain spectral
flattening via alternating projections, (3) projected subgradient steps,
(4) coordinate descent polish, (5) multi-restart and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = fft(a, size)
    fb = fft(b, size)
    return np.real(ifft(fa * fb, size))[:n + m - 1]


def fast_score(a):
    """Fast score computation: 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def compute_convolution(a):
    """Compute self-convolution."""
    return fft_convolve(a, a)


def lp_direction(a, M, verbose=False):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns g or None if infeasible.
    """
    N = len(a)
    K = 2 * N - 1
    # Build constraint matrix: A_ub[k, j] = a[k-j] if 0 <= k-j < N
    A_ub = np.zeros((K, N))
    for k in range(K):
        j_start = max(0, k - N + 1)
        j_end = min(N - 1, k)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            A_ub[k, indices] = a[k - indices]

    c_obj = -np.ones(N)
    b_ub = M * np.ones(K)
    bounds = [(0.0, None)] * N

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def spectral_flatten(a, alpha=0.5):
    """
    Spectral flattening via alternating projections.
    Flattens the power spectrum of the zero-padded sequence.
    """
    N = len(a)
    L = 1
    while L < 2 * N - 1:
        L <<= 1

    a_pad = np.zeros(L)
    a_pad[:N] = a
    A = fft(a_pad)
    mag = np.abs(A)
    phase = np.angle(A)

    # Avoid log of zero
    mag_safe = np.maximum(mag, 1e-15)
    target_mag = np.sqrt(np.sum(mag_safe ** 2) / L)

    # Interpolate magnitude towards flat
    new_mag = mag_safe ** (1.0 - alpha) * target_mag ** alpha
    A_new = new_mag * np.exp(1j * phase)
    x = np.real(ifft(A_new))

    # Keep only first N entries, project to non-negative
    a_new = x[:N]
    a_new = np.maximum(0.0, a_new)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def subgradient_step(a, lr):
    """
    One step of projected subgradient descent on max(conv(a,a))
    with normalization sum(a)=1, a>=0.
    """
    N = len(a)
    c = compute_convolution(a)
    k_star = int(np.argmax(c))

    # Subgradient of c[k_star] w.r.t. a[j]
    g = np.zeros(N)
    for j in range(N):
        val = 0.0
        if 0 <= k_star - j < N:
            val += a[k_star - j]
        if 0 <= j - k_star < N:
            val += a[j - k_star]
        g[j] = val

    # Update: a_new = a - lr * g, then project onto simplex
    a_raw = a - lr * g
    # Project onto {x >= 0, sum(x) = 1}
    a_proj = project_simplex(a_raw)
    return a_proj


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-10:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def coordinate_descent_polish(a_init, time_limit, rng):
    """
    Coordinate descent: for each element, try to find a better value
    via local search while keeping others fixed.
    """
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)
    current_score = fast_score(a)
    best = a.copy()
    best_score = current_score
    N = len(a)
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        order = rng.permutation(N)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-8:
                continue

            base = a[j]
            # Try multiplicative factors
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                trial = base * frac
                a[j] = trial
                s = np.sum(a)
                if s < 0.01:
                    a[j] = base
                    break
                normalized = a / s
                new_score = fast_score(normalized)
                if new_score < current_score:
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = a / np.sum(a)
                        no_improve = 0
                    a = a / s  # maintain normalization
                else:
                    a[j] = base
                    if frac > 1.0:
                        break

        if not improved:
            no_improve += 1
            # Random kick to escape
            n_kick = max(1, N // 20)
            idx = rng.choice(N, size=n_kick, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.1, size=n_kick))
            a = np.maximum(0, a)
            s = np.sum(a)
            if s > 0.01:
                a = a / s
                current_score = fast_score(a)
                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()
                    no_improve = 0

        if no_improve > 50:
            break

        if iteration % 10 == 0:
            print(f"      CD iter={iteration}, cur={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")

    return best, best_score


def make_initialization(N, rng, method='random'):
    """Create initial sequence."""
    if method == 'random':
        a = rng.uniform(0.1, 2.0, size=N)
    elif method == 'broadband':
        t = np.arange(N)
        a = np.ones(N)
        for k in range(1, 20):
            freq = k * (1.0 + 0.5 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.1 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / N + phase)
        a = np.maximum(0.01, a)
    elif method == 'structured':
        # Try a pattern that might give low autocorrelation
        a = np.ones(N)
        period = max(2, int(np.sqrt(N)))
        for i in range(N):
            if (i // period) % 2 == 0:
                a[i] = 1.0 + 0.3 * rng.random()
            else:
                a[i] = 0.3 + 0.2 * rng.random()
    else:
        a = np.ones(N)
    a = a / np.sum(a)
    return a

                    a_try = (1.0 - t) * a + t * g

                            improved_this_iter = True

        # Method 2: Spectral flattening (every iteration)

                improved_this_iter = True

            print(f"  Phase 3: no improvement (best still {best_score:.6f})")

    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining < 30:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        return final_list

    # Phase 4: Multi-length exploration
    best_score, best_seq = try_multiple_lengths(remaining, best_score, best_seq, rng, verbose=True)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with official evaluator
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_eval:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
