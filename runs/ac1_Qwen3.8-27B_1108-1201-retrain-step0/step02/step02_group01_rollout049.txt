"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent with soft-max 
peak focusing, combined with Adam refinement and coordinate descent polish.

Key ideas:
1. Multiplicative (mirror) updates on the simplex for natural positivity enforcement.
2. Soft-max temperature annealing: start with moderate T (focus on top few peaks),
   increase T over time to focus on the single maximum.
3. Periodic random perturbations to escape local minima.
4. Final coordinate descent polish for fine-grained improvements.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_fft(a, w):
    """Compute g[j] = 2 * sum_k w[k] * a[k-j] using FFT.
    
    w has length 2n-1 (convolution lags), a has length n.
    Returns g of length n.
    """
    n = len(a)
    total = len(w) + n - 1
    fsize = 1 << (total - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    r = np.real(ifft(W * np.conj(A)))
    return 2.0 * r[:n]


def softmax_weights(c, T):
    """Compute soft-max weights on the convolution lags."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c)
    scaled = T * (c - c_max)
    # scaled <= 0, max is 0
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        idx = np.argmax(c)
        w = np.zeros(len(c))
        w[idx] = 1.0
        return w
    return exp_scaled / total


def normalize_to_S(a, S):
    """Normalize a to have sum S."""
    s = a.sum()
    if s < 1e-300:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_mirror_descent(a_init, time_budget, n_top=20, T_start=20.0, T_end=500.0, eta_scale=0.5):
    """Mirror descent with soft-max peak focusing.
    
    Parameters:
    - a_init: initial sequence (will be normalized)
    - time_budget: time in seconds
    - n_top: not used directly (soft-max handles this via T)
    - T_start, T_end: temperature range
    - eta_scale: step size scale factor
    """
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_to_S(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    progress = 0.0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule: exponential increase
        T = T_start * (T_end / T_start) ** progress
        
        # Soft-max weights
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
        
        w = softmax_weights(c, T)
        
        # Subgradient
        g = compute_subgrad_fft(a, w)
        
        # Step size: inversely proportional to the max of |g|
        g_max = np.abs(g).max()
        if g_max < 1e-15:
            break
        eta = eta_scale / (T * g_max)
        
        # Mirror descent step (multiplicative)
        # a_new = a * exp(-eta * g), normalize
        log_update = -eta * g
        # Clip to avoid overflow
        log_update = np.clip(log_update, -50.0, 50.0)
        a_new = a * np.exp(log_update)
        a_new = normalize_to_S(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < cur:
            a = a_new
        else:
            # Halve the step size
            eta /= 4.0
            log_update = -eta * g
            log_update = np.clip(log_update, -50.0, 50.0)
            a_new = a * np.exp(log_update)
            a_new = normalize_to_S(a_new, S)
            c_new = fft_convolve_self(a_new)
            obj_new = c_new.max()
            if obj_new < cur:
                a = a_new
        
        # Periodic perturbation every ~2 seconds
        if int(elapsed * 0.5) != int((elapsed - 0.01) * 0.5):
            # Small random perturbation
            rng = np.random.default_rng(hash((n, int(elapsed))) & 0xFFFFFFFF)
            perturbation = rng.standard_normal(n) * 0.01
            a = np.maximum(a * (1.0 + perturbation), 0.0)
            a = normalize_to_S(a, S)
    
    return best_a, best_obj


def run_adam_logparams(a_init, time_budget, T_start=10.0, T_end=200.0, lr_start=0.05):
    """Adam optimizer on log-parameters with soft-max objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = normalize_to_S(a, S)
    theta = np.log(a / a.sum())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_theta = theta.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        T = T_start * (T_end / T_start) ** progress
        
        a = S * np.exp(theta - theta.max()) 
        a = a / a.sum() * S
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()
            best_theta = theta.copy()
        
        w = softmax_weights(c, T)
        grad_a = compute_subgrad_fft(a, w)
        
        # Chain rule for log-parameters
        dot = np.dot(grad_a, a)
        grad_theta = grad_a * a - (a / S) * dot
        
        gnorm = np.linalg.norm(grad_theta)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_start * (0.5 ** (progress * 2.0))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj


def coordinate_descent_polish(a, time_budget, max_sweeps=100):
    """Coordinate descent to polish the sequence."""
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    a = a.copy().astype(np.float64)
    a = normalize_to_S(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original <= 1e-12:
                continue
            
            # Try decreasing a[i]
            # The effect on c_k is: c_k decreases by 2*a[i]*a[k-i] (approximately)
            # We want to find the largest decrease that doesn't increase the max.
            
            # Simple approach: try a few values
            for frac in [0.9, 0.8, 0.7, 0.5, 0.3]:
                a[i] = original * frac
                s = a.sum()
                if s < 1e-10:
                    a[i] = original
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_norm.copy()
                    improved = True
                    a = a_norm
                    original = a[i]  # Update for next iteration
                    break
                else:
                    a[i] = original
            else:
                # Try increasing a[i] slightly
                a[i] = original * 1.2
                s = a.sum()
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                a[i] = original
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj
