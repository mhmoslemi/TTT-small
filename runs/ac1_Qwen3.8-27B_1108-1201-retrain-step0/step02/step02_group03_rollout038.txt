"""
Optimized search for non-negative step function minimizing peak autocorrelation ratio.
Uses smoothed-max gradient descent with Adam, multi-scale annealing, LP direction finding,
and local polish.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def local_score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))

def softmax_normalized(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def compute_grad(theta, S, tau):
    """
    Compute gradient of smooth_max(conv(a,a), tau) w.r.t. theta,
    where a = softmax(theta) * S.
    Returns (value, grad_theta, a, c).
    """
    n = len(theta)
    a = softmax_normalized(theta) * S
    c = fft_convolve(a, a)
    
    # Smooth max weights
    c_max = np.max(c)
    e = np.exp((c - c_max) / tau)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))
    
    # Gradient of value w.r.t. a:
    # grad_a[i] = sum_k w[k] * a[k-i] (for valid k-i)
    # = conv(a, w_rev)[2n-2-i] as derived
    n_fft = 1
    while n_fft < 2 * n - 1:
        n_fft <<= 1
    A = np.fft.rfft(a, n_fft)
    W_rev = np.fft.rfft(w[::-1], n_fft)
    conv_aw = np.fft.irfft(A * W_rev, n_fft)[:3*n-3]
    grad_a = conv_aw[n-1:2*n-1][::-1]
    
    # Chain rule: grad_theta[j] = a[j] * (grad_a[j] - dot(a, grad_a) / S)
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner / S)
    
    return value, grad_theta, a, c

def optimize_adam(n, a_init, time_budget, rng, tau_start=None, tau_end=None):
    """
    Adam optimization with annealed temperature.
    a_init should be the initial sequence (will be normalized).
    """
    S = np.sqrt(2.0 * n)
    a = a_init.copy()
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a) * S
    
    # Initialize theta from a
    theta = np.log(a / S + 1e-15)
    
    if tau_start is None:
        tau_start = 50.0 / n * S  # scale with S
    if tau_end is None:
        tau_end = 0.1 / n * S
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 3.0
    
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = local_score(a)
    
    # Multi-scale: divide time into phases with decreasing tau
    n_phases = 8
    phase_times = np.linspace(0, 1, n_phases + 1)
    
    for phase in range(n_phases):
        phase_start = phase_times[phase]
        phase_end = phase_times[phase + 1]
        tau_phase_start = tau_start * (tau_end / tau_start) ** phase_start
        tau_phase_end = tau_start * (tau_end / tau_start) ** phase_end
        
        while time.monotonic() - t0 < time_budget * phase_end:
            elapsed = (time.monotonic() - t0) / time_budget
            tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.5)
            
            value, grad_theta, a, c = compute_grad(theta, S, tau)
            
            t += 1
            m = beta1 * m + (1 - beta1) * grad_theta
            v = beta2 * v + (1 - beta2) * grad_theta ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            lr_t = lr / np.sqrt(1 + 0.1 * t)
            theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if t % 500 == 0:
                a_check = softmax_normalized(theta) * S
                score = local_score(a_check)
                if score < best_score:
                    best_score = score
                    best_theta = theta.copy()
    
    a_final = softmax_normalized(best_theta) * S
    score_final = local_score(a_final)
    return a_final, min(score_final, best_score)

def lp_direction(a, M, max_iter=100):
    """
    Solve LP: maximize sum(b) s.t. conv(a, b)[k] <= M for all k, b >= 0.
    Returns the optimal b (normalized).
    """
    n = len(a)
    # Build constraint matrix A_eq shape (2n-1, n)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = a[k - i]
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.success:
        b = res.x
        b = np.clip(b, 0, None)
        s = np.sum(b)
        if s > 1e-10:
            return b / s
    return None

        s = np.sum(a)
        if s < 1e-10:

        else:

                i_max = min(n - 1, lag)
                for i in range(i_min, i_max + 1):

            n_red = max(2, n // 8)
            top_idx = np.argsort(contrib)[-n_red:]
            bot_idx = np.argsort(contrib)[:n_red]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, n_red))
            candidate[bot_idx] *= np.exp(rng.normal(0.05, 0.02, n_red))

            n_blocks = rng.integers(2, 5)

                bs = int(rng.integers(0, n - bw))

        elif r < 0.8:

            candidate = current + 0.2 * (sym - current)
            candidate += 0.01 * rng.standard_normal(n)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0, 0.02, n))
        
        candidate = np.clip(candidate, 1e-10, None)
        candidate = candidate / np.sum(candidate) * S

        if new_score < current_score:
            current = candidate
            current_score = new_score

                best = current.copy()

                    current = candidate
                    current_score = new_score

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative step functions.
Uses smoothed-max gradient descent with Adam, multi-scale temperature annealing,
scipy L-BFGS refinement, peak-directed local search, and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def softmax_normalized(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def compute_value_grad(theta, S, tau):
    """
    Compute smooth_max(conv(a,a), tau) and gradient w.r.t. theta.
    a = softmax(theta) * S.
    Returns (value, grad_theta, a, c).
    """
    n = len(theta)
    a = softmax_normalized(theta) * S
    c = fft_convolve(a, a)

    # Log-sum-exp approximation of max
    c_max = np.max(c)
    e = np.exp((c - c_max) / tau)
    w = e / np.sum(e)
    value = c_max + tau * np.log(np.sum(e))

    # Gradient of value w.r.t. a_i = sum_k w_k * a[k-i] (valid k-i)
    # Computed as: conv(a, w_reversed)[n-1 : 2n-1][::-1]
    n_fft = 1
    while n_fft < 3 * n - 1:
        n_fft <<= 1
    A = np.fft.rfft(a, n_fft)
    W_rev = np.fft.rfft(w[::-1], n_fft)
    conv_aw = np.fft.irfft(A * W_rev, n_fft)[:3 * n - 3]
    grad_a = conv_aw[n - 1:2 * n - 1][::-1]

    # Chain rule through softmax * S
    inner = np.dot(a, grad_a)
    grad_theta = a * (grad_a - inner / S)

    return value, grad_theta, a, c


def optimize_adam(n, a_init, time_budget, seed, tau_mult_start=40.0, tau_mult_end=0.2):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Normalizes to sum = sqrt(2n). Returns (best_a, best_score).
    """
    S = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = np.clip(a_init, 1e-10, None)
    a = a / np.sum(a) * S
    theta = np.log(a / S + 1e-15)

    tau_start = tau_mult_start / n * S
    tau_end = tau_mult_end / n * S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 2.0

    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = local_score(a)
    patience = 0
    patience_limit = 15000

    while time.monotonic() - t0 < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        value, grad_theta, a_cur, c = compute_value_grad(theta, S, tau)

        # Gradient clipping
        gnorm = np.linalg.norm(grad_theta)
        if gnorm > 100.0:
            grad_theta = grad_theta * 100.0 / gnorm

        t += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate with warmup and decay
        warmup_steps = 2000
        if t < warmup_steps:
            lr = base_lr * t / warmup_steps
        else:
            lr = base_lr / np.sqrt(1.0 + 0.05 * (t - warmup_steps))

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic check
        if t % 2000 == 0:
            a_check = softmax_normalized(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                patience = 0
            else:
                patience += 2000

            # If stuck and still have time, perturb
            if patience > patience_limit and time.monotonic() - t0 < time_budget * 0.8:
                theta += rng.normal(0, 0.1, n)
                patience = 0

    a_final = softmax_normalized(best_theta) * S
    score_final = local_score(a_final)
    if score_final < best_score:
        best_score = score_final
        a_final = a_final.copy()
    return a_final, best_score


def optimize_lbfgs(n, a_init, time_budget, seed):
    """
    Use scipy L-BFGS-B to minimize smooth_max(conv, tau) with moderate tau.
    Returns (best_a, best_score).
    """
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 1e-10, None)
    a = a / np.sum(a) * S
    theta0 = np.log(a / S + 1e-15)

    tau = 8.0 / n * S  # moderate smoothing

    def objective(theta):
        a_tmp = softmax_normalized(theta) * S
        c = fft_convolve(a_tmp, a_tmp)
        c_max = np.max(c)
        e = np.exp((c - c_max) / tau)
        return c_max + tau * np.log(np.sum(e))

    def gradient(theta):
        _, g, _, _ = compute_value_grad(theta, S, tau)
        return g

    t0 = time.monotonic()
    max_iter = 500

    try:
        res = minimize(objective, theta0, jac=gradient, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': 20000,
                                'ftol': 1e-10, 'gtol': 1e-8})
    except Exception:
        a_result = softmax_normalized(theta0) * S
        return a_result, local_score(a_result)

    theta_res = res.x
    a_res = softmax_normalized(theta_res) * S
    score = local_score(a_res)

    # Also check with lower tau
    tau2 = 2.0 / n * S
    try:
        res2 = minimize(objective_tau2 := lambda th: compute_value_grad(th, S, tau2)[0],
                        theta_res, jac=lambda th: compute_value_grad(th, S, tau2)[1],
                        method='L-BFGS-B', options={'maxiter': 300, 'maxfun': 10000})
        a_res2 = softmax_normalized(res2.x) * S
        score2 = local_score(a_res2)
        if score2 < score:
            a_res, score = a_res2, score2
    except Exception:
        pass

    # Time check
    elapsed = time.monotonic() - t0
    if elapsed > time_budget:
        pass  # Just return what we have

    return a_res, score


def optimize_combined(n, a_init, time_budget, seed):
    """
    Two-stage optimization: L-BFGS then Adam refinement.
    """
    # Stage 1: L-BFGS (use 30% of budget)
    t1 = max(5.0, 0.3 * time_budget)
    a1, s1 = optimize_lbfgs(n, a_init, t1, seed)

    # Stage 2: Adam refinement (use remaining budget)
    t2 = max(5.0, time_budget - t1 - 2.0)
    a2, s2 = optimize_adam(n, a1, t2, seed + 1000,
                           tau_mult_start=20.0, tau_mult_end=0.1)

    if s2 < s1:
        return a2, s2
    return a1, s1

            sym = 0.5 * (current + current[::-1])
            candidate = current + 0.2 * (sym - current)
            candidate += 0.008 * rng.standard_normal(n)
        elif r < 0.85:
            # Sparsification: reduce some elements significantly
            n_sparse = max(1, n // 6)
            sparse_idx = rng.choice(n, size=n_sparse, replace=False)
            candidate[sparse_idx] *= np.exp(rng.normal(-0.2, 0.1, n_sparse))
            dense_idx = np.setdiff1d(np.arange(n), sparse_idx)
            if len(dense_idx) > 0:
                n_inc = min(8, len(dense_idx))
                inc_idx = rng.choice(dense_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.15, 0.05, n_inc))
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0.0, 0.015, n))

        candidate = np.clip(candidate, 1e-10, None)
        s_val = np.sum(candidate)
        if s_val < 1e-10:
            candidate = current.copy()
        else:
            candidate = candidate / s_val * S

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 50:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score

        temp *= cooling

    return best, best_score


def upsample_seq(a, new_n, seed):
    """Upsample a sequence to a longer length."""
    rng = np.random.default_rng(seed)
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.02 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for optimal non-negative step function.
    Uses L-BFGS + Adam hybrid optimization, multi-start with diverse
    initializations, coarse-to-fine upscaling, and local polish.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s0 = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = start_seq.copy()

    # Phase 1: Multi-start optimization at various lengths
    print("\n=== Phase 1: Multi-start optimization ===")
    lengths_to_try = [256, 384, 512, 768]

    # If provided sequence has a different length, add it
    if start_seq is not None and len(start_seq) not in lengths_to_try:
        lengths_to_try.insert(0, len(start_seq))

    total_init_budget = 0.65 * time_budget
    idx_global = 0

    for n in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break

        # Generate initializations for this length
        inits = []

        if start_seq is not None:
            if len(start_seq) == n:
                inits.append(('provided', start_seq.copy()))
            elif len(start_seq) < n:
                a_up = upsample_seq(start_seq, n, seed + n)
                inits.append(('upsampled', a_up))
            else:
                # Downsample by interpolation
                a_dn = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(start_seq)), start_seq)
                a_dn = np.clip(a_dn, 0.01, None)
                inits.append(('downsampled', a_dn / np.sum(a_dn)))

        inits.append(('fft', fft_phase_init(n, seed + 1000 + n)))
        inits.append(('chirp', chirp_init(n, seed + 2000 + n)))
        inits.append(('multitone', multi_tone_init(n, seed + 3000 + n, n_tones=4 + n // 128)))
        inits.append(('random', random_init(n, seed + 4000 + n, spread=0.4)))

        # Time allocation
        n_remaining_lengths = len(lengths_to_try) - lengths_to_try.index(n)
        n_remaining_inits = sum(
            len(start_seq) is not None and len(start_seq) == n or True
            for _ in [0]
        )
        # Simpler: allocate based on total remaining work
        remaining_init_work = 0
        for n2 in lengths_to_try[lengths_to_try.index(n):]:
            remaining_init_work += 4 + (1 if start_seq is not None and len(start_seq) == n2 else 0)
        time_per_init = max(10.0, min(150.0, total_init_budget / max(1, remaining_init_work)))
        # Scale down for larger n (slower per iteration)
        time_per_init = time_per_init * min(1.0, 384.0 / n)
        time_per_init = max(8.0, time_per_init)

        for name, init in inits:
            if time.monotonic() > deadline - 120:
                break
            idx_global += 1
            t_alloc = min(time_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 8:
                break

            print(f"  n={n}, init={name}, t_alloc={t_alloc:.0f}s")
            a_opt, score = optimize_combined(n, init, t_alloc, seed + 5000 + idx_global)
            print(f"    -> score={score:.6f}")

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}, {name}) ***")

    # Phase 2: Coarse-to-fine upscaling
    print("\n=== Phase 2: Coarse-to-fine upscaling ===")
    if best_seq is not None:
        for target_n in [512, 1024]:
            if time.monotonic() > deadline - 90:
                break
            if len(best_seq) >= target_n:
                continue
            a_up = upsample_seq(best_seq, target_n, seed + 6000)
            t_alloc = min(90.0, deadline - time.monotonic() - 60)
            if t_alloc < 15:
                break
            print(f"  Upsample to n={target_n}, t={t_alloc:.0f}s")
            a_opt, score = optimize_combined(target_n, a_up, t_alloc, seed + 6000 + target_n)
            print(f"    -> score={score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Local polish on best candidate
    print(f"\n=== Phase 3: Local polish (n={len(best_seq) if best_seq is not None else 0}) ===")
    if best_seq is not None:
        polish_time = min(200.0, max(20.0, deadline - time.monotonic() - 20))
        print(f"  Polishing with t={polish_time:.0f}s, current score={best_score:.6f}")
        polished, p_score = local_polish(best_seq, polish_time, seed + 7000, top_k=8)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
        print(f"  After polish: {best_score:.6f}")

    # Phase 4: Additional restarts if time remains
    remaining = deadline - time.monotonic() - 15
    if remaining > 40 and best_seq is not None:
        print(f"\n=== Phase 4: Additional restarts (t_remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = min(4, int(remaining // 50))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            a_new = best_seq.copy() * np.exp(rng.normal(0, 0.15, size=n_best))
            a_new = np.clip(a_new, 0.01, None)
            a_new = a_new / np.sum(a_new)
            t_alloc = min(40.0, (deadline - time.monotonic() - 15) / (n_restarts - r))
            print(f"  Restart {r+1}/{n_restarts}, t={t_alloc:.0f}s")
            a_opt, score = optimize_adam(n_best, a_new, t_alloc, seed + 8000 + r,
                                         tau_mult_start=15.0, tau_mult_end=0.15)
            print(f"    -> score={score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} ***")
                # Quick polish
                if deadline - time.monotonic() > 25:
                    p2, s2 = local_polish(best_seq, min(20.0, deadline - time.monotonic() - 10),
                                          seed + 9000 + r, top_k=6)
                    if s2 < best_score:
                        best_score = s2
                        best_seq = p2.copy()
                        print(f"    After quick polish: {best_score:.6f}")

    # Final output
    print(f"\n{'=' * 50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = local_score(np.array(final_list))
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
