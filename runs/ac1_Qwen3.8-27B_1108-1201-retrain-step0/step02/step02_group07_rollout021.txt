```python
    # (f*g)_k = sum_{j} f_j * g_{k-j}
```

```python
"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the linear programming direction-finding approach from the state of the art,
combined with adaptive step sizes, multiple restarts, and gradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_constraint_matrix(f, n):
    """Build the sparse constraint matrix for the LP.
    A[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0
    """
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()


def solve_direction_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns the optimal g.
    """
    n = len(f)
    A = build_constraint_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        g = res.x
        g = np.maximum(g, 0)
        return g
    else:
        return None


def lp_iterate(f, M, t):
    """One LP iteration: find direction g, update f.
    Returns (new_f, new_M, improved).
    """
    g = solve_direction_lp(f, M)
    if g is None:
        return f, M, False
    
    g_sum = g.sum()
    if g_sum < 1e-12:
        return f, M, False
    
    # Check if g is essentially f (no improvement possible)
    g_norm = g / g_sum
    if np.max(np.abs(g_norm - f)) < 1e-8:
        return f, M, False
    
    # Rescale g to sum = 1
    g = g / g_sum
    
    # Convex combination
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        return f, M, False
    f_new /= s
    
    new_conv = fast_convolve(f_new)
    new_M = new_conv.max()
    
    return f_new, new_M, True


def lp_optimize(f0, time_limit, t_init=0.3, verbose=False):
    """Full LP-based optimization loop."""
    f = np.maximum(f0, 0.0).copy()
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t = t_init
    patience = 0
    max_patience = 30
    iters = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        f_new, new_M, improved = lp_iterate(f, M, t)
        iters += 1
        
        if not improved:
            patience += 1
            t *= 0.7
            if t < 0.005:
                break
            continue
        
        if new_M < M - 1e-12:
            f = f_new
            M = new_M
            patience = 0
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_f = f.copy()
                if verbose:
                    print("    iter %d: M=%.8f, obj=%.8f, t=%.4f" % (iters, M, cur_obj, t))
        else:

            patience += 1
            t *= 0.8
        
        if patience >= max_patience:
            t = t_init * 0.5
            patience = 0
            if t < 0.005:
                break
    
    return best_f, best_obj


def compute_subgradient_smooth(conv, a, temperature=30.0):
    """Compute smoothed subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temperature * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-6:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def mirror_descent_polish(a0, time_limit, t_temp=50.0, eta_init=0.2):
    """Mirror descent polishing step."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t = t_temp
    eta = eta_init
    patience = 0
    max_pat = 30
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient_smooth(conv, a, t)
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t *= 1.4
            if eta < 0.003:
                break
    return best_a, best_obj


def local_search(a0, time_limit, rng):
    """Local search: perturb peaks and redistribute."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        # Find peak positions
        peak_thresh = max_c * 0.9
        peak_pos = np.where(conv >= peak_thresh)[0]
        if len(peak_pos) == 0:
            peak_pos = np.array([np.argmax(conv)])
        
        k = int(rng.choice(peak_pos))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg = a[js:je + 1]
        seg_rev = a[k - je:k - js + 1][::-1]
        products = seg * seg_rev
        total = products.sum()
        if total < 1e-15:
            continue
        probs = products / total
        # Move mass from high-contribution position to low-contribution
        source = js + int(rng.choice(je - js + 1, p=probs))
        target = int(rng.integers(n))
        
        delta = a[source] * rng.uniform(0.02, 0.1)
        if delta < 1e-10:
            continue
        a[source] -= delta
        a[target] += delta
        if a[source] < 0:
            a[source] += delta
            a[target] -= delta
            continue
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        
        if new_obj < cur_obj:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Revert
            a[source] += delta
            a[target] -= delta
    
    return best_a, best_obj
```

```python
"""
LP-direction optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal descent directions (following the state-of-the-art
LP approach), combined with mirror descent polishing, adaptive step sizes, local search,
and diverse initializations across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_lp_matrix(f, n):
    """Build sparse constraint matrix A where A[k,i] = f[k-i] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()


def solve_direction_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns optimal g vector or None on failure.
    """
    n = len(f)
    f_safe = np.maximum(f, 1e-12)
    A = build_lp_matrix(f_safe, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        g = np.maximum(res.x, 0.0)
        return g
    return None


def lp_step(f, M, t):
    """One LP step: find direction, update f. Returns (new_f, new_M, success)."""
    g = solve_direction_lp(f, M)
    if g is None:
        return f, M, False
    g_sum = g.sum()
    if g_sum < 1e-12:
        return f, M, False
    g_norm = g / g_sum
    if np.max(np.abs(g_norm - f)) < 1e-10:
        return f, M, False
    f_new = (1.0 - t) * f + t * g_norm
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        return f, M, False
    f_new /= s
    new_conv = fast_convolve(f_new)
    new_M = new_conv.max()
    return f_new, new_M, True

def compute_smoothed_subgrad(conv, a, temp=40.0):
    """Smoothed subgradient of max(conv) w.r.t. a using softmax."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    scaled = temp * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-7:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg

        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)

def local_search(a0, time_limit, rng):
    """Peak-targeted local search."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        thresh = max_c * 0.85
        peaks = np.where(conv >= thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])
        k = int(rng.choice(peaks))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg = a[js:je + 1]
        seg_rev = a[k - je:k - js + 1][::-1]
        prods = seg * seg_rev
        total = prods.sum()
        if total < 1e-15:
            continue
        probs = prods / total
        source = js + int(rng.choice(je - js + 1, p=probs))
        target = int(rng.integers(n))
        delta = a[source] * rng.uniform(0.02, 0.08)
        if delta < 1e-10:
            continue
        a[source] -= delta
        a[target] += delta
        if a[source] < 0:
            a[source] += delta
            a[target] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        if new_obj < cur_obj:
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[source] += delta
            a[target] -= delta
    return best_a, best_obj

        return float(evaluate_sequence(a_arr.tolist()))
    except Exception:
        return float('inf')


def get_parent_seq():
    """Get parent sequence from global if available."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def gen_inits(n, rng, parent):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        if g.sum() > 1e-10:
            g /= g.sum()
            inits['gauss_%s' % sf] = g
    for freq in [1, 2, 3, 4]:
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        s = np.maximum(s, 0)
        if s.sum() > 1e-10:
            s /= s.sum()
            inits['cos_%d' % freq] = s
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    # Chicane
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dirichlet_%s' % alpha] = r
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    # Spectral flatness-inspired
    target_spec = np.ones(n)
    target_spec[n // 4:3 * n // 4] = 0.7
    a_spec = np.fft.irfft(target_spec * np.exp(2j * np.pi * rng.uniform(0, 1, n) * x / n), n)
    a_spec = np.maximum(a_spec, 0)
    if a_spec.sum() > 1e-10:
        a_spec /= a_spec.sum()
        inits['spectral'] = a_spec
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main optimizer: LP-direction method with mirror descent polishing.
    Multi-phase: scan -> deep LP -> polish -> local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    parent = get_parent_seq()

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        score = safe_eval(a_norm)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_norm.copy()
            print("  *** New best: score=%.6f, n=%d ***" % (score, len(a_norm)))

    # Phase 1: Quick scan
    print("=== Phase 1: Quick scan ===")
    phase1_end = t_start + min(400.0, (deadline - t_start) * 0.18)
    scan_sizes = [128, 256, 512, 1024]
    scan_results = []

    for n in scan_sizes:
        if time.monotonic() > phase1_end - 60:
            break
        inits = gen_inits(n, rng, parent)
        n_inits = len(inits)
        time_per_init = max(3.0, (phase1_end - 60 - time.monotonic()) / max(1, n_inits))

        for name, a0 in inits.items():
            if time.monotonic() > phase1_end - 60:
                break
            t_alloc = min(time_per_init, phase1_end - 60 - time.monotonic())
            if t_alloc < 2.0:
                break
            a_lp, obj_lp = lp_optimize(a0, time_limit=t_alloc * 0.6, t_init=0.3)
            a_md, obj_md = mirror_descent(a_lp, time_limit=t_alloc * 0.4, t_temp=40.0, eta_init=0.3)
            score = safe_eval(a_md)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_md))
            record(a_md)
            record(a_lp)

    print("  Scan done. Best so far: %.6f" % best_score)
    scan_results.sort(key=lambda x: x[0])

    # Phase 2: Deep LP optimization on top candidates
    print("=== Phase 2: Deep LP optimization ===")
    phase2_end = t_start + min(1600.0, (deadline - t_start) * 0.7)
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt.copy()))
            seen.add(key)
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    n_deep = len(top_configs)
    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_end - 300:
            break
        t_alloc = min((phase2_end - 300 - time.monotonic()), 300.0)
        if t_alloc < 10.0:
            break
        print("  [%d/%d] n=%d (%s), budget=%.0fs" % (idx + 1, n_deep, n, name, t_alloc))

        # Alternating LP and mirror descent
        a_cur = a0.copy()
        num_rounds = 4
        sub_t = t_alloc / num_rounds
        for r in range(num_rounds):
            if time.monotonic() > phase2_end - 300:
                break
            rem = phase2_end - 300 - time.monotonic()
            if rem < 3:
                break
            a_cur, _ = lp_optimize(a_cur, time_limit=sub_t * 0.55, t_init=0.2,
                                   verbose=(r == 0))
            record(a_cur)
            if time.monotonic() > phase2_end - 300:
                break
            a_cur, _ = mirror_descent(a_cur, time_limit=sub_t * 0.45,
                                      t_temp=50.0 + 30.0 * r, eta_init=0.15 - 0.03 * r)
            record(a_cur)

    print("  Deep LP done. Best: %.6f" % best_score)

    # Phase 3: Extended local refinement
    print("=== Phase 3: Local refinement ===")
    phase3_end = deadline - 8.0
    if best_sequence is not None and time.monotonic() < phase3_end - 30:
        remaining = phase3_end - 8 - time.monotonic()
        n_rounds = min(6, max(2, int(remaining / 40)))
        t_per = remaining / n_rounds
        for i in range(n_rounds):
            if time.monotonic() > phase3_end - 8:
                break
            rng_local = np.random.default_rng(seed + 10000 + i)
            a_start = best_sequence.copy()
            # Slight perturbation for diversity
            if i > 0:
                a_pert = a_start + rng_local.normal(0, 0.01, len(a_start))
                a_pert = np.maximum(a_pert, 0)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    a_start = a_pert

            if i % 2 == 0:
                a_opt, _ = local_search(a_start, time_limit=t_per * 0.5, rng=rng_local)
                a_opt, _ = mirror_descent(a_opt, time_limit=t_per * 0.3,
                                          t_temp=100.0, eta_init=0.05)
            else:
                a_opt, _ = lp_optimize(a_start, time_limit=t_per * 0.5, t_init=0.1)
                a_opt, _ = local_search(a_opt, time_limit=t_per * 0.3, rng=rng_local)
            record(a_opt)

    # Phase 4: Final polish
    print("=== Phase 4: Final polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        a_best = best_sequence.copy()
        rem = deadline - 5 - time.monotonic()
        if rem > 5:
            a_opt, _ = mirror_descent(a_best, time_limit=rem * 0.5,
                                      t_temp=200.0, eta_init=0.03)
            record(a_opt)
            if time.monotonic() < deadline - 5:
                rem2 = deadline - 5 - time.monotonic()
                if rem2 > 3:
                    rng_f = np.random.default_rng(seed + 99999)
                    a_opt2, _ = local_search(a_best, time_limit=rem2, rng=rng_f)
                    record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128.0
        best_score = safe_eval(best_sequence)

    print("=== FINAL: score=%.6f, n=%d ===" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
```
