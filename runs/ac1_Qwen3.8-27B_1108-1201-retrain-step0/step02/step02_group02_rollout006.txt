def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, 0.0, 0
    c = fft_convolve(a, a)
    peak_val = np.max(c)
    peak_idx = int(np.argmax(c))
    score = 2.0 * n * peak_val / (S * S)
    return score, peak_val, peak_idx


def compute_subgradient_multi(a, n_peaks=3, tau_frac=0.0):
    """
    Compute a smoothed subgradient using the top n_peaks convolution values.
    Returns (smoothed_value, gradient, weights, conv).
    """
    n = len(a)
    c = fft_convolve(a, a)
    if tau_frac <= 0.0:
        # Exact subgradient at the max
        k_star = int(np.argmax(c))
        peak_val = c[k_star]
        g = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j < n:
                val += a[k_star - j]
            if 0 <= j - k_star < n:
                val += a[j - k_star]
            g[j] = val
        return peak_val, g, None, c
    else:
        # Soft-max over all convolution values
        c_max = np.max(c)
        tau = tau_frac * c_max
        if tau < 1e-10:
            tau = 1e-10
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / np.sum(exp_shifted)
        value = c_max + tau * np.log(np.sum(exp_shifted))
        
        # Gradient: sum_k w[k] * grad conv_k

                continue
            for j in range(n):

                    g[j] += w[k] * a[k - j]
                if 0 <= j - k < n:
                    g[j] += w[k] * a[j - k]
        return value, g, w, c

def spectral_init(n, rng):
    """
    Create an initial sequence based on spectral construction.
    Uses a combination of sinusoidal components with incommensurate frequencies

        amp = 0.1 / np.sqrt(k + 1)
        t = np.arange(n)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)

def binary_init(n, rng):

def interpolated_init(n, source_seq, rng):

    m = len(source_seq)

    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)

                    a_cd, s_cd = coordinate_descent_polish(best_seq, t_cd, rng, verbose=False)
                    s_cd_actual = compute_score(a_cd)[0]
                    if s_cd_actual < best_score:
                        best_score = s_cd_actual
                        best_seq = a_cd.copy()
    
    # === Try one alternative initialization if time remains ===
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n--- ALTERNATIVE START ---")
        t_alt = min(50.0, deadline - time.monotonic() - 30)
        if t_alt > 20:
            # Try a different initialization
            a_alt = binary_init(n_main, rng)
            a_alt2, s_alt = nesterov_subgradient(a_alt, t_alt, rng, verbose=False)
            s_alt_actual = compute_score(a_alt2)[0]
            print(f"  Alt start result: score={s_alt_actual:.6f}")
            if s_alt_actual < best_score:
                best_score = s_alt_actual
                best_seq = a_alt2.copy()
                print(f"  *** New best from alt: {best_score:.6f} ***")
    
    # Return best sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*60}")
        # Make sure our local score matches
        local = compute_score(np.array(final_list))[0]
        print(f"    local_score={local:.6f}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
