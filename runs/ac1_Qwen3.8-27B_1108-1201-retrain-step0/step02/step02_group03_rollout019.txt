"""
Optimization of non-negative step functions minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direct subgradient descent on the autocorrelation peak with equalization of
top lags, momentum, adaptive step size, and multi-scale coarse-to-fine search.
Key novelty: direct max-subgradient (not soft-max), edge-biased initializations
exploiting the structural asymmetry of non-negative autocorrelations, and
larger sequence lengths for better theoretical bounds.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_grad_top_lags(a, n_lags=3):
    """
    Compute gradient of the average of top-n_lags autocorrelation values.
    Returns (gradient, top_lags_indices, top_lags_values).
    """
    n = len(a)
    c = fft_convolve(a, a)
    k = min(n_lags, len(c))
    top_idx = np.argsort(c)[-k:]
    
    g = np.zeros(n)
    a_rev = a[::-1]
    for lag in top_idx:
        m_start = max(0, lag - n + 1)
        m_end = min(n - 1, lag)
        if m_end >= m_start:
            # g[m] += 2 * a[lag - m] for m in [m_start, m_end]
            # a[lag-m] = a_rev[n-1-(lag-m)] = a_rev[n-1-lag+m]
            idx_start = n - 1 - lag + m_start
            idx_end = n - 1 - lag + m_end
            g[m_start:m_end + 1] += 2.0 * a_rev[idx_start:idx_end + 1]
    
    # Normalize gradient by number of top lags
    g /= k
    return g, top_idx, c[top_idx]


def subgrad_optimize(a_init, time_limit, rng, n_lags=3, momentum=0.9, 
                     lr_init=0.05, lr_decay=0.9999, use_line_search=True):
    """
    Direct subgradient descent minimizing max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses momentum and optional line search.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    step = 0
    v = np.zeros(n)  # momentum
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        g, top_idx, top_vals = compute_grad_top_lags(a, n_lags)
        
        # Project gradient to tangent space of sum(a)=1
        g = g - np.mean(g)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * g
        direction = -v
        
        # Ensure direction preserves sum (remove mean)
        direction = direction - np.mean(direction)
        
        # Compute step size
        if use_line_search:
            # Backtracking line search
            norm_dir = np.sum(direction ** 2)
            if norm_dir < 1e-20:
                break
            # Initial step
            alpha = lr / np.sqrt(norm_dir + 1e-10)
            # Try up to 5 halvings
            accepted = False
            for _ in range(5):
                a_trial = a + alpha * direction
                a_trial = np.clip(a_trial, 0, None)
                s_trial = np.sum(a_trial)
                if s_trial < 1e-10:
                    alpha *= 0.5
                    continue
                a_trial = a_trial / s_trial
                # Check if max conv decreased
                c_trial = fft_convolve(a_trial, a_trial)
                c_curr = fft_convolve(a, a)
                if np.max(c_trial) < np.max(c_curr):
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                alpha = 0.0
        else:
            alpha = lr
        
        # Update
        a_new = a + alpha * direction
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        # Check improvement
        if step % 50 == 0:
            sc = local_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        a = a_new
        step += 1
        lr *= lr_decay
    
    # Final check
    sc = local_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def equalization_optimize(a_init, time_limit, rng, n_lags=5):
    """
    Equalization approach: try to make the top-n_lags autocorrelation values equal.
    Uses Newton-like updates on the "excess" at each top lag.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    step = 0
    lr = 0.02
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        k = min(n_lags, len(c))
        top_idx = np.argsort(c)[-k:]
        top_vals = c[top_idx]
        target = np.mean(top_vals)
        
        # Compute "excess" at each top lag
        excess = top_vals - target  # These should be driven to 0
        
        # Combined gradient: sum of excess[k] * grad(c[k])
        g = np.zeros(n)
        a_rev = a[::-1]
        for j, lag in enumerate(top_idx):
            m_start = max(0, lag - n + 1)
            m_end = min(n - 1, lag)
            if m_end >= m_start:
                idx_start = n - 1 - lag + m_start
                idx_end = n - 1 - lag + m_end
                g[m_start:m_end + 1] += 2.0 * excess[j] * a_rev[idx_start:idx_end + 1]
        
        # Project to tangent space
        g = g - np.mean(g)
        
        # Update
        a_new = a - lr * g
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        if step % 100 == 0:
            sc = local_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        a = a_new
        step += 1
        if step % 200 == 0:
            lr *= 0.95
    
    sc = local_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def soft_max_optimize(a_init, time_limit, rng, tau_decay=0.9995, lr=0.1):
    """
    Soft-max based gradient descent (improved from previous approach).
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = local_score(a)
    tau = max(1.0 / n, 1e-6)
    
    t0 = time.monotonic()
    step = 0
    v1 = np.zeros(n)
    v2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        
        # Soft-max
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        e = np.exp(shifted)
        w = e / np.sum(e)
        
        # Gradient of soft-max(conv(a,a)) w.r.t. a
        # = 2 * correlation(a, w) appropriately indexed
        g = np.zeros(n)
        a_rev = a[::-1]
        for lag in range(len(w)):
            if w[lag] > 1e-10:
                m_start = max(0, lag - n + 1)
                m_end = min(n - 1, lag)
                if m_end >= m_start:
                    idx_start = n - 1 - lag + m_start
                    idx_end = n - 1 - lag + m_end
                    g[m_start:m_end + 1] += 2.0 * w[lag] * a_rev[idx_start:idx_end + 1]
        
        # Project to tangent space
        g = g - np.mean(g)
        
        # Adam update
        step += 1
        v1 = b1 * v1 + (1 - b1) * g
        v2 = b2 * v2 + (1 - b2) * (g ** 2)
        v1_hat = v1 / (1 - b1 ** step)
        v2_hat = v2 / (1 - b2 ** step)
        step_size = lr / (np.sqrt(v2_hat) + eps)
        
        a_new = a - step_size * v1_hat
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        if step % 100 == 0:
            sc = local_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        a = a_new
        tau *= tau_decay
    
    sc = local_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def init_fft_phase(n, rng, n_iter=100):
    """Gerchberg-Saxton style: flatten magnitude spectrum."""
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-12] = 1e-12
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def init_edge_biased(n, rng, beta=1.0):
    """Initialize with edge-biased shape (large at edges, small at center)."""

    a = 1.0 + beta * (d ** 1.5)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)

def upsample(a, new_n, rng, noise_level=0.02):
    """Upsample a sequence to a longer length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += noise_level * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)

                for i in range(m_start, m_end + 1):
                    j = lag - i
                    contrib[i] += current[i] * current[j]
            n_reduce = max(2, n // 12)
            top_idx = np.argsort(contrib)[-n_reduce:]
            bot_idx = np.argsort(contrib)[:n_reduce]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.02, size=len(top_idx)))
            candidate[bot_idx] *= np.exp(rng.normal(0.04, 0.02, size=len(bot_idx)))
        elif r < 0.6:
            # Block swap: reduce one block, increase another
            width = max(1, int(rng.integers(1, max(2, n // 25))))
            start1 = int(rng.integers(0, n - width))
            start2 = int(rng.integers(0, n - width))
            candidate[start1:start1 + width] *= np.exp(rng.normal(-0.06, 0.02, size=width))
            candidate[start2:start2 + width] *= np.exp(rng.normal(0.06, 0.02, size=width))
        elif r < 0.75:
            # Multi-block perturbation
            n_blocks = int(rng.integers(2, 5))
            for _ in range(n_blocks):
                bw = max(1, int(rng.integers(1, max(2, n // 15))))
                bs = int(rng.integers(0, n - bw))
                candidate[bs:bs + bw] *= np.exp(rng.normal(0.0, 0.04, size=bw))
        elif r < 0.85:
            # Edge emphasis: boost edges, reduce center
            n_edge = max(1, n // 10)
            edge_scale = np.exp(rng.normal(0.02, 0.01))
            center_scale = np.exp(rng.normal(-0.02, 0.01))
            candidate[:n_edge] *= edge_scale
            candidate[-n_edge:] *= edge_scale
            candidate[n_edge:n - n_edge] *= center_scale
        else:
            # Global noise
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            candidate = current
        else:
            candidate = candidate / s * np.sum(current)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 10:
                accept_prob = np.exp(-delta / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search minimizing 2*n*max(conv)/sum^2.
    Phase 1: Diverse initialization (FFT, edge-biased, random, dome, Barker)
    Phase 2: Subgradient + equalization + soft-max optimization
    Phase 3: Coarse-to-fine upscaling
    Phase 4: Local search polish
    Phase 5: Final verification
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check for provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Phase 1: Generate diverse initial candidates
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    
    # Try multiple lengths - larger n gives better theoretical bound
    lengths_to_try = [256, 512, 1024, 2048, 4096]
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        t_n_start = time.monotonic()
        
        # Generate multiple inits for this length
        inits = []
        
        a1 = init_fft_phase(n, rng, n_iter=80)
        inits.append(('fft', a1))
        
        a2 = init_edge_biased(n, rng, beta=0.5)
        inits.append(('edge0.5', a2))
        
        a3 = init_edge_biased(n, rng, beta=1.0)
        inits.append(('edge1.0', a3))
        
        a4 = init_dome(n, rng, power=2.0)
        inits.append(('dome', a4))
        
        a5 = init_random(n, rng, spread=0.3)
        inits.append(('random', a5))
        
        a6 = init_barker_modulated(n, rng)
        inits.append(('barker', a6))
        
        for name, a in inits:
            score = local_score(a)
            print(f"  n={n}, {name}: score={score:.6f}")
            candidates.append((score, n, a.copy(), name))
        
        print(f"  [n={n} done in {time.monotonic()-t_n_start:.1f}s]")
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{c[0]:.4f}', c[1], c[3]) for c in candidates[:5]]}")
    
    # Phase 2: Optimize top candidates with multiple methods
    print("\n=== Phase 2: Optimization ===")
    n_top = min(6, len(candidates))
    
    for i in range(n_top):
        if time.monotonic() > deadline - 120:
            break
        score0, n, a, name = candidates[i]
        t_alloc = min(60.0, max(10.0, (deadline - time.monotonic() - 120) / (n_top - i)))
        # Scale for larger n
        t_alloc *= min(1.0, 1024.0 / max(n, 1))
        t_alloc = max(5.0, t_alloc)
        
        print(f"  [{i+1}/{n_top}] n={n}, {name}, score={score0:.6f}, t={t_alloc:.0f}s")
        
        # Method 1: Subgradient descent
        t_sub = min(t_alloc * 0.4, max(5.0, t_alloc * 0.4))
        a_sub, s_sub = subgrad_optimize(a, t_sub, rng, n_lags=3, 
                                          momentum=0.9, lr_init=0.03,
                                          lr_decay=0.9999)
        print(f"    subgrad: {s_sub:.6f}")
        if s_sub < best_score:
            best_score = s_sub
            best_seq = a_sub.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        
        # Method 2: Equalization (from subgrad result)
        if time.monotonic() < deadline - 60:
            t_eq = min(t_alloc * 0.3, max(5.0, t_alloc * 0.3))
            a_eq, s_eq = equalization_optimize(a_sub, t_eq, rng, n_lags=5)
            print(f"    equalize: {s_eq:.6f}")
            if s_eq < best_score:
                best_score = s_eq
                best_seq = a_eq.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        
        # Method 3: Soft-max (from equalization result)
        if time.monotonic() < deadline - 60:
            t_sm = min(t_alloc * 0.3, max(5.0, t_alloc * 0.3))
            a_sm, s_sm = soft_max_optimize(a_eq, t_sm, rng)
            print(f"    softmax: {s_sm:.6f}")
            if s_sm < best_score:
                best_score = s_sm
                best_seq = a_sm.copy()
                print(f"    *** New best: {best_score:.6f} (n={n}) ***")
    
    # Phase 3: Coarse-to-fine
    print(f"\n=== Phase 3: Coarse-to-fine ===")
    if best_seq is not None:
        # Try upscaling the best sequence
        current_n = len(best_seq)
        target_ns = [x for x in [512, 1024, 2048, 4096, 8192] if x > current_n]
        
        for target_n in target_ns:
            if time.monotonic() > deadline - 90:
                break
            a_up = upsample(best_seq, target_n, rng, noise_level=0.01)
            t_alloc = min(45.0, max(10.0, (deadline - time.monotonic() - 90)))
            t_alloc *= min(1.0, 1024.0 / target_n)
            t_alloc = max(5.0, t_alloc)
            
            print(f"  Upsample {current_n} -> {target_n}, t={t_alloc:.0f}s")
            
            a_opt, s_opt = subgrad_optimize(a_up, t_alloc * 0.5, rng, n_lags=3,
                                             momentum=0.9, lr_init=0.02,
                                             lr_decay=0.9999)
            if time.monotonic() < deadline - 30:
                a_opt2, s_opt2 = equalization_optimize(a_opt, t_alloc * 0.3, rng, n_lags=5)
                s_opt = min(s_opt, s_opt2)
                if s_opt2 < s_opt:
                    a_opt = a_opt2
            
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")
                current_n = target_n
    
    # Phase 4: Local search polish
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None:
        polish_time = min(180.0, max(15.0, (deadline - time.monotonic() - 30)))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, time={polish_time:.0f}s")
        polished, p_score = local_search(best_seq, polish_time, rng, top_k=5)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
    
    # Phase 5: Additional targeted optimization with remaining time
    remaining = deadline - time.monotonic() - 20
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 5: Final optimization (time={remaining:.0f}s) ===")
        n_best = len(best_seq)
        
        # Try a few random perturbations + optimize
        n_attempts = min(3, int(remaining // 30))
        for att in range(n_attempts):
            if time.monotonic() > deadline - 25:
                break
            a_pert = best_seq * np.exp(rng.normal(0, 0.05, size=n_best))
            a_pert = np.clip(a_pert, 0.01, None)
            a_pert = a_pert / np.sum(a_pert)
            
            t_alloc = min(25.0, (deadline - time.monotonic() - 25) / max(1, n_attempts - att))
            a_opt, s_opt = subgrad_optimize(a_pert, t_alloc * 0.6, rng, n_lags=3,
                                             momentum=0.9, lr_init=0.02,
                                             lr_decay=0.9999)
            if time.monotonic() < deadline - 20:
                a_opt2, s_opt2 = equalization_optimize(a_opt, t_alloc * 0.3, rng, n_lags=4)
                if s_opt2 < s_opt:
                    s_opt = s_opt2
                    a_opt = a_opt2
            
            print(f"  Attempt {att+1}: score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # Final verification
    print(f"\n{'='*60}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        local_check = local_score(final_list)
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: local_score={local_check:.6f}, evaluate_sequence={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
