**Key enhancement: "Peak shaving" as a separate step**

After each batch of gradient steps, I'll:
1. Find the current peak location k*
2. Compute the contribution of each a[i] to (a*a)[k*]: contrib[i] = a[k*-i] (for valid i)
3. Reduce the top-m contributors by a small amount
4. Add the removed mass to the bottom-m contributors (or to random positions)
5. Check if the score improved

This is a form of "water-filling" that directly targets the peak.

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    value = max_c + tau * np.log(np.sum(exp_vals))
    # Gradient: d/d a[j] of sum_k w_k * (a*a)[k]
    # (a*a)[k] = sum_i a[i] * a[k-i]
    # d/d a[j] = sum_k w_k * (a[k-j] + a[j-k]) for valid indices
    # = 2 * sum_k w_k * a[k-j] (for valid k-j)
    # This is the cross-correlation of w and a, evaluated at lag j
    size = 2 * n
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    # cross-correlation: corr[j] = sum_k w[k] * a[k-j]
    # In FFT: IFFT(W * conj(A))[j]
    corr = np.fft.irfft(fw * np.conj(fa), size)[:n]
    grad = 2.0 * corr
    return value, grad

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def init_hann(n, rng, noise_level=0.01):

    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)

    if tau_start is None:
        tau_start = 1.0 / n
    if tau_end is None:

    theta = np.log(a_init / np.sum(a_init) + 1e-15)

    m = np.zeros(n)

    best_score = np.inf

    a_best = softmax_normalize(best_theta) * S
    return a_best, best_score

def peak_shave(a, rng, n_shave=10):
    """Directly reduce the peak by lowering top contributors and raising bottom ones."""
    n = len(a)
    conv = fft_convolve(a, a)
    peak_idx = int(np.argmax(conv))
    
    # Contributors to conv[peak_idx]: a[i] * a[peak_idx - i] for valid i
    valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
    contrib = np.zeros(n)
    for i in valid_i:
        j = peak_idx - i
        contrib[i] = a[i] * a[j]
    
    # Reduce top contributors, increase bottom contributors
    sorted_idx = np.argsort(contrib)
    top = sorted_idx[-n_shave:]
    bottom = sorted_idx[:n_shave]
    
    amt = 0.01 * np.mean(a[top])
    a[top] -= amt
    a[bottom] += amt * len(top) / len(bottom)
    a = np.maximum(a, 0.0)
    
    return a

def coordinate_descent(a_init, time_limit, rng):

    a = a_init.copy()
