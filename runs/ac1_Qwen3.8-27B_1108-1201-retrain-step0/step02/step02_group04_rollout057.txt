def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_and_conv(a):
    """Compute score and convolution for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf, None
    c = fft_convolve(a, a)
    max_c = np.max(c)
    score = 2.0 * n * max_c / (sum_a ** 2)
    return score, c


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We need x = max(v - tau, 0) with sum(x) = target_sum
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find the largest rho such that sum_{i<=rho} (u[i] - tau) = target_sum
    # where tau = (sum_{i<=rho} u[i] - target_sum) / rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - target_sum) / rho
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - target_sum) / rho_max
    return np.maximum(v - tau, 0.0)


def subgradient(a, c, active_lags):
    """Compute subgradient of max(c) w.r.t. a.
    c[k] = sum_i a[i]*a[k-i], dc[k]/da[j] = 2*a[k-j] if valid.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(active_lags)
    return grad


def subgradient_fast(a, c, active_lags):
    """Faster subgradient computation using the structure of the convolution."""
    n = len(a)
    # For each active lag k, the gradient contribution to position j is 2*a[k-j]
    # This is basically 2 * (a shifted) for each active lag.
    grad = np.zeros(2 * n - 1)
    for k in active_lags:
        # The gradient at position j gets contribution 2*a[k-j] for j where 0<=k-j<n
        # This is equivalent to: for j in [max(0,k-n+1), min(n-1,k)], grad[j] += 2*a[k-j]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(active_lags)
    return grad[:n]

def compute_subgrad(a, active_lags):
    """Compute the subgradient efficiently."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        # g[j] = 2*a[k-j] for j where 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        # a[k-j] for j in that range = a[k-hi : k-lo+1] reversed
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a[k-hi], a[k-hi+1], ..., a[k-lo]
            grad[lo:hi+1] += 2.0 * a[k - hi : k - lo + 1][::-1]
    grad /= len(active_lags)
    return grad


def optimize_subgrad(a_init, time_budget, rng, verbose=True):
    """Projected subgradient descent on the simplex."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    
    # Normalize to sum = 1
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best = a.copy()
    best_score, _ = score_and_conv(a)
    t0 = time.monotonic()
    
    lr = 0.1  # Initial learning rate
    step = 0
    patience = 0
    best_patience_score = best_score
    
    while time.monotonic() - t0 < time_budget:
        # Compute convolution
        c = fft_convolve(a, a)
        max_c = np.max(c)
        current_score = 2.0 * n * max_c  # since sum(a) = 1
        
        # Find active lags (within 1% of max)
        threshold = max_c * 0.99
        active_lags = np.where(c >= threshold)[0]
        
        if len(active_lags) == 0:
            active_lags = np.where(c >= max_c)[0]
        
        # Compute subgradient
        grad = compute_subgrad(a, active_lags)
        
        # Project gradient onto the tangent space of the simplex
        # The tangent space is {v : sum(v) = 0}
        grad_tangent = grad - np.mean(grad)
        
        # Take a step
        a_new = a - lr * grad_tangent
        a_new = project_simplex(a_new, 1.0)
        
        # Evaluate
        new_score, _ = score_and_conv(a_new)
        
        if np.isfinite(new_score) and new_score < current_score:
            a = a_new
            patience = 0
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                best_patience_score = best_score
        else:
            patience += 1
            # Reduce learning rate
            if patience > 10:
                lr *= 0.5
                patience = 0
        
        step += 1
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      subgrad step={step}, score={2*n*max_c:.8f}, best={best_score:.8f}, lr={lr:.6f}, time={elapsed:.1f}s")
        
        # Early stopping if no improvement for a long time
        if step % 1000 == 0 and patience > 50:
            break
    
    return best, best_score

Uses smoothed gradient descent (log-sum-exp) on the probability simplex

import numpy as np
import time


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    return fft_convolve(a, a)


def score_from_conv(c, n, sum_a):
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - 1.0) / rho_max
    return np.maximum(v - tau, 0.0)

    Objective: 2n * soft_max(conv(a,a), tau) / (sum(a))^2

    n = len(a)
    
    # Normalize
    s = np.sum(a)
    a = a / s
    
    best = a.copy()
    best_score = 2.0 * n * np.max(compute_conv(a))  # sum = 1
    
    t0 = time.monotonic()
    lr = 0.01
    momentum = 0.9
    vel = np.zeros(n)
    tau_start = 0.1
    tau_end = 0.001
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        # Anneal tau
        progress = min(1.0, step / 5000)
        tau = tau_start * (tau_end / tau_start) ** progress

        grad_tan = grad - np.mean(grad)
        
        # Momentum step
        vel = momentum * vel - lr * grad_tan
        a_new = a + vel
        a_new = project_simplex(a_new)
        
        # Check improvement (using actual score, not smoothed)
        actual_score = 2.0 * n * np.max(compute_conv(a_new))
        
        if actual_score < best_score:
            best = a_new.copy()
            best_score = actual_score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.5
            momentum = 0.8
            no_improve = 0
        
        step += 1
        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        GD step={step}, tau={tau:.5f}, score={actual_score:.8f}, best={best_score:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")

        if no_improve > 200:
            break

Uses smoothed gradient descent (log-sum-exp) on the probability simplex with momentum,
followed by exact coordinate descent polishing. Multiple initializations and sequence
lengths are explored to avoid local minima.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    """Compute auto-convolution."""
    return fft_convolve(a, a)


def score_exact(a):
    """Exact score computation (scale-invariant)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = compute_conv(a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - 1.0) / rho_max
    return np.maximum(v - tau, 0.0)


def smooth_score_grad(a, tau):
    """Compute smoothed score and its gradient on simplex (sum(a)=1 assumed).
    Score = 2n * soft_max(conv(a,a), tau)
    """
    n = len(a)
    c = compute_conv(a)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(c_shifted / tau)
    w = exp_c / np.sum(exp_c)
    soft_max_val = c_max + tau * np.log(np.sum(exp_c))
    score = 2.0 * n * soft_max_val

    # Gradient: d(soft_max)/da[j] = sum_k w[k] * 2*a[k-j] (if valid)
    # = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, w[j:j + n])

    return score, grad, c, w


def gradient_descent_smoothed(a_init, time_budget, verbose=True):
    """Smoothed gradient descent on the probability simplex with momentum."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best = a.copy()
    best_c = compute_conv(a)
    best_score = 2.0 * n * np.max(best_c)

    t0 = time.monotonic()
    lr = 0.05
    momentum = 0.9
    vel = np.zeros(n)
    tau_start = 0.05
    tau_end = 0.0005
    total_steps_planned = 10000
    step = 0
    no_improve = 0
    min_improve = 1e-10

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, step / total_steps_planned)
        tau = tau_start * (tau_end / tau_start) ** progress

        score_sm, grad, c, w = smooth_score_grad(a, tau)

        # Project gradient to tangent space (sum = 0)
        grad_tan = grad - np.mean(grad)

        # Momentum update
        vel = momentum * vel - lr * grad_tan
        a_new = a + vel
        a_new = project_simplex(a_new)

        # Check with exact score
        c_new = compute_conv(a_new)
        actual_score = 2.0 * n * np.max(c_new)

        if actual_score < best_score - min_improve:
            best = a_new.copy()
            best_score = actual_score
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        step += 1

        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.7
            momentum = min(0.95, momentum + 0.05)
            no_improve = 0

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"          GD step={step}, tau={tau:.6f}, score={actual_score:.8f}, "
                  f"best={best_score:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")

        if no_improve > 300:
            break

    return best, best_score


def coordinate_descent(a_init, time_budget, verbose=True):
    """Exact coordinate descent on the probability simplex."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best = a.copy()
    c = compute_conv(a)
    best_score = 2.0 * n * np.max(c)
    t0 = time.monotonic()

    sweep = 0
    stalled = 0

    while time.monotonic() - t0 < time_budget:
        improved = False
        a_sum = 1.0  # maintained as sum(a) = 1

        for j in range(n):
            if time.monotonic() - t0 > time_budget:
                break

            a_old = a[j]

            # Precompute: for each k, the coefficient of u and u^2 in c_new[k]
            # c_new[k] = c[k] + 2*u*a[k-j]*1_{valid} + u^2*1_{k=2j}
            # where u = x - a_old, x is new value of a[j]

            # Build the linear and quadratic coefficients
            lin = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                idx = k - j
                if 0 <= idx < n:
                    lin[k] = 2.0 * a[idx]

            quad_coeff = np.zeros(2 * n - 1)
            if 2 * j < 2 * n - 1:
                quad_coeff[2 * j] = 1.0

            # Try candidate u values (x = a_old + u, x >= 0 => u >= -a_old)
            # x ranges over [0, 1000], but realistically [0, ~5/n] for good scores
            max_x = max(5.0 / n, a_old * 3.0 + 1.0 / n)
            candidates_x = np.concatenate([
                [0.0],
                np.linspace(0.0, max_x, 30),
                [a_old]
            ])
            candidates_x = np.unique(np.clip(candidates_x, 0.0, 1000.0))

            best_x = a_old
            best_x_score = 2.0 * n * np.max(c)  # current score (sum=1)

            for x in candidates_x:
                u = x - a_old
                new_sum = a_sum + u
                if new_sum < 1e-10:
                    continue
                c_new = c + u * lin + (u ** 2) * quad_coeff
                score_x = 2.0 * n * np.max(c_new) / (new_sum ** 2)
                # But we want to maintain sum=1, so we need to rescale
                # Actually, since score is scale-invariant, we can just use:
                # score = 2n * max(c_new) / new_sum^2
                # This is the correct scale-invariant score.
                if score_x < best_x_score:
                    best_x_score = score_x
                    best_x = x

            # Apply update
            if best_x != a_old:
                u = best_x - a_old
                a[j] = best_x
                c = c + u * lin + (u ** 2) * quad_coeff
                a_sum += u
                # Renormalize to sum = 1
                a = a / a_sum
                c = c / (a_sum ** 2)
                a_sum = 1.0
                improved = True

                cur_score = 2.0 * n * np.max(c)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

        sweep += 1
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"          CD sweep={sweep}, best_score={best_score:.8f}, t={elapsed:.1f}s")

        if not improved:
            stalled += 1
            if stalled > 3:
                break
        else:
            stalled = 0

    return best, best_score


def random_perturb_init(n, rng, alpha=0.3):
    """Create a random perturbation of the uniform sequence."""
    a = np.ones(n)
    noise = rng.standard_normal(n) * alpha
    a = a + noise
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    return a


def structured_init(n, rng, method='gaussian'):
    """Create structured initializations."""
    if method == 'gaussian':
        x = np.arange(n) - (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
    elif method == 'exponential':
        x = np.arange(n)
        a = np.exp(-x / (n / 3.0))
    elif method == 'bimodal':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    elif method == 'random_positive':
        a = rng.random(n) + 0.1
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main optimization driver. Tries multiple strategies:
    1. Optimize the provided sequence (if available)
    2. Try structured initializations
    3. Try random perturbations
    4. Coordinate descent polishing
    Returns the best sequence found.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(seq, score):
        nonlocal best_score, best_seq
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    NEW BEST: score={score:.8f}, n={len(seq)}, time_left={deadline - time.monotonic():.1f}s")

    # --- Phase 0: Initial evaluations ---
    print("=== Phase 0: Initial evaluation ===")

    # Check provided sequence
    a_start = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a_start = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = score_exact(a_start)
        print(f"  Provided sequence: n={len(a_start)}, score={s:.8f}")
        if np.isfinite(s) and s < best_score:
            update_best(a_start, s)

    # Try uniform sequences of various lengths
    for n in [128, 256, 512, 1024]:
        uni = np.ones(n)
        s = score_exact(uni)
        print(f"  Uniform n={n}: score={s:.8f}")
        if s < best_score:
            update_best(uni, s)

    # --- Phase 1: Gradient descent on provided/best sequence ---
    print("\n=== Phase 1: Gradient descent ===")

    # Optimize the best found so far
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        time_alloc = min(600.0, (deadline - time.monotonic()) * 0.4)
        print(f"  Optimizing n={n}, budget={time_alloc:.0f}s")
        result, result_score = gradient_descent_smoothed(best_seq, time_alloc, verbose=True)
        update_best(result, result_score)

    # Try structured initializations
    for n in [128, 256, 512]:
        for method in ['gaussian', 'bimodal', 'exponential']:
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng, method)
            # Quick gradient descent
            time_alloc = min(120.0, (deadline - time.monotonic()) * 0.2)
            if time_alloc < 10:
                continue
            result, result_score = gradient_descent_smoothed(a_init, time_alloc, verbose=False)
            s = score_exact(result)
            if s < best_score:
                update_best(result, s)
            else:
                elapsed_info = f"  Structured {method} n={n}: score={s:.8f} (not best)"
                print(elapsed_info)

    # Try random perturbations
    for n in [128, 256, 512]:
        for attempt in range(3):
            if time.monotonic() > deadline - 30:
                break
            a_init = random_perturb_init(n, rng, alpha=0.5)
            time_alloc = min(120.0, (deadline - time.monotonic()) * 0.15)
            if time_alloc < 10:
                continue
            result, result_score = gradient_descent_smoothed(a_init, time_alloc, verbose=False)
            s = score_exact(result)
            if s < best_score:
                update_best(result, s)
            else:
                print(f"  Random n={n} attempt={attempt}: score={s:.8f} (not best)")

    # --- Phase 2: Coordinate descent polishing ---
    print(f"\n=== Phase 2: Coordinate descent (best so far: {best_score:.8f}) ===")

    if best_seq is not None and time.monotonic() < deadline - 30:
        time_alloc = min(900.0, (deadline - time.monotonic()) * 0.5)
        print(f"  CD on best sequence, budget={time_alloc:.0f}s")
        result, result_score = coordinate_descent(best_seq, time_alloc, verbose=True)
        update_best(result, result_score)

        # Second round of CD after another GD pass
        if time.monotonic() < deadline - 60:
            time_gd = min(300.0, (deadline - time.monotonic()) * 0.3)
            if time_gd > 30:
                result2, s2 = gradient_descent_smoothed(result, time_gd, verbose=False)
                update_best(result2, s2)
                if time.monotonic() < deadline - 30:
                    time_cd2 = min(300.0, (deadline - time.monotonic()) * 0.5)
                    result3, s3 = coordinate_descent(result2, time_cd2, verbose=True)
                    update_best(result3, s3)

    # --- Phase 3: More random restarts if time allows ---
    print(f"\n=== Phase 3: Additional search (best so far: {best_score:.8f}) ===")

    attempts = 0
    while time.monotonic() < deadline - 20:
        attempts += 1
        n = rng.choice([128, 256, 384, 512])
        alpha = rng.uniform(0.3, 1.0)
        a_init = random_perturb_init(n, rng, alpha=alpha)

        time_gd = min(200.0, (deadline - time.monotonic()) * 0.3)
        if time_gd < 20:
            break
        result, result_score = gradient_descent_smoothed(a_init, time_gd, verbose=False)
        s = score_exact(result)

        if s < best_score:
            update_best(result, s)
            # Polish with CD
            if time.monotonic() < deadline - 30:
                time_cd = min(300.0, (deadline - time.monotonic()) * 0.4)
                result2, s2 = coordinate_descent(result, time_cd, verbose=False)
                update_best(result2, s2)
        else:
            print(f"  Restart {attempts} n={n} alpha={alpha:.2f}: score={s:.8f} (not best)")

    # --- Final validation ---
    print(f"\n=== FINAL: best_score={best_score:.8f} ===")

    if best_seq is None:
        best_seq = np.ones(256)

    # Ensure all values are valid
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_list = [float(x) for x in best_seq]

    # Validate with evaluate_sequence
    try:
        final_score = evaluate_sequence(final_list)
        print(f"  evaluate_sequence confirms: {final_score:.8f}")
        if not np.isfinite(final_score):
            # Fallback to uniform
            final_list = [1.0] * 256
            print("  Fallback to uniform!")
    except Exception as e:
        print(f"  Validation error: {e}")

    return final_list
