def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)

def subgradient_at_peak(a):

    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):

    return peak_val, g, k_star


def project_simplex(v, S):
    """Project v onto {x : x >= 0, sum(x) = S}."""
    n = len(v)
    # Standard algorithm (Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        # Simplify: g_theta = a/S * (g_a - S*inner/S)... 
        # Let me redo this.
        # L = value(a) where a = softmax(theta) * S
        # dL/dtheta_j = sum_i (dL/da_i)(da_i/dtheta_j)
        # da_i/dtheta_j = S * softmax_j * (delta_ij - softmax_i) = a_j * (delta_ij - a_i/S)
        # So dL/dtheta_j = sum_i g_a_i * a_j * (delta_ij - a_i/S)
        # = g_a_j * a_j - a_j * (sum_i g_a_i * a_i / S)
        # = a_j * (g_a_j - (sum_i g_a_i * a_i) / S)
        
        weighted_sum = np.dot(g_a, a)
        g_theta = a * (g_a - weighted_sum / S_target)
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        # Learning rate schedule
        base_lr = 1.0 / np.sqrt(step + 1)
        if elapsed_frac < 0.2:
            lr = base_lr * 2.0
        elif elapsed_frac < 0.6:
            lr = base_lr
        else:
            lr = base_lr * 0.3
        
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if step % 100 == 0:
            a_check = softmax_normalize(theta) * S_target
            score = local_score(a_check)
            if score < best_score:

                best_theta = theta.copy()
                no_improve = 0
            else:

            # Escape mechanism
            if no_improve > 2000:

                if r < 0.3:

                elif r < 0.6:

                    c_cur = fft_convolve(a_check, a_check)
                    peak_idx = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= peak_idx - i < n:
                            contribs[i] = a_check[i] * a_check[peak_idx - i]
                    top_idx = np.argsort(contribs)[::-1][:max(4, n // 8)]
                    theta[top_idx] += 0.15 * rng.standard_normal(len(top_idx))
                else:
                    # Add a structured perturbation
                    freq = rng.uniform(1, 5)
                    t_arr = np.arange(n)
                    pert = 0.1 * np.cos(2 * np.pi * freq * t_arr / n + rng.uniform(0, 2*np.pi))

    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()
    
    return softmax_normalize(best_theta) * S_target, best_score


def subgradient_descent(n, a_init, time_limit, rng):
    """
    Direct subgradient descent with projection onto simplex.
    """
    S_target = np.sqrt(2.0 * n)
    a = a_init.astype(np.float64).copy()
    a = project_simplex(a, S_target)
    
    best = a.copy()
    best_score = local_score(a)
    no_improve = 0
    step = 0
    
    t0 = time.monotonic()
    print(f"    Initial score: {best_score:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        peak_val, g, k_star = subgradient_at_peak(a)
        
        # Step size: eta = c / sqrt(step)
        eta = 1.0 / np.sqrt(step + 1)
        # Scale gradient to be comparable in magnitude to a
        g_scale = np.max(np.abs(g))
        if g_scale > 0:
            g = g / g_scale * S_target
        
        a_new = a - eta * g
        a_new = project_simplex(a_new, S_target)
        
        # Check score
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Escape mechanism
        if no_improve > 5000:
            # Restart from best with perturbation
            a = best + 0.05 * S_target * rng.standard_normal(n)
            a = np.clip(a, 0, None)
            a = project_simplex(a, S_target)
            no_improve = 0
        
        if step % 5000 == 0:
            print(f"    step={step}, score={local_score(a):.6f}, best={best_score:.6f}")
    
    return best, best_score
