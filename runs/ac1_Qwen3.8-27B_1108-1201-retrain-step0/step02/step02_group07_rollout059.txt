    Find direction g >= 0 maximizing sum(g) subject to (a*g)(k) <= M for all k.
    Returns the normalized direction g/sum(g).
    """
    n = len(a)
    # Convolution (a*g)(k) = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_{k-j} * g_j
    # For k = 0, ..., 2n-2

    # (a*g)(k) = sum_{j} A[k,j] * g_j where A[k,j] = a_{k-j} if 0 <= k-j <= n-1 else 0

        s = g.sum()
        if s > 1e-12:
            return g / s
    return None

"""
LP-guided and FISTA hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal feasible directions, combined with
FISTA acceleration and adaptive annealing for local escape.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_normalized(a):
    """Objective with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix C such that C @ g = a * g (linear conv)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                C[k, j] = a[idx]
    return C

def lp_direction(a, M, method='highs'):
    """LP: max sum(g) s.t. g >= 0, (a*g)(k) <= M for all k."""
    n = len(a)
    C = build_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method=method)
    if res.success:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-12:
            return g / s
    return None

def lp_optimize(a0, time_limit, t_init=0.5):
    """LP-based optimization with adaptive step size."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t = t_init
    
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit:
        g = lp_direction(a, M)
        if g is None:
            break
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            t *= 0.5
            continue
        a_new /= s
        
        new_conv = fast_convolve(a_new)
        new_M = new_conv.max()
        new_obj = 2.0 * n * new_M
        
        if new_obj < best_obj - 1e-10:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            M = new_M
            conv = new_conv
            stall = 0
        else:
            stall += 1
            if stall > 10:
                t *= 0.5
                stall = 0
            if t < 0.01:
                break
    
    return best_a, best_obj

def fast_convolve(a):

    result = np.fft.irfft(A * A, fft_size)

def obj_norm(a):

    conv = fast_convolve(a)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_mat(a):
    """Build C (2n-1)x n such that C @ g = linear_conv(a, g)."""
    n = len(a)
    C = np.zeros((2*n-1, n))

        je = min(n-1, k)
        for j in range(js, je+1):

    return C

def lp_solve(a, M):
    """Solve LP: max 1^T g s.t. C@g <= M*1, g >= 0."""
    n = len(a)
    C = build_conv_mat(a)

    return None

        src = js + int(rng.choice(je - js + 1, p=contrib[js:je+1]/total))
        tgt = int(rng.integers(n))

        a[src] -= delta
        a[tgt] += delta
        if a[src] < 0:
            a[src] += delta
            a[tgt] -= delta
            continue

            new_conv = fast_convolve(a_new)
            new_M = new_conv.max()
            new_obj = 2.0 * n * new_M
            if new_obj < best_obj - 1e-11:
                best_obj = new_obj
                best_a = a_new.copy()
                a = a_new
                M = new_M
                stall = 0
            else:
                stall += 1
        iter_count += 1
        if stall > 30:
            t *= 0.5
            stall = 0
            if t < 0.003:
                break
        if iter_count % 50 == 0 and t > 0.1:
            t = max(0.1, t * 0.95)
    return best_a, best_obj


def fista_optimize(a0, time_limit, beta=40.0):
    """FISTA with adaptive log-sum-exp smoothing."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = obj_norm(a)
    best_a = a.copy()
    t0 = time.monotonic()
    t_ista = 1.0
    y = a.copy()
    a_prev = a.copy()
    L = 4.0 * n * max(a.max(), 1e-6)
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        if max_c < 1e-15:
            break
        scaled = beta * conv_y / max_c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv_y)
            w[np.argmax(conv_y)] = 1.0
        else:
            w /= w_sum
        grad = np.zeros(n)
        for i in range(n):
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            j_hi = k_hi - i
            grad[i] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[:j_hi + 1])
        step = 1.0 / max(L, 1e-10)
        z = y - step * grad
        a_new = project_simplex(z)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_ista ** 2)) / 2.0
        y = a_new + ((t_ista - 1.0) / t_new) * (a_new - a_prev)
        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0.0)
        s2 = a_new.sum()
        if s2 < 1e-12:
            break
        a_new /= s2
        obj_val = obj_norm(a_new)
        if obj_val < best_obj - 1e-10:
            best_obj = obj_val
            best_a = a_new.copy()
        L = max(L * 0.995, 4.0 * n * a_new.max() * 0.5)
        t_ista = t_new
        iter_count += 1
    return best_a, best_obj


def anneal_informed(a0, time_limit, rng):
    """Informed simulated annealing targeting peak reduction."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.02 * (1.0 - frac) ** 2 + 0.0002
        max_c = conv.max()
        if max_c < 1e-15:
            break
        peak_thresh = max_c * 0.75
        peaks = np.where(conv >= peak_thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])
        k = int(rng.choice(peaks))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        contrib = np.zeros(je - js + 1)
        for idx, j in enumerate(range(js, je + 1)):
            contrib[idx] = a[j] * a[k - j]
        total = contrib.sum()
        if total < 1e-15:
            continue
        src_off = int(rng.choice(je - js + 1, p=contrib / total))
        src = js + src_off
        tgt = int(rng.integers(n))
        if src == tgt:
            continue
        delta = a[src] * rng.uniform(0.01, 0.06)
        if delta < 1e-10:
            continue
        a[src] -= delta
        a[tgt] += delta
        if a[src] < 0:
            a[src] += delta
            a[tgt] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        d = new_obj - cur_obj
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[src] += delta
            a[tgt] -= delta
    return best_a, best_obj


def anneal_random(a0, time_limit, rng):
    """Random pairwise transfer annealing."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = 0.03 * (1.0 - frac) ** 1.5 + 0.001
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = a[i] * rng.uniform(0.005, 0.04)
        if delta < 1e-10:
            continue
        a[i] -= delta
        a[j] += delta
        if a[i] < 0:
            a[i] += delta
            a[j] -= delta
            continue
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        d = new_obj - cur_obj
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            a[i] += delta
            a[j] -= delta
    return best_a, best_obj


def safe_eval(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_arr = np.minimum(a_arr, 1000.0)
        if a_arr.sum() < 0.01:
            return float('inf')
        if not np.all(np.isfinite(a_arr)):
            return float('inf')
        if a_arr.size == 0:
            return float('inf')
        return float(evaluate_sequence(a_arr.tolist()))
    except Exception:
        return float('inf')

def gen_initialization(name, n, rng, parent):
    """Generate a single initialization by name."""
    x = np.arange(n, dtype=np.float64)
    if name == 'uniform':
        return np.ones(n) / n
    if name == 'gauss_015':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.15 * n)) ** 2)
        g /= g.sum()
        return g
    if name == 'gauss_025':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.25 * n)) ** 2)
        g /= g.sum()
        return g
    if name == 'gauss_035':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.35 * n)) ** 2)
        g /= g.sum()
        return g
    if name == 'bimodal':
        b = np.zeros(n)
        w = max(1, n // 5)
        b[:w] = 1.0
        b[-w:] = 1.0
        b /= b.sum()
        return b
    if name == 'bimodal_wide':
        b = np.zeros(n)
        w = max(1, n // 3)
        b[:w] = 1.0
        b[-w:] = 1.0
        b /= b.sum()
        return b
    if name == 'ramp':
        r = np.arange(1, n + 1, dtype=np.float64)
        r /= r.sum()
        return r
    if name == 'random_05':
        return rng.dirichlet(np.ones(n) * 0.5)
    if name == 'random_10':
        return rng.dirichlet(np.ones(n) * 1.0)
    if name == 'random_02':
        return rng.dirichlet(np.ones(n) * 0.2)
    if name == 'sin1':
        s = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
        s = np.maximum(s, 0)
        if s.sum() > 0:
            s /= s.sum()
            return s
        return np.ones(n) / n
    if name == 'sin2':
        s = 0.5 * (1.0 + np.cos(4.0 * np.pi * x / n))
        s = np.maximum(s, 0)
        if s.sum() > 0:
            s /= s.sum()
            return s
        return np.ones(n) / n
    if name == 'chirp':
        c = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * x / (2.0 * n)))
        c = np.maximum(c, 0)
        if c.sum() > 1e-10:
            c /= c.sum()
            return c
        return np.ones(n) / n
    if name == 'parent':
        if parent is not None and parent.size == n:
            p = np.maximum(parent, 0).astype(np.float64)
            p /= p.sum()
            return p
        return None
    if name.startswith('parent_interp'):
        if parent is not None:
            p = np.maximum(parent, 0).astype(np.float64)
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                p /= p.sum()
                return p
        return None
    return None
