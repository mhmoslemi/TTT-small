"""
L-BFGS optimizer with softmax parameterization and log-sum-exp smoothing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses temperature annealing and multi-scale search.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def logsumexp_smooth(conv, T):
    """Compute smoothed max and softmax weights."""
    max_c = conv.max()
    scaled = (conv - max_c) / T
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    smooth_max = max_c + T * log_sum
    p = exp_scaled / exp_scaled.sum()
    return smooth_max, p


def compute_grad_sliding(p, a):
    """Compute g[i] = sum_j p[i+j] * a[j] via FFT.
    p has length 2n-1, a has length n.
    Returns g of length n.
    """
    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()
    P = rfft(p, N)
    A = rfft(a, N)
    g = irfft(np.conj(A) * P, N)[:n]
    return g


def objective_and_grad(z, n, T):
    """Compute objective and gradient w.r.t. z (softmax parameterization)."""
    # Softmax to get a
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp smoothing
    max_c = conv.max()
    scaled = (conv - max_c) / T
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    smooth_obj = 2.0 * n * (max_c + T * np.log(exp_scaled.sum()))
    
    # Gradient w.r.t. a
    g = compute_grad_sliding(p, a)
    grad_a = 4.0 * n * g
    
    # Gradient w.r.t. z (softmax chain rule)
    mean_grad = np.dot(grad_a, a)
    grad_z = a * (grad_a - mean_grad)
    
    return smooth_obj, grad_z

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_grad_sliding(p, a):
    """Compute g[i] = sum_j p[i+j] * a[j] via FFT.
    p has length 2n-1, a has length n. Returns g of length n.

    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()
    P = rfft(p, N)

            func, z, jac=True,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-10, 'maxls': 30}
        )
        return result.x, result.nit
    except Exception:
        return z, 0


def propose_candidate(seed=42, budget_s=2400):
    """
    Main entry point. L-BFGS with softmax parameterization and adaptive
    log-sum-exp smoothing. Multi-scale search with temperature annealing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = safe_evaluate(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            elapsed = time.monotonic() - t_start
            print("  [%.0fs] New best: score=%.6f, n=%d" % (elapsed, score, len(a)))

    parent = get_parent()
    if parent is not None:
        print("Parent sequence loaded: n=%d, score=%.6f" % (len(parent), safe_evaluate(parent)))

    # Phase 1: Multi-scale scan
    print("Phase 1: Multi-scale L-BFGS scan...")

    n_values = [128, 256, 512, 1024, 2048, 4096]
    results = []

    for n in n_values:
        if time.monotonic() > deadline - 500:
            break

        inits = generate_inits_z(n, rng, parent)
        n_inits = len(inits)
        phase1_time_left = deadline - time.monotonic() - 500
        time_per_init = max(3.0, min(30.0, phase1_time_left / max(1, n_inits * len(n_values))))

        for name, z0 in inits.items():
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 500)
            if t_alloc < 2.0:
                break

            # Adaptive temperature schedule
            a_init = softmax(z0)
            conv_init = fast_convolve(a_init)
            max_c_init = conv_init.max()

            # 3-4 temperature levels
            n_chunks = 3
            T_vals = [max_c_init * 0.4, max_c_init * 0.15, max_c_init * 0.05, max_c_init * 0.02]
            T_vals = T_vals[:n_chunks]

            z_cur = z0.copy()
            for T_val in T_vals:
                if time.monotonic() > deadline - 500:
                    break
                maxiter_chunk = max(50, int(200 * t_alloc / max(1.0, n_chunks * 10.0)))
                z_cur, _ = run_lbfgs_chunk(z_cur, n, T_val, maxiter_chunk, 1e-5)
                a_cur = softmax(z_cur)
                s = a_cur.sum()
                if s > 1e-10:
                    a_cur = a_cur / s
                record(a_cur)

            # Record final
            a_final = softmax(z_cur)
            s = a_final.sum()
            if s > 1e-10:
                a_final = a_final / s
            results.append((safe_evaluate(a_final), n, name, a_final))
            record(a_final)

    print("  Phase 1 done. Best: %.6f" % best_score)

    # Phase 2: Deep optimization on top candidates
    print("Phase 2: Deep optimization...")

    results.sort(key=lambda x: (0 if np.isfinite(x[0]) else 1, x[0]))
    top_candidates = []
    seen_keys = set()
    for score, n, name, a in results:
        key = (n, name)
        if key not in seen_keys and len(top_candidates) < 12:
            if np.isfinite(score):
                top_candidates.append((n, a.copy()))
                seen_keys.add(key)

    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))

    phase2_time = deadline - time.monotonic() - 300
    n_deep = len(top_candidates)
    time_per_deep = max(15.0, min(200.0, phase2_time / max(1, n_deep)))

    for idx, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 300)
        if t_alloc < 8.0:
            break

        print("  [%d/%d] n=%d, budget=%.0fs" % (idx + 1, n_deep, n, t_alloc))

        a0_clip = np.maximum(a0, 1e-15)
        z0 = np.log(a0_clip / a0_clip.sum())

        # Multi-temperature deep optimization with adaptive T
        a_cur = softmax(z0)
        conv_cur = fast_convolve(a_cur)
        max_c = conv_cur.max()

        n_T_levels = 5
        t_per_T = t_alloc / n_T_levels

        for T_level in range(n_T_levels):
            if time.monotonic() > deadline - 300:
                break

            T_frac = 0.25 * (0.4 ** T_level)
            T_val = max_c * T_frac

            # Adapt T based on current conv
            a_tmp = softmax(z0)
            conv_tmp = fast_convolve(a_tmp)
            T_val = max(conv_tmp.max() * T_frac, 1e-10)

            maxiter_chunk = max(100, int(500))
            z0, n_iters = run_lbfgs_chunk(z0, n, T_val, maxiter_chunk, 1e-6)

            a_cur = softmax(z0)
            s = a_cur.sum()
            if s > 1e-10:
                a_cur = a_cur / s
            record(a_cur)

            # Update max_c for next level
            conv_cur = fast_convolve(a_cur)
            max_c = conv_cur.max()

    print("  Phase 2 done. Best: %.6f" % best_score)

    # Phase 3:exploitation of best with very fine tuning
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 3: Fine tuning...")

        n = len(best_sequence)
        a0 = np.maximum(best_sequence, 1e-15)
        a0 = a0 / a0.sum()
        z0 = np.log(a0)

        t_remaining = deadline - time.monotonic() - 10

        # Very low temperature fine-tuning
        conv0 = fast_convolve(a0)
        max_c0 = conv0.max()

        # Try several very low temperatures
        T_fine_list = [max_c0 * 0.01, max_c0 * 0.005, max_c0 * 0.001]
        t_per_fine = t_remaining / (len(T_fine_list) + 2)

        for T_val in T_fine_list:
            if time.monotonic() > deadline - 10:
                break
            if t_per_fine < 2.0:
                break
            z0, _ = run_lbfgs_chunk(z0, n, T_val, 500, 1e-7)
            a_cur = softmax(z0)
            record(a_cur)

        # Try perturbing the best and re-optimizing
        if time.monotonic() < deadline - 15:
            a_best = best_sequence.copy()
            for perturb_trial in range(3):
                if time.monotonic() > deadline - 15:
                    break
                perturbation = rng.normal(0, 0.1, size=len(a_best))
                a_pert = np.maximum(a_best + perturbation, 1e-15)
                a_pert = a_pert / a_pert.sum()
                z_pert = np.log(a_pert)
                conv_pert = fast_convolve(a_pert)
                T_pert = conv_pert.max() * 0.05
                z_pert, _ = run_lbfgs_chunk(z_pert, n, T_pert, 300, 1e-6)
                a_res = softmax(z_pert)
                record(a_res)

    # Phase 4: Try a slightly different n around the best
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("Phase 4: Adjacent n exploration...")
        n_best = len(best_sequence)
        candidate_ns = []
        for delta in [64, 128, -64, -128]:
            n_try = n_best + delta
            if n_try >= 32 and n_try <= 8192 and n_try != n_best:
                candidate_ns.append(n_try)

        for n_try in candidate_ns[:3]:
            if time.monotonic() > deadline - 20:
                break
            # Interpolate best to new length
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                 np.linspace(0, 1, n_best), best_sequence)
            a_interp = np.maximum(a_interp, 1e-15)
            a_interp = a_interp / a_interp.sum()
            z_interp = np.log(a_interp)
            conv_interp = fast_convolve(a_interp)
            T_interp = conv_interp.max() * 0.1
            z_opt, _ = run_lbfgs_chunk(z_interp, n_try, T_interp, 300, 1e-6)
            a_opt = softmax(z_opt)
            record(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.6f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
