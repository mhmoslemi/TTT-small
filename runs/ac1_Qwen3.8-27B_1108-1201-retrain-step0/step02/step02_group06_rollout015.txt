Element-wise line search optimization for minimizing peak autocorrelation

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_from_conv(n, conv, sum_a):
    """Compute score from precomputed convolution and sum."""
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * np.max(conv) / (sum_a * sum_a)

"""
Element-wise line search with multi-scale exploration for minimizing peak
autocorrelation of non-negative step functions. Uses guaranteed-improvement
1D searches, vectorized convolution updates, and adaptive sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast score: 2n * max(convolve(a,a)) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def compute_unaffected_max(conv, j, n):
    """Max of conv outside the affected range [j, j+n-1]."""
    k_max = min(j + n, 2 * n - 1)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[j:k_max] = False
    if np.any(mask):
        return np.max(conv[mask])
    return 0.0


def element_line_search(a, j, conv, sum_a, n, unaffected_max):
    """
    Find optimal a[j] via 1D search. Returns (new_x, new_conv, new_score).
    """
    aj = a[j]
    k_max = min(j + n, 2 * n - 1)
    ks = np.arange(j, k_max)
    diffs = ks - j

    # Precompute base values for affected conv entries
    # For k != 2j: new_conv[k] = conv[k] + 2*(x-aj)*a[diffs]
    # For k == 2j: new_conv[k] = conv[k] - aj^2 + x^2
    base = conv[ks].copy()
    linear_coeff = np.zeros(len(ks))
    for idx in range(len(ks)):
        if ks[idx] == 2 * j:
            base[idx] = conv[ks[idx]] - aj * aj
        else:
            linear_coeff[idx] = 2.0 * a[diffs[idx]]
            base[idx] = conv[ks[idx]] - 2.0 * aj * a[diffs[idx]]

    has_quadratic = (ks == 2 * j)
    quad_idx = np.where(has_quadratic)[0]

    def eval_score(x):
        new_sum = sum_a - aj + x
        if new_sum < 1e-15:
            return np.inf
        vals = base + linear_coeff * x
        if len(quad_idx) > 0:
            for qi in quad_idx:
                vals[qi] = base[qi] + x * x
        local_max = np.max(vals)
        total_max = max(unaffected_max, local_max)
        return 2.0 * n * total_max / (new_sum * new_sum)

    # Initial candidates
    best_x = aj
    best_s = eval_score(aj)

    candidates = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0]) * aj
    candidates = np.concatenate([candidates, [0.001, 0.01, aj * 0.8, aj * 1.2]])
    candidates = np.clip(candidates, 0.0, 10.0 * max(aj, 0.01))
    candidates = np.unique(np.round(candidates, 10))

    for cx in candidates:
        s = eval_score(cx)
        if s < best_s:
            best_s = s
            best_x = cx

    # Refine with scipy minimize_scalar in a bracket around best
    if best_x > 1e-10:
        lo = max(0.0, best_x * 0.5)
        hi = best_x * 2.0 + 0.01
    else:
        lo = 0.0
        hi = max(aj * 0.5 + 0.01, 0.01)

    try:
        res = minimize_scalar(eval_score, bounds=(lo, hi), method='bounded',
                              options={'maxiter': 50, 'xatol': 1e-10})
        if res.fun < best_s:
            best_s = res.fun
            best_x = res.x
    except Exception:
        pass

    # Update conv
    new_conv = conv.copy()
    new_vals = base + linear_coeff * best_x
    if len(quad_idx) > 0:
        for qi in quad_idx:
            new_vals[qi] = base[qi] + best_x * best_x
    new_conv[ks] = new_vals

    return best_x, new_conv, best_s


def full_sweep(a, conv, sum_a, n, order=None):
    """One full sweep of element-wise line search."""
    if order is None:
        order = np.arange(n)
    improved = 0
    for j in order:
        unaffected_max = compute_unaffected_max(conv, j, n)
        new_x, new_conv, new_score = element_line_search(a, j, conv, sum_a, n, unaffected_max)
        if abs(new_x - a[j]) > 1e-12:
            a[j] = new_x
            conv = new_conv
            sum_a = sum_a - a[j] + new_x
            improved += 1
    return a, conv, sum_a, improved


def spectral_init(n, rng, n_iter=50):
    """Spectral iterative initialization."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    for _ in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        corr = np.fft.irfft(power, 2 * n)[:n]
        alpha = 1.0 / (n + 1.0)
        a = a - alpha * corr
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def ramp_init(n, rng, style='linear'):
    """Various ramp initializations."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if style == 'linear':
        a = 1.0 + 0.5 * t
    elif style == 'quadratic':
        a = 1.0 + 0.5 * t ** 2
    elif style == 'sinusoidal':
        a = 1.0 + 0.4 * np.abs(np.sin(2 * np.pi * t))
    elif style == 'bump':
        a = 1.0 + 0.5 * np.exp(-((t - 0.5) ** 2) / 0.05)
    elif style == 'two_bump':
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
    elif style == 'sqrt':
        a = 1.0 + 0.5 * np.sqrt(t)
    elif style == 'exponential':
        a = np.exp(0.5 * t)
    else:
        a = 1.0 + 0.3 * t
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def upsample(a_old, n_new):
    """Upsample a sequence to a new length via FFT band-limited interpolation."""
    n_old = len(a_old)
    # Use FFT-based interpolation
    A = np.fft.rfft(a_old, n_old)
    A_new = np.zeros(n_new // 2 + 1, dtype=np.complex128)
    n_copy = min(len(A), n_new // 2 + 1)
    A_new[:n_copy] = A[:n_copy]
    a_new = np.fft.irfft(A_new, n_new)
    a_new = np.clip(a_new, 0.0, None)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = np.ones(n_new) / n_new
    else:
        a_new = a_new / s
    return a_new


def optimize_sequence(a_init, time_limit, rng, verbose=True):
    """
    Optimize a sequence using element-wise line search with periodic
    perturbation for escaping local minima.
    """
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a))
        s = len(a)
    a = a / s
    n = len(a)
    conv = fft_convolve(a, a)
    sum_a = 1.0

    current_score = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    sweep_count = 0
    patience = 0
    max_patience = 15
    perturbation_count = 0

    while time.monotonic() - t0 < time_limit:
        # Determine order: random or sequential
        if sweep_count % 5 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        a, conv, sum_a, improved = full_sweep(a, conv, sum_a, n, order)
        sweep_count += 1

        current_score = 2.0 * n * np.max(conv) / (sum_a * sum_a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if verbose and sweep_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      sweep={sweep_count}, best_score={best_score:.6f}, "
                  f"current={current_score:.6f}, t={elapsed:.1f}s")

        if improved == 0:
            patience += 1

        if patience >= max_patience:
            # Perturbation to escape local minimum
            perturbation_count += 1
            strength = min(0.1, 0.01 * perturbation_count)
            a = a * np.exp(rng.normal(0, strength, n))
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            conv = fft_convolve(a, a)
            sum_a = 1.0
            patience = 0

    return best_a, best_score


def pair_optimization(a, conv, sum_a, n, rng, num_pairs=50):
    """Apply a few pair optimizations for additional improvement."""
    for _ in range(num_pairs):
        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            continue
        # Alternate 1D searches for j and k
        for _ in range(3):
            unaffected_max = compute_unaffected_max(conv, j, n)
            new_x, conv, _ = element_line_search(a, j, conv, sum_a, n, unaffected_max)
            sum_a = sum_a - a[j] + new_x
            a[j] = new_x
            unaffected_max = compute_unaffected_max(conv, k, n)
            new_x, conv, _ = element_line_search(a, k, conv, sum_a, n, unaffected_max)
            sum_a = sum_a - a[k] + new_x
            a[k] = new_x
    return a, conv, sum_a

        n_best = len(best_seq)
        a_norm = best_seq / np.sum(best_seq)
        t_polish = max(10.0, deadline - time.monotonic() - 30)
        print(f"  Polishing n={n_best} with {t_polish:.0f}s budget")

        # Use element-wise search more aggressively with different orders
        conv = fft_convolve(a_norm, a_norm)
        sum_a = 1.0
        a = a_norm.copy()

        polish_deadline = time.monotonic() + t_polish
        sweep = 0
        while time.monotonic() < polish_deadline:
            order = rng.permutation(n_best)
            a, conv, sum_a, improved = full_sweep(a, conv, sum_a, n_best, order)
            sweep += 1
            sc = 2.0 * n_best * np.max(conv) / (sum_a * sum_a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                if sweep % 5 == 0:
                    print(f"    polish sweep {sweep}: score={sc:.6f}")
            if improved == 0 and sweep % 3 == 0:
                # Small perturbation
                a = a * np.exp(rng.normal(0, 0.005, n_best))
                a = np.clip(a, 0.0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_seq / np.sum(best_seq)
                conv = fft_convolve(a, a)
                sum_a = 1.0

        final_sc = score_fast(best_seq)
        print(f"  After polish: score={final_sc:.6f}")

    # Phase 5: Try a few more diverse inits if time remains
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 5: Diverse exploration ===")
        remaining = deadline - time.monotonic() - 60
        n_try = len(best_seq)
        for trial in range(4):
            if time.monotonic() > deadline - 60:
                break
            styles = ['sinusoidal', 'two_bump', 'bump', 'quadratic']
            a_try = ramp_init(n_try, rng, styles[trial % len(styles)])
            # Blend with current best
            blend = 0.3 * a_try + 0.7 * (best_seq / np.sum(best_seq))
            blend = blend / np.sum(blend)
            t_per = max(10.0, remaining / (5 - trial))
            a_best, sc = optimize_sequence(blend, t_per, rng, verbose=False)
            sc_actual = score_fast(a_best)
            print(f"  trial {trial} ({styles[trial % len(styles)]}): score={sc_actual:.6f}")
            if sc_actual < best_score:
                best_score = sc_actual
                best_seq = a_best.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
