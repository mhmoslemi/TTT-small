import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_subgradient(conv, a, temperature):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    # Softmax with temperature
    scaled = conv / max_c / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Compute gradient: sg[i] = sum_k w[k] * a[k-i] (where valid)
    sg = np.zeros(n)
    # For each k, the contribution to sg[i] is w[k] * a[k-i] for i = k-j where j is valid
    # Equivalently: sg = IFFT-like operation

    for k in range(len(w)):

"""
Step function optimizer combining adaptive mirror descent, peak-directed coordinate descent,
and diverse initialization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(conv, a, temp):
    n = len(a)
    mx = conv.max()
    if mx < 1e-15:
        return np.zeros(n)
    s = conv / (mx * temp)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for k in range(len(w)):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
    return sg

def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Hann window
    h = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    h = np.maximum(h, 0)
    h /= h.sum()
    inits['hann'] = h
    
    # Raised cosine (Hanning)
    r = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    r = np.maximum(r, 1e-10)
    r /= r.sum()
    inits['raised_cos'] = r
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - n/2) / (n * 0.25))**2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Sine
    for freq in [1, 2, 3]:
        s = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        s = np.maximum(s, 1e-10)
        s /= s.sum()
        inits['sin_%d' % freq] = s
    
    # Random Dirichlet
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dirich_%s' % alpha] = r
    
    # Power law
    for p in [0.5, 2.0, 4.0]:
        pl = (1 - np.abs(2 * x / n - 1))**p
        pl = np.maximum(pl, 1e-10)
        pl /= pl.sum()
        inits['power_%s' % p] = pl
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    alt = np.maximum(alt, 0)
    alt /= alt.sum()
    inits['alternating'] = alt
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp
    
    return inits

"""
Step function optimizer using adaptive mirror descent, peak-directed coordinate descent,
and diverse initialization to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(conv, a, temp):
    """Smoothed subgradient of max(conv) via softmax weighting. O(n^2) but fast for n<=1024."""
    n = len(a)
    mx = conv.max()
    if mx < 1e-15:
        return np.zeros(n)
    s = conv / (mx * temp)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for k in range(len(w)):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    return sg


def make_gradient_fast(conv, a, temp):
    """FFT-accelerated gradient computation."""
    n = len(a)
    mx = conv.max()
    if mx < 1e-15:
        return np.zeros(n)
    s = conv / (mx * temp)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # sg[i] = sum_k w[k] * a[k-i] for valid indices
    # This is cross-correlation: sg = correlate(w, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # correlation: sg[i] = sum_j w[i+j] * a[j]
    # = IFFT(conj(W) * A) with appropriate slicing
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    # The correlation result: corr[i] = sum_j a[j] * w[i+j] (circular, but we padded)
    # Actually we need: sg[i] = sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]
    # With zero-padding, the circular correlation gives us exactly this for valid indices
    # corr is of length fft_size, we need indices 0..n-1
    sg = corr[:n]
    return sg


def mirror_descent_opt(a0, time_limit, temp_init=0.35, eta_init=0.5, seed=0):
    """Adaptive mirror descent with log-space momentum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    temp = temp_init
    eta = eta_init
    patience = 0
    max_pat = 80
    log_a = np.log(np.maximum(a, 1e-30))
    momentum = np.zeros(n)
    mu = 0.7
    t0 = time.monotonic()
    use_fft_grad = n > 128
    while time.monotonic() - t0 < time_limit:
        if use_fft_grad:
            sg = make_gradient_fast(conv, a, temp)
        else:
            sg = compute_gradient(conv, a, temp)
        sg_c = sg - sg.mean()
        # Nesterov look-ahead in log space
        log_a_next = log_a - mu * momentum
        log_update = -eta * (sg_c - momentum * 0.1)
        log_update -= log_update.max()
        log_a_new = log_a_next + log_update
        log_a_new -= log_a_new.max()
        a = np.exp(log_a_new)
        a = np.maximum(a, 0.0)
        a = project_simplex(a)
        log_a = np.log(np.maximum(a, 1e-30))
        momentum = mu * momentum + (1.0 - mu) * (np.log(np.maximum(a, 1e-30)) - log_a + log_update)
        log_a = np.log(np.maximum(a, 1e-30))
        conv = fast_convolve(a)
        cur = 2.0 * n * conv.max()
        if cur < best_obj - 1e-10:
            best_obj = cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience >= max_pat:
            patience = 0
            temp *= 1.35
            eta *= 0.6
            mu = max(0.0, mu - 0.05)
            if eta < 0.003:
                elapsed = time.monotonic() - t0
                frac = min(1.0, elapsed / max(time_limit, 1.0))
                eta = eta_init * (1.0 - 0.75 * frac)
                temp = temp_init * (1.0 + 6.0 * frac)
                mu = 0.5 * (1.0 - frac)
    return best_a, best_obj

def peak_cd(a0, time_limit, seed=0):
    """Peak-directed coordinate descent with targeted mass transfer."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    max_stall = 5

    while time.monotonic() - t0 < time_limit:
        progress = False
        # Find significant peaks
        thresh = cur_max * 0.82
        peaks = np.where(conv >= thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # For each significant peak, try to reduce it
        for k in peaks[:min(5, len(peaks))]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            j_arr = np.arange(js, je + 1)
            contribs = a[j_arr] * a[k - j_arr]
            total = contribs.sum()
            if total < 1e-15:
                continue
            # Top contributors to this peak
            top_n = min(6, len(j_arr))
            order = np.argsort(contribs)[::-1][:top_n]
            for _ in range(top_n):
                j = int(j_arr[order[_]])
                if a[j] < 1e-9:
                    continue
                # Try moving mass to target positions
                # Prefer targets that don't contribute to peaks
                n_targets = 10
                for _t in range(n_targets):
                    target = int(rng.integers(0, n))
                    if target == j:
                        continue
                    found = False
                    for df in [0.06, 0.025, 0.01, 0.004]:
                        delta = a[j] * df
                        if delta < 1e-11:
                            continue
                        a[j] -= delta
                        a[target] += delta
                        nc = fast_convolve(a)
                        nm = nc.max()
                        if nm < cur_max - 1e-12:
                            cur_max = nm
                            conv = nc
                            cur_obj = 2.0 * n * cur_max
                            progress = True
                            found = True
                            if cur_obj < best_obj:
                                best_obj = cur_obj
                                best_a = a.copy()
                            break
                        else:
                            a[j] += delta
                            a[target] -= delta
                    if found:
                        # Continue trying to reduce this peak further
                        continue
                if found:
                    if time.monotonic() - t0 > time_limit:
                        break

        if not progress:
            stall_count += 1
            # Try broader random moves
            for _ in range(30):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                for df in [0.03, 0.01]:
                    delta = a[i] * df
                    if delta < 1e-10:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    nc = fast_convolve(a)
                    nm = nc.max()
                    if nm < cur_max - 1e-12:
                        cur_max = nm
                        conv = nc
                        cur_obj = 2.0 * n * cur_max
                        progress = True
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
            if not progress:
                if stall_count >= max_stall:
                    break
            else:
                stall_count = 0
        else:
            stall_count = 0

    return best_a, best_obj


def symmetric_cd(a0, time_limit, seed=0):
    """Coordinate descent that enforces symmetry a[i] = a[n-1-i]."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    # Enforce symmetry
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym /= a_sym.sum()
    a = a_sym
    conv = fast_convolve(a)
    cur_max = conv.max()
    cur_obj = 2.0 * n * cur_max
    best_obj = cur_obj
    best_a = a.copy()
    t0 = time.monotonic()
    half_n = (n + 1) // 2

    while time.monotonic() - t0 < time_limit:
        progress = False
        thresh = cur_max * 0.82
        peaks = np.where(conv >= thresh)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        for k in peaks[:3]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            j_arr = np.arange(js, je + 1)
            contribs = a[j_arr] * a[k - j_arr]
            order = np.argsort(contribs)[::-1][:4]
            for _ in range(min(4, len(order))):
                j = int(j_arr[order[_]])
                if j >= half_n:
                    j = n - 1 - j  # Use the symmetric partner
                if a[j] < 1e-9:
                    continue
                for _t in range(6):
                    target = int(rng.integers(0, half_n))
                    if target == j:
                        continue
                    for df in [0.04, 0.015, 0.005]:
                        delta = a[j] * df
                        if delta < 1e-11:
                            continue
                        # Move symmetrically
                        a[j] -= delta
                        a[n - 1 - j] -= delta
                        a[target] += delta
                        a[n - 1 - target] += delta
                        a = np.maximum(a, 0)
                        a = project_simplex(a)
                        nc = fast_convolve(a)
                        nm = nc.max()
                        if nm < cur_max - 1e-12:
                            cur_max = nm
                            conv = nc
                            cur_obj = 2.0 * n * cur_max
                            progress = True
                            if cur_obj < best_obj:
                                best_obj = cur_obj
                                best_a = a.copy()
                            # Restore symmetry
                            a = 0.5 * (a + a[::-1])
                            a = project_simplex(a)
                            break
                        else:
                            # Undo
                            a[j] += delta
                            a[n - 1 - j] += delta
                            a[target] -= delta
                            a[n - 1 - target] -= delta
                            a = np.maximum(a, 0)
                            a = project_simplex(a)
                            a = 0.5 * (a + a[::-1])
                            a = project_simplex(a)

        if not progress:
            break

    return best_a, best_obj


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Hann window
    h = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    h = np.maximum(h, 1e-10)
    h /= h.sum()
    inits['hann'] = h

    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2) / (n * 0.2)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi = np.maximum(bi, 1e-10)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine
    for freq in [1, 2, 3, 4]:
        s = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        s = np.maximum(s, 1e-10)
        s /= s.sum()
        inits['cos_%d' % freq] = s

    # Random (different concentrations)
    for alpha in [0.05, 0.2, 0.5, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits['dir_%s' % str(alpha)] = r

    # Power shapes
    for p in [0.3, 0.7, 2.0, 5.0]:
        t = np.abs(2 * x / n - 1)
        pl = (1 - t ** 2) ** p
        pl = np.maximum(pl, 1e-10)
        pl /= pl.sum()
        inits['pow_%s' % str(p)] = pl

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.2
    alt = np.maximum(alt, 0)
    alt /= alt.sum()
    inits['alt'] = alt

    # Two-tone
    tt = np.ones(n)
    tt[::3] = 0.1
    tt = np.maximum(tt, 0)
    tt /= tt.sum()
    inits['two_tone'] = tt

    # Chirp
    chirp = 0.5 * (1 + np.cos(2 * np.pi * x ** 2 / (2 * n)))
    chirp = np.maximum(chirp, 1e-10)
    chirp /= chirp.sum()
    inits['chirp'] = chirp

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0).copy()
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_i'] = pp

    return inits

            seen.add(key)
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_budget = min(1200.0, (deadline - time.monotonic()) * 0.42)
    time_per_deep = max(8.0, phase2_budget / max(1, len(top_configs)))

    for idx, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 800:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 800)
        if t_alloc < 5.0:
            break
        print("  [%d/%d] n=%d %s, budget=%.0fs" % (idx + 1, len(top_configs), n, name, t_alloc))

        a_cur = a0.copy()
        sub = t_alloc / 6.0

        # Stage 1: Mirror descent (exploratory)
        a_cur, _ = mirror_descent_simple(a_cur, time_limit=sub, temp_init=0.25, eta_init=0.5, seed=seed + 10)
        record(a_cur)

        # Stage 2: Mirror descent with momentum (exploitative)
        a_cur, _ = mirror_descent_opt(a_cur, time_limit=sub, temp_init=0.4, eta_init=0.35, seed=seed + 20)
        record(a_cur)

        # Stage 3: Peak-directed CD
        a_cur, _ = peak_cd(a_cur, time_limit=sub, seed=seed + 30)
        record(a_cur)

        # Stage 4: Mirror descent (fine)
        a_cur, _ = mirror_descent_simple(a_cur, time_limit=sub, temp_init=1.0, eta_init=0.15, seed=seed + 40)
        record(a_cur)

        # Stage 5: Peak CD again
        a_cur, _ = peak_cd(a_cur, time_limit=sub, seed=seed + 50)
        record(a_cur)

        # Stage 6: Very fine mirror descent
        a_cur, _ = mirror_descent_simple(a_cur, time_limit=sub, temp_init=3.0, eta_init=0.06, seed=seed + 60)
        record(a_cur)

    print("  Phase 2 complete. Best: %.8f" % best_score)

    # === Phase 3: Symmetric exploitation ===
    if best_sequence is not None and time.monotonic() < deadline - 200:
        print("Phase 3: Symmetric CD...")
        n_best = len(best_sequence)
        t_sym = min(200.0, (deadline - time.monotonic()) - 200)
        if t_sym > 10:
            a_sym_start = 0.5 * (best_sequence + best_sequence[::-1])
            a_sym_start = np.maximum(a_sym_start, 0)
            a_sym_start /= a_sym_start.sum()
            a_opt, _ = symmetric_cd(a_sym_start, time_limit=t_sym * 0.5, seed=seed + 200)
            record(a_opt)
            # Also try non-symmetric CD on the symmetric result
            if time.monotonic() < deadline - 200:
                a_opt2, _ = peak_cd(a_opt, time_limit=t_sym * 0.3, seed=seed + 210)
                record(a_opt2)

    print("  Phase 3 complete. Best: %.8f" % best_score)

    # === Phase 4: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Final polish...")
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_best = best_sequence.copy()
            a_best = np.maximum(a_best, 0)
            a_best /= a_best.sum()

            # Fine CD
            t1 = remaining * 0.5
            a_opt, _ = peak_cd(a_best, time_limit=t1, seed=seed + 500)
            record(a_opt)

            if time.monotonic() < deadline - 10:
                rem2 = deadline - time.monotonic() - 10
                if rem2 > 5:
                    # Ultra-fine mirror descent
                    a_opt2, _ = mirror_descent_simple(best_sequence, time_limit=rem2 * 0.6,
                                                      temp_init=5.0, eta_init=0.03, seed=seed + 600)
                    record(a_opt2)

            if time.monotonic() < deadline - 10:
                rem3 = deadline - time.monotonic() - 10
                if rem3 > 3:
                    # Final CD
                    a_opt3, _ = peak_cd(best_sequence, time_limit=rem3, seed=seed + 700)
                    record(a_opt3)

    print("  Phase 4 complete. Best: %.8f" % best_score)

    # === Phase 5: Explore new lengths if time remains ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic() - 30
        if rem > 60:
            print("Phase 5: Try new lengths...")
            n_best = len(best_sequence)
            for n_try in [n_best - 32, n_best + 32, n_best + 64]:
                if n_try < 32 or n_try > 1024:
                    continue
                if time.monotonic() > deadline - 30:
                    break
                # Interpolate best to new length
                a_interp = np.interp(np.linspace(0, 1, n_try),
                                     np.linspace(0, 1, n_best),
                                     best_sequence)
                a_interp = np.maximum(a_interp, 0)
                a_interp /= a_interp.sum()
                t_alloc = min(30.0, rem * 0.3)
                a_opt, _ = mirror_descent_simple(a_interp, time_limit=t_alloc,
                                                 temp_init=0.3, eta_init=0.4, seed=seed + n_try)
                record(a_opt)
                if time.monotonic() < deadline - 30:
                    t_cd = min(20.0, (deadline - time.monotonic()) - 30)
                    if t_cd > 3:
                        a_opt2, _ = peak_cd(a_opt, time_limit=t_cd, seed=seed + n_try + 1)
                        record(a_opt2)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = safe_evaluate(best_sequence)

    print("FINAL SCORE: %.8f" % best_score)
    print("Sequence length: %d" % len(best_sequence))

    return [float(x) for x in best_sequence]
