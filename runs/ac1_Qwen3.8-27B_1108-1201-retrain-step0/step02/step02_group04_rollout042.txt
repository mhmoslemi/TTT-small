def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (sum_a ** 2)

def smooth_max_and_grad(a, T):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)

    max_c = np.max(conv)
    logits = (conv - max_c) / T

    # = 2 * (a_rev * weights)[j + n - 1]

    aw = fft_convolve(a_rev, weights)
    grad = 2.0 * aw[n-1 : 2*n-1]
    
    return smooth_max, grad, weights, conv
