Optimization of non-negative step functions to minimize peak autocorrelation ratio.
Uses multi-phase approach: diverse initialization, smooth gradient descent with 
soft-max objective, subgradient refinement, and peak-directed local search.

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def get_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def soft_max_and_grad(c, tau):
    """Compute soft-max of c and gradient w.r.t. c."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    svm = c_max + tau * np.log(s)
    return svm, w

"""
Multi-phase optimization for minimizing peak autocorrelation ratio of non-negative
step functions. Uses diverse initialization, Adam gradient descent with annealed
soft-max, subgradient refinement, and peak-directed local search with subgradient
informed mutations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def get_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def softmax_stable(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def score_grad_softmax(theta, S, tau):
    """
    Compute soft-max score and gradient w.r.t. theta for a = softmax(theta)*S.
    score = 2n * soft_max(conv(a,a), tau) / S^2
    """
    n = len(theta)
    a = softmax_stable(theta) * S
    c = fft_convolve(a, a)
    
    # Soft max
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s
    svm = c_max + tau * np.log(s)
    score = 2.0 * n * svm / (S * S)
    
    # Gradient of svm w.r.t. a[i]:
    # d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n
    # d(svm)/d(a[i]) = sum_k w[k] * 2*a[k-i]
    # = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * cross_correlation(w, a)[i]
    # where cross_corr(i) = sum_k w[k] * a[k-i]
    
    size = 1
    while size < 2 * (2 * n - 1) - 1:
        size <<= 1
    W = np.fft.fft(w, size)
    A = np.fft.fft(a, size)
    # cross_corr[i] = sum_k w[k] * a[k-i] = ifft(conj(FFT(a)) * FFT(w))[i]
    corr = np.fft.ifft(np.conj(A) * W).real[:n]
    g_a = 2.0 * corr
    
    # d(score)/d(a[i]) = 2n * (g_a[i]*S - 2*svm) / S^3
    g_a = 2.0 * n * (g_a * S - 2.0 * svm) / (S ** 3)
    
    # Transform to theta gradient: g_theta = a * (g_a - dot(a, g_a)/S)
    inner = np.dot(a, g_a)
    g_theta = a * (g_a - inner / S)
    
    return score, g_theta


def compute_subgrad_direct(a, top_lags):
    """
    Compute subgradient of max_k conv[k] w.r.t. a.
    g[i] = (1/|S|) * sum_{k in S} 2*a[k-i] (for 0 <= k-i < n)

    g = np.zeros(n)
    for k in top_lags:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            idx = np.arange(i_end - i_start + 1)[::-1]
            g[i_start:i_end + 1] += a[idx]
    g *= 2.0 / len(top_lags)
    return g

            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            delta = new_score - current_score
            if delta < temp * 5:
                if rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    current = candidate
                    current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def fft_phase_init(n, rng, n_iter=100):
    """Gerchberg-Saxton style: flatten magnitude spectrum with positivity."""
    N_fft = 1
    while N_fft < 2 * n:
        N_fft <<= 1
    
    a = np.ones(n) / n
    for it in range(n_iter):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        a_full = np.fft.ifft(phase).real
        a = np.clip(a_full[:n], 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def random_init(n, rng, spread=0.5):
    a = 1.0 + spread * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def cosine_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a / np.sum(a)


def upsample_interp(a, new_n, rng):
    """Upsample via interpolation with small noise."""
    n = len(a)
    if new_n <= n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp += 0.01 * rng.standard_normal(new_n)
    a_interp = np.clip(a_interp, 0.001, None)
    return a_interp / np.sum(a_interp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function.
    Combines diverse initialization, Adam gradient descent with annealed soft-max,
    subgradient refinement, and peak-directed local search.
    """
    t_global = time.monotonic()
    time_budget = max(60.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check for provided starting sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        s = get_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # === Phase 1: Diverse Initialization ===
    print("\n=== Phase 1: Initialization ===")
    candidates = []
    
    lengths = [256, 384, 512, 768, 1024]
    
    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        
        inits = []
        inits.append(('fft', fft_phase_init(n, rng, n_iter=80)))
        inits.append(('fft2', fft_phase_init(n, rng, n_iter=150)))
        inits.append(('rand', random_init(n, rng, spread=0.4)))
        inits.append(('cos', cosine_init(n, rng)))
        
        for name, a in inits:
            a_norm = a / np.sum(a) * np.sqrt(2.0 * n)
            sc = get_score(a_norm)
            print(f"  n={n}, {name}: score={sc:.6f}")
            candidates.append((sc, n, a_norm.copy()))
        
        if time.monotonic() > deadline - 600:
            break
    
    if best_seq is not None:
        candidates.append((best_score, len(best_seq), best_seq.copy()))
    
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5: {[(f'{c[0]:.4f}', c[1]) for c in candidates[:5]]}")
    
    # === Phase 2: Gradient Optimization ===
    print("\n=== Phase 2: Adam Gradient Optimization ===")
    n_opt = min(6, len(candidates))
    time_per_gradient = min(120.0, max(20.0, (deadline - time.monotonic() - 300) / n_opt))
    
    optimized = []
    for i in range(n_opt):
        if time.monotonic() > deadline - 200:
            break
        sc0, n, a = candidates[i]
        # Scale time allocation by complexity
        t_alloc = time_per_gradient * min(1.0, 384.0 / n)
        t_alloc = max(10.0, t_alloc)
        
        print(f"  [{i+1}/{n_opt}] n={n}, score0={sc0:.6f}, t={t_alloc:.1f}s")
        theta_init = np.log(a / np.sqrt(2.0 * n) + 1e-12)
        a_opt, s_opt = optimize_adam(theta_init, n, t_alloc, rng)
        print(f"    -> score={s_opt:.6f}")
        
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        
        optimized.append((s_opt, n, a_opt))
    
    optimized.sort(key=lambda x: x[0])
    
    # === Phase 3: Subgradient Refinement ===
    print("\n=== Phase 3: Subgradient Refinement ===")
    for i in range(min(3, len(optimized))):
        if time.monotonic() > deadline - 150:
            break
        sc, n, a = optimized[i]
        t_alloc = min(40.0, max(10.0, (deadline - time.monotonic() - 100) / (min(3, len(optimized)) - i)))
        print(f"  [{i+1}] n={n}, score={sc:.6f}, t={t_alloc:.1f}s")
        a_sub, s_sub = subgradient_optimize(a, t_alloc, rng, top_k=3, eta0=0.03, decay=0.98)
        print(f"    -> score={s_sub:.6f}")
        if s_sub < best_score:
            best_score = s_sub
            best_seq = a_sub.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
    
    # === Phase 4: Local Search ===
    print(f"\n=== Phase 4: Local Search (best so far: {best_score:.6f}) ===")
    if best_seq is not None:
        t_alloc = min(300.0, max(30.0, deadline - time.monotonic() - 60))
        print(f"  n={len(best_seq)}, t={t_alloc:.1f}s")
        a_ls, s_ls = local_search_peaked(best_seq, t_alloc, rng, top_k=6)
        print(f"  After local search: {s_ls:.6f}")
        if s_ls < best_score:
            best_score = s_ls
            best_seq = a_ls.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 5: Coarse-to-Fine ===
    print("\n=== Phase 5: Coarse-to-Fine ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_cur = len(best_seq)
        for target_n in [min(n_cur * 2, 1024), min(n_cur * 2, 1536)]:
            if target_n <= n_cur or time.monotonic() > deadline - 80:
                break
            a_up = upsample_interp(best_seq, target_n, rng)
            a_up = a_up / np.sum(a_up) * np.sqrt(2.0 * target_n)
            t_alloc = min(60.0, max(15.0, deadline - time.monotonic() - 60))
            print(f"  Upsample to n={target_n}, t={t_alloc:.1f}s")
            theta_init = np.log(a_up / np.sqrt(2.0 * target_n) + 1e-12)
            a_opt, s_opt = optimize_adam(theta_init, target_n, t_alloc, rng)
            print(f"    -> score={s_opt:.6f}")
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"    *** New best: {best_score:.6f} (n={target_n}) ***")
                # Quick local search on the new best
                if time.monotonic() < deadline - 40:
                    t_ls = min(30.0, deadline - time.monotonic() - 30)
                    a_ls2, s_ls2 = local_search_peaked(best_seq, t_ls, rng, top_k=5)
                    if s_ls2 < best_score:
                        best_score = s_ls2
                        best_seq = a_ls2.copy()
                        print(f"    After LS: {best_score:.6f}")
    
    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 25:
        t_polish = min(60.0, max(10.0, deadline - time.monotonic() - 20))
        print(f"  Polishing n={len(best_seq)}, score={best_score:.6f}, t={t_polish:.1f}s")
        a_final, s_final = local_search_peaked(best_seq, t_polish, rng, top_k=4)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"  Final: {best_score:.6f}")
    
    # Verify with evaluate_sequence
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return [1.0] * 256
