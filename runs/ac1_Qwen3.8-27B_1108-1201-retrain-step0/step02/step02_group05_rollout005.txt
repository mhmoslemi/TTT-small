Multi-phase optimization using LP-based alternating minimization (SOTA-inspired),

import numpy as np

import cvxpy as cp
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def solve_lp_subproblem(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns optimal b.
    """
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b))
    
    # (a*b)(k) = sum_i a_i * b_{k-i} for valid i, k-i
    # This is linear in b
    # We can write it as A @ b <= M where A[k, j] = a[k-j] if valid
    # But for n=256, this matrix is 511 x 256, which is fine
    
    constraints = []
    # Build constraint matrix
    # For each k in 0..2n-2, (a*b)(k) = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * b[j]
    # So coefficient of b[j] in constraint k is a[k-j] if max(0,k-n+1) <= j <= min(n-1,k)
    
    # Efficient: use convolution structure
    # (a*b)(k) = (convolve(a_rev, b))(k + n - 1) where a_rev = a[::-1]
    # Actually let me just build it directly for clarity
    
    # For n=256, we have 511 constraints and 256 variables
    # Let's build the matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    
    constraints.append(A @ b <= M)
    
    prob = cp.Problem(objective, constraints)
    
    # Try CLARABEL first (fast for LPs), fall back to others
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            prob.solve(solver=cp.SCS)
    
    if b.value is None:
        return a.copy()  # fallback
    
    b_val = np.maximum(b.value, 0)
    if np.sum(b_val) < 1e-10:
        return a.copy()
    
    return b_val

def lp_based_iterations(a_init, n_iters=100, step_start=0.3, step_end=0.05, deadline=None):
    """
    Perform LP-based alternating minimization iterations.
    """

    a = np.maximum(a, 0)
    a = a / np.sum(a)
    best = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        M = np.max(fft_convolve(a, a))
        M *= (1 + 1e-6)  # small tolerance for numerical issues
        
        b = solve_lp_subproblem(a, M)

        s_b = np.sum(b)

def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    cp_ = c ** (p - 1)
    sum_cp = np.sum(cp_ * c)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp_)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta

def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Cosine window
    t = np.arange(n)
    inits['cosine'] = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
    inits['cosine'] = np.maximum(inits['cosine'], 0.01)
    inits['cosine'] = inits['cosine'] / np.sum(inits['cosine'])
    
    # Raised cosine (narrower)
    inits['raised'] = np.sin(np.pi * t / (n - 1)) ** 2
    inits['raised'] = np.maximum(inits['raised'], 0.01)
    inits['raised'] = inits['raised'] / np.sum(inits['raised'])
    
    # Chirp
    for alpha in [0.2, 0.5, 0.8]:
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a = np.clip(a, 0.01, None)
        a += 0.05 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        inits[f'chirp_{alpha}'] = a / np.sum(a)
    
    # Random
    for spread in [0.1, 0.3, 0.5]:
        a = 1.0 + spread * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        inits[f'rand_{spread}'] = a / np.sum(a)
    
    # Gaussian (bell)
    for width_frac in [0.1, 0.2, 0.3]:
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((t - center) / (n * width_frac)) ** 2)
        a += 0.02 * rng.standard_normal(n)
        a = np.clip(a, 0.005, None)
        inits[f'bell_{width_frac}'] = a / np.sum(a)
    
    # Linear ramp
    inits['ramp'] = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = inits['ramp'] / np.sum(inits['ramp'])
    
    # Two-peak
    a = np.zeros(n)
    a[n//4] = 1
    a[3*n//4] = 1
    a = a + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    inits['two_peak'] = a / np.sum(a)
    
    return inits

def polish_local_search(a_init, time_limit, rng):
    """
    Fast local search with targeted peak reduction and coordinate descent.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution once
        c = fft_convolve(current, current)
        peak_val = np.max(c)
        
        # Identify peak locations
        peak_indices = np.where(c > 0.95 * peak_val)[0]
        
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.40:
            # Targeted peak reduction
            # Find which pairs contribute most to the peak
            for pk in peak_indices[:5]:  # top few peaks
                # c[pk] = sum_i current[i] * current[pk-i]
                contribs = np.zeros(n)
                for i in range(min(n, pk + 1)):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                top_k = max(5, n // 15)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                # Compensate by increasing others
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = min(15, len(other) // 2)
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.60:
            # Coordinate descent-like: pick a few coordinates and optimize
            n_coords = rng.integers(3, 10)
            idx = rng.choice(n, size=n_coords, replace=False)
            for i in idx:
                # Try increasing and decreasing
                for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]:
                    trial = current.copy()
                    trial[i] = max(0.01/n, trial[i] * (1 + delta))
                    trial = trial / np.sum(trial)
                    ts = local_score(trial)
                    if ts < current_score:
                        current = trial
                        current_score = ts
                        candidate = trial.copy()
                        break
        elif r < 0.75:
            # Symmetric perturbation
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.02, size=n))
        elif r < 0.90:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))
        
        candidate = np.maximum(candidate, 1e-10)
        candidate = candidate / np.sum(candidate)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):
            current = candidate
            current_score = new_score
        
        temperature = max(temperature * cooling, 1e-10)
    
    return best, best_score

def run_gradient_refinement(a_init, deadline, max_time=120):
    """Run L-BFGS with p-norm annealing on a given initialization."""
    n = len(a_init)
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(a_init)
    
    p_schedule = [16, 32, 64, 128, 256]
    time_per_p = max_time / len(p_schedule)
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        
        p_deadline = time.monotonic() + time_per_p
        
        result = minimize(
            pnorm_obj_grad, theta.copy(),
            method='L-BFGS-B',
            jac=True, args=(p,),
            options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10, 
                     'maxfun': 1500, 'maxls': 50}
        )
        theta_new = result.x
        a_new = softmax(theta_new)
        s = local_score(a_new)
        
        if s < best_score:
            best_score = s
            best_theta = theta_new.copy()
        
        if time.monotonic() > p_deadline:
            break
    
    return softmax(best_theta), best_score

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Uses LP-based alternating minimization (convex subproblems), L-BFGS gradient
refinement with p-norm annealing, and targeted local search polish.
Key innovation: direct LP subproblem that is linear in the new sequence given
the current one, enabling efficient convex optimization at each step.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def local_score(a):
    """Score for normalized sequence (sum=1): 2n * max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def build_constraint_matrix(n):
    """
    Build the structure for the LP constraint matrix.
    A[k, j] = 1 if a[k-j] would be the coefficient, i.e., k-j in [0, n-1].
    We precompute the indices.
    """
    rows = []
    cols = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_subproblem(a, M, row_idx, col_idx):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Uses precomputed index structure for the banded matrix.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build the constraint: A @ b <= M
    # A[k, j] = a[k-j] for valid (k,j) pairs
    # Using the precomputed indices
    a_values = a[col_idx]  # a[k-j] for each (k,j) pair
    # We need A @ b where A has entries a[k-j] at position (k,j)
    # This is: for each k, sum_j a[k-j] * b[j] <= M
    # We can write this as: the entries of b weighted by a
    
    # Efficient approach: for each k, (a*b)(k) = sum of a[k-j]*b[j] for valid j
    # Let's build it as a sparse-matrix-like product
    
    # Actually for n=256, let's just build the dense matrix
    A = np.zeros((2 * n - 1, n))
    A[row_idx, col_idx] = a_values
    # Fix: row_idx and col_idx give positions, a_values gives the values
    # A[row, col] = a[k-col] where k=row, so a[row - col]
    # We already have this: for pair (k,j), A[k,j] = a[k-j]
    # a_values[i] = a[col_idx[i]] but that's not right
    # Let me rebuild properly
    
    # For pair (k, j): A[k, j] = a[k - j]
    A[row_idx, col_idx] = a[row_idx - col_idx]
    
    constraints = [A @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return a.copy()
    
    if b_var.value is None:
        return a.copy()
    
    b_val = np.maximum(b_var.value, 0.0)
    if np.sum(b_val) < 1e-10:
        return a.copy()
    
    return b_val


def lp_based_iterations(a_init, deadline, max_iters=500, step_start=0.25, step_end=0.05):
    """
    Perform LP-based alternating minimization iterations.
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    n = len(a)
    best = a.copy()
    best_score = local_score(a)
    
    # Precompute constraint matrix indices
    row_idx, col_idx = build_constraint_matrix(n)
    
    t0 = time.monotonic()
    stall_count = 0
    stall_threshold = 30
    last_improvement = best_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(a, a)
        M = np.max(c) * (1.0 + 1e-8)
        
        b = solve_lp_subproblem(a, M, row_idx, col_idx)
        
        # Rescale b to match sum of a
        s_a = np.sum(a)
        s_b = np.sum(b)
        if s_b > 1e-10:
            b = b * (s_a / s_b)
        
        # Linear interpolation with decaying step
        progress = it / max(1, max_iters - 1)
        t = step_start + (step_end - step_start) * progress
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / np.sum(a_new)
        
        score = local_score(a_new)
        
        if score < best_score:
            best_score = score
            best = a_new.copy()
            if score < last_improvement - 1e-6:
                last_improvement = score
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        
        a = a_new
        
        # If stalled, try a larger step or restart
        if stall_count > stall_threshold:
            stall_count = 0
            # Perturb and continue
            a = best * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)
    
    return best, best_score


def pnorm_obj_grad(theta, p):
    """
    Objective and gradient for 2n * ||conv(softmax(theta), softmax(theta))||_p.
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_safe = np.maximum(c, 1e-30)
    cp_ = c_safe ** (p - 1)
    sum_cp = np.sum(cp_ * c_safe)
    if sum_cp < 1e-30:
        return 0.0, np.zeros(n)
    
    pnorm = sum_cp ** (1.0 / p)
    score = 2.0 * n * pnorm
    
    factor = 4.0 * n * (sum_cp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, cp_)
    grad_a = factor * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def soft_max_obj_grad(theta, tau):
    """
    Objective and gradient for 2n * soft_max(conv(softmax(theta), softmax(theta)), tau).
    """
    n = len(theta)
    a = softmax(theta)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    soft_max_val = c_max + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    score = 2.0 * n * soft_max_val
    
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, w)
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return score, grad_theta


def run_gradient_optimization(a_init, deadline):
    """Run L-BFGS with p-norm annealing then soft-max."""
    n = len(a_init)
    theta = np.log(a_init + 1e-30)
    best_theta = theta.copy()
    best_score = local_score(a_init)
    
    # Phase A: p-norm annealing

                best_theta = theta_new.copy()
            theta = best_theta

        else:

        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):

    # Random
    for spread in [0.1, 0.3, 0.6]:
        a = 1.0 + spread * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        inits[f'rand_{spread}'] = a / np.sum(a)
    
    # Gaussian
    for wf in [0.15, 0.25, 0.4]:
        center = (n - 1) / 2.0
        a = np.exp(-0.5 * ((t - center) / (n * wf)) ** 2)
        a += 0.01 * rng.standard_normal(n)
        a = np.clip(a, 0.005, None)
        inits[f'bell_{wf}'] = a / np.sum(a)
    
    # Two-lobe
    a = np.zeros(n)
    a[n // 4] = 1.0
    a[3 * n // 4] = 1.0
    a += 0.2 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    inits['two_lobe'] = a / np.sum(a)
    
    # Periodic-like
    for freq in [2, 3, 5]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t / n)
        a = np.clip(a, 0.01, None)
        a += 0.05 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        inits[f'periodic_{freq}'] = a / np.sum(a)
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization:
    1. LP-based alternating minimization with diverse initializations
    2. L-BFGS gradient refinement (p-norm then soft-max annealing)
    3. Extended LP iterations on best
    4. Second gradient pass
    5. Local search polish
    6. Length exploration
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 50.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 1e-8, 1000.0)
        start_seq = start_seq / np.sum(start_seq)
        s = local_score(start_seq)
        print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()
    
    # === Phase 1: LP-based optimization with diverse initializations ===
    print("\n=== Phase 1: LP-based alternating minimization ===")
    
    n = 256
    inits = make_initializations(n, rng)
    
    # Add starting sequence
    if start_seq is not None:
        if len(start_seq) == n:
            inits['start'] = start_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, start_seq)
            a_interp = np.maximum(a_interp, 1e-6)
            inits['start_interp'] = a_interp / np.sum(a_interp)
    
    # Also add perturbed best
    if best_seq is not None:
        for pert in [0.05, 0.1, 0.2]:
            a_pert = best_seq * np.exp(rng.normal(0, pert, n))
            a_pert = np.maximum(a_pert, 1e-6)
            inits[f'best_pert_{pert}'] = a_pert / np.sum(a_pert)
    
    # Sort by initial score
    init_scores = {name: local_score(a) for name, a in inits.items()}
    sorted_inits = sorted(init_scores.items(), key=lambda x: x[1])
    
    print(f"  Generated {len(sorted_inits)} initializations")
    for name, score in sorted_inits[:8]:
        print(f"    {name}: {score:.6f}")
    
    # Run LP on top initializations
    max_lp_time = 50.0  # seconds per LP run
    n_lp_allowed = int((deadline - t_global - 400) / max_lp_time)
    n_lp_runs = min(len(sorted_inits), max(3, n_lp_allowed))
    
    for i, (name, score) in enumerate(sorted_inits[:n_lp_runs]):
        if time.monotonic() > deadline - 400:
            break
        
        a_init = inits[name]
        lp_deadline = time.monotonic() + max_lp_time
        print(f"\n  LP run {i+1}/{n_lp_runs}: init='{name}', score={score:.6f}")
        
        a_best, s_best = lp_based_iterations(
            a_init, deadline=lp_deadline,
            max_iters=400, step_start=0.25, step_end=0.05
        )
        
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** NEW BEST: {s_best:.6f} ***")
        else:
            print(f"    LP best: {s_best:.6f}")
    
    # === Phase 2: Gradient refinement ===
    print(f"\n=== Phase 2: Gradient refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        grad_deadline = min(time.monotonic() + 350, deadline - 300)
        a_grad, s_grad = run_gradient_optimization(best_seq, grad_deadline)
        if s_grad < best_score:
            best_score = s_grad
            best_seq = a_grad.copy()
            print(f"  *** NEW BEST: {s_grad:.6f} ***")
        else:
            print(f"  Gradient best: {s_grad:.6f}")
    
    # === Phase 3: Extended LP on current best ===
    print(f"\n=== Phase 3: Extended LP (best={best_score:.6f}) ===")
    if best_seq is not None:
        lp_deadline = min(time.monotonic() + 180, deadline - 200)
        a_lp, s_lp = lp_based_iterations(
            best_seq, deadline=lp_deadline,
            max_iters=600, step_start=0.12, step_end=0.03
        )
        if s_lp < best_score:
            best_score = s_lp
            best_seq = a_lp.copy()
            print(f"  *** NEW BEST: {s_lp:.6f} ***")
        else:
            print(f"  Extended LP: {s_lp:.6f}")
    
    # === Phase 4: Second gradient pass ===
    print(f"\n=== Phase 4: Second gradient pass (best={best_score:.6f}) ===")
    if best_seq is not None:
        grad_deadline = min(time.monotonic() + 150, deadline - 120)
        a_grad2, s_grad2 = run_gradient_optimization(best_seq, grad_deadline)
        if s_grad2 < best_score:
            best_score = s_grad2
            best_seq = a_grad2.copy()
            print(f"  *** NEW BEST: {s_grad2:.6f} ***")
        else:
            print(f"  2nd gradient: {s_grad2:.6f}")
    
    # === Phase 5: Local search polish ===
    print(f"\n=== Phase 5: Local search polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        polish_time = max(60, deadline - time.monotonic() - 10)
        print(f"  Time available for polish: {polish_time:.0f}s")
        polished, p_score = polish_local_search(best_seq, polish_time, rng)
        if p_score < best_score:
            best_score = p_score
            best_seq = polished.copy()
            print(f"  *** NEW BEST: {p_score:.6f} ***")
        else:
            print(f"  Polish: {p_score:.6f}")
    
    # === Phase 6: Try nearby lengths ===
    if time.monotonic() < deadline - 80:
        print(f"\n=== Phase 6: Length exploration (best={best_score:.6f}) ===")
        for n_try in [240, 272, 300, 320]:
            if time.monotonic() > deadline - 30:
                break
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 1e-6)
            a_try = a_try / np.sum(a_try)
            
            s_try = local_score(a_try)
            # Quick LP
            lp_dl = time.monotonic() + 40
            a_best_try, s_lp_try = lp_based_iterations(
                a_try, deadline=lp_dl, max_iters=150, step_start=0.2, step_end=0.05
            )
            
            total_best = min(s_try, s_lp_try)
            if total_best < best_score:
                best_score = total_best
                if s_lp_try < s_try:
                    best_seq = a_best_try.copy()
                print(f"  n={n_try}: score={total_best:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n_try}: score={total_best:.6f}")
    
    # Final verification
    print(f"\n{'='*50}")
    print(f"FINAL: local_score={best_score:.6f}, n={len(best_seq)}")
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"evaluate_sequence confirms: {final_score:.6f}")
    print(f"{'='*50}")
    
    return final_list
