    # This is 2 * (a_reversed * w)[j + n - 1]
    # where a_reversed = a[::-1]
    # Because (a_rev * w)[k] = sum_i a_rev[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]
    # Let m = n-1-i, so i = n-1-m:
    # = sum_m a[m] * w[k-n+1+m]
    # We want this to equal sum_m a[m] * w[j+m] when k-n+1 = j, i.e., k = j+n-1
    # So g_j = 2 * (a_rev * w)[j + n - 1]
    
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n-1:2*n-1]
    
    return value, g, c, w, c_max

def optimize_gradient(n, theta, time_limit, rng, verbose=True):

def simulate_annealing_polish(a_init, time_limit, rng, verbose=True):
    """Simulated annealing with peak-directed moves."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.99997
    step = 0
    
    if verbose:
        print(f"  SA start: score={best_score:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        r = rng.random()
        
        if r < 0.3:
            # Peak-directed: reduce top contributor, increase random element
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                if 0 <= peak_idx - i < n:
                    contribs[i] = current[i] * current[peak_idx - i]
            top_k = max(4, n // 16)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                amt = current[j_red] * rng.uniform(0.01, 0.05)
                current[j_red] -= amt
                current[j_inc] += amt
                current[j_red] = max(0.0, current[j_red])
        elif r < 0.5:
            # Scale a random element
            j = int(rng.integers(0, n))
            current[j] *= np.exp(rng.normal(0, 0.1))
            current[j] = max(0.0, current[j])
        elif r < 0.7:
            # Scale a random block
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.05))
            current[start:start + width] *= factor
            current = np.clip(current, 0.0, 1000.0)
        elif r < 0.85:
            # Swap portions of two blocks
            w = max(2, int(rng.integers(2, max(3, n // 10))))
            i1 = int(rng.integers(0, n - w))
            i2 = int(rng.integers(0, n - w))
            if i2 != i1:
                current[i1:i1+w], current[i2:i2+w] = current[i2:i2+w].copy(), current[i1:i1+w].copy()
        else:
            # Global small perturbation
            current *= np.exp(rng.normal(0, 0.003, size=n))
            current = np.clip(current, 0.0, 1000.0)
        
        new_score = score_fast(current)
        
        if new_score < current_score:
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if step % 5000 == 0 and verbose:
                    print(f"    SA step={step}, best={best_score:.6f}, temp={temp:.6f}")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score
        
        temp = max(temp * cooling, 1e-8)
        
        # Restart if temperature gets too low
        if temp <= 1e-8:
            current = best + 0.01 * rng.standard_normal(n)
            current = np.clip(current, 0.0, 1000.0)
            current_score = score_fast(current)
            temp = 0.001
    
    return best, best_score

def coordinate_descent_polish(a_init, time_limit, rng, verbose=True):
    """Coordinate descent: for each coordinate, try a few multipliers."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    iters = 0
    
    if verbose:
        print(f"  CD start: score={best_score:.6f}")
    
    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if current[j] < 1e-8:
                continue
            
            base = current[j]
            best_local = base
            best_local_score = current_score
            
            for frac in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                current[j] = base * frac
                s = score_fast(current)
                if s < best_local_score:
                    best_local_score = s
                    best_local = base * frac
            
            current[j] = best_local
            if best_local != base:
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
        
        iters += 1
        if not improved and iters > 2:
            # Try a random kick
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.2)))
            current_score = score_fast(current)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        
        if iters % 50 == 0 and verbose:
            print(f"    CD iter={iters}, best={best_score:.6f}")
    
    return best, best_score

"""
Multi-phase optimization for minimizing autocorrelation peak of non-negative sequences.
Combines refined soft-max gradient descent, simulated annealing with peak-directed moves,
and coordinate descent polish. Uses the scale-invariance of the score to fix sum(a)=sqrt(2n).
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def softmax(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def softmax_conv_grad(a, tau):
    """
    Compute soft-max value and gradient of max of convolution.
    Returns (value, gradient_wrt_a, convolution, weights).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_sh = np.exp(shifted)
    sum_exp = np.sum(exp_sh)
    w = exp_sh / sum_exp
    value = c_max + tau * np.log(sum_exp)

    # Gradient: g_j = 2 * sum_m a_m * w_{j+m} = 2 * (a_rev * w)[j+n-1]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]

    return value, g, c, w, c_max


def gradient_descent(n, theta_init, time_limit, rng):
    """
    Refined gradient descent with adaptive soft-max temperature in theta-space.
    Uses Adam optimizer with periodic restarts on stagnation.
    """
    S_target = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0

    a = softmax(theta) * S_target
    best_score = score_fast(a)
    best_theta = theta.copy()
    no_improve = 0
    patience = 4000

    t0 = time.monotonic()
    print(f"    GradOpt start: score={best_score:.6f}, n={n}")

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit

        # Temperature: exponential decay from high to very low
        tau_hi = 8.0 / n
        tau_lo = 0.005 / n
        tau = tau_hi * (tau_lo / tau_hi) ** elapsed

        a = softmax(theta) * S_target
        value, g_a, c, w, c_max = softmax_conv_grad(a, tau)

        # Convert to theta-space: g_theta[k] = a_k * (g_a[k] - dot(a, g_a)/S_target)
        inner = np.dot(a, g_a) / S_target
        g_theta = a * (g_a - inner)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g_theta
        v_buf = beta2 * v_buf + (1 - beta2) * g_theta ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        # Learning rate: decays slowly
        lr = 3.0 / (1.0 + 0.05 * step) ** 0.4

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic score check and escape
        if step % 300 == 0:
            a_check = softmax(theta) * S_target
            sc = score_fast(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()
                no_improve = 0
                if step % 3000 == 0:
                    print(f"      step={step}, score={sc:.6f}, best={best_score:.6f}, tau={tau:.2e}")
            else:
                no_improve += 300

            if no_improve > patience:
                # Escape mechanism
                r = rng.random()
                if r < 0.4:
                    # Gaussian reheat around best
                    theta = best_theta + 0.4 * rng.standard_normal(n)
                elif r < 0.7:
                    # Targeted: perturb peak contributors
                    a_cur = softmax(best_theta) * S_target
                    c_cur = fft_convolve(a_cur, a_cur)
                    # Find all near-peak locations
                    peak_val = np.max(c_cur)
                    peak_indices = np.where(c_cur >= peak_val * 0.99)[0]
                    # Compute contributions to the worst peak
                    worst_peak = int(np.argmax(c_cur))
                    contribs = np.zeros(n)
                    for i in range(n):
                        if 0 <= worst_peak - i < n:
                            contribs[i] = a_cur[i] * a_cur[worst_peak - i]
                    top_k = np.argsort(contribs)[-max(6, n // 12):]
                    theta[top_k] += 0.35 * rng.standard_normal(len(top_k))
                    # Also boost low contributors
                    bot_k = np.argsort(contribs)[:max(6, n // 12)]
                    theta[bot_k] -= 0.2 * rng.standard_normal(len(bot_k))
                else:
                    # Block scaling perturbation
                    block = max(2, n // 16)
                    start = int(rng.integers(0, max(1, n - block)))
                    theta[start:start + block] += 0.2 * rng.standard_normal(block)

                no_improve = 0
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                step = 0

    a_final = softmax(theta) * S_target
    final_score = score_fast(a_final)
    if final_score < best_score:
        best_score = final_score
        best_theta = theta.copy()

    return best_theta, best_score


def sa_polish(a_init, time_limit, rng):
    """
    Simulated annealing with peak-directed mutation strategies.
    Maintains non-negativity naturally.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    temp = 0.003
    cooling = 0.99996
    step = 0

    print(f"    SA start: score={best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        step += 1
        r = rng.random()

        if r < 0.35:
            # Peak-directed: reduce top contributor, boost random
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                if 0 <= peak_idx - i < n:
                    contribs[i] = current[i] * current[peak_idx - i]
            top_k = max(4, n // 16)
            top_idx = np.argsort(contribs)[::-1][:top_k]
            j_red = top_idx[int(rng.integers(0, top_k))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red:
                frac = rng.uniform(0.02, 0.06)
                amt = current[j_red] * frac
                current[j_red] -= amt
                current[j_inc] += amt
                current[j_red] = max(0.0, current[j_red])
        elif r < 0.55:
            # Scale random element
            j = int(rng.integers(0, n))
            current[j] *= np.exp(rng.normal(0, 0.12))
            current[j] = max(0.0, current[j])
        elif r < 0.7:
            # Scale random block
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            factor = np.exp(rng.normal(0, 0.06))
            current[start:start + width] *= factor
            current = np.clip(current, 0.0, 1000.0)
        elif r < 0.85:
            # Two-element swap
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                current[i1], current[i2] = current[i2], current[i1]
        else:
            # Global micro-perturbation
            current *= np.exp(rng.normal(0, 0.004, size=n))
            current = np.clip(current, 0.0, 1000.0)

        new_score = score_fast(current)

        if new_score < current_score:
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if step % 10000 == 0:
                    print(f"      SA step={step}, best={best_score:.6f}, temp={temp:.6f}")
        elif temp > 1e-10 and rng.random() < np.exp(-(new_score - current_score) / temp):
            current_score = new_score

        temp = max(temp * cooling, 1e-10)

        # Reheat when stuck
        if temp <= 1e-9:
            current = best + abs(0.01 * rng.standard_normal(n))
            current = np.clip(current, 0.0, 1000.0)
            current_score = score_fast(current)
            temp = 0.002
            if current_score < best_score:
                best_score = current_score
                best = current.copy()

    return best, best_score


def coord_descent(a_init, time_limit, rng):
    """Coordinate descent with geometric试探 steps."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score

    t0 = time.monotonic()
    iters = 0
    fracs = [0.6, 0.75, 0.85, 0.92, 1.0, 1.08, 1.18, 1.3, 1.5]

    print(f"    CD start: score={best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if current[j] < 1e-10:
                # Try turning on a zero element
                current[j] = current.mean() * 0.1 * rng.uniform(0.5, 2.0)
                s = score_fast(current)
                if s < current_score:
                    current_score = s
                    improved = True
                    if s < best_score:
                        best_score = s
                        best = current.copy()
                else:
                    current[j] = 0.0
                continue

            base = current[j]
            best_local = base
            best_local_score = current_score

            for frac in fracs:
                current[j] = base * frac
                s = score_fast(current)
                if s < best_local_score:
                    best_local_score = s
                    best_local = base * frac

            if best_local != base:
                current[j] = best_local
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()

        iters += 1

        if not improved and iters > 3:
            # Random kick to escape
            n_kicks = rng.integers(1, 4)
            for _ in range(n_kicks):
                j = int(rng.integers(0, n))
                current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.25)))
            current_score = score_fast(current)
            if current_score < best_score:
                best_score = current_score
                best = current.copy()

        if iters % 100 == 0:
            print(f"      CD iter={iters}, best={best_score:.6f}")

    return best, best_score


def make_init(n, rng, method='broadband'):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if method == 'broadband':
        a = np.ones(n)
        for k in range(1, 20):
            freq = k * (1.0 + 0.6 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.04 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'sparse':
        a = np.ones(n) * 0.3
        n_bumps = n // 8
        for _ in range(n_bumps):
            center = int(rng.integers(0, n))
            width = rng.uniform(2, 6)
            amp = rng.uniform(0.5, 1.5)
            a += amp * np.exp(-(t - center) ** 2 / (2 * width ** 2))
        return np.clip(a, 0.01, None)
    else:
        return np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, None)
