F(c)[m] = C * sum_{k=0}^{2n-2} exp(-2pi*i*m*k/N) = C * (1 - exp(-2pi*i*m*(2n-1)/N)) / (1 - exp(-2pi*i*m/N))

This is a Dirichlet kernel (sinc-like function), not a delta.

So F(a) = sqrt(F(c)) = sqrt(C * D_N(m)) where D_N is the Dirichlet kernel.

Then a = IFFT(sqrt(C * D_N)). This might not be non-negative!

Let's check: if a = IFFT(sqrt(C * D_N)), is a >= 0?

This is the "Fourier factorization" of the autocorrelation. For a valid autocorrelation, F(c) must be non-negative real (which it is, since c is non-negative and even). So sqrt(F(c)) is well-defined (real, non-negative).

But a = IFFT(sqrt(F(c))) might have negative values! The condition for a >= 0 is not automatically satisfied.

So the "ideal" solution (uniform c) may not correspond to a non-negative a. We need to find the closest "achievable" c to the uniform target.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_subgradient(c, a, top_k=8):
    """Compute subgradient of max(c) w.r.t. a, using top-k peaks.
    grad[m] = 2 * sum_k w_k * a[k-m] for k in top peaks"""
    n = len(a)
    # Get top-k lag positions
    if top_k < len(c):
        top_idx = np.argsort(c)[::-1][:top_k]
    else:
        top_idx = np.arange(len(c))
    
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / len(top_idx)
    
    # grad[m] = 2 * sum_k w[k] * a[k-m] where 0 <= k-m < n
    # This is 2 * (w * a_rev)[n-1 : 2n-1] where a_rev = a[::-1]
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    # full_conv has length (2n-1) + n - 1 = 3n-2
    # We need indices [n-1, 2n-2] of full_conv
    end_idx = min(2 * n - 1, len(full_conv))
    grad = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def nesterov_subgradient(a_init, S, time_budget, top_k=8, beta0=0.9):

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    y = a.copy()  # Nesterov's extrapolated point
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = np.inf
    
    # Initial evaluation
    c = fft_convolve_self(a)
    best_obj = c.max()
    
    alpha = 0.1  # initial step size
    t = 0
    beta = beta0
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Nesterov extrapolation
        beta_t = (1 + t) / (3 + t)  # standard Nesterov schedule
        # Actually, let's use a simpler scheme:
        # y = a + beta_t * (a - a_prev)
        # but we need to track a_prev
        
        # Simpler: just use gradient descent with momentum
        # a_new = a - alpha * grad + momentum * (a - a_prev)
        
        c_y = fft_convolve_self(y)
        grad = compute_subgradient(c_y, y, top_k)
        
        # Projected gradient step
        a_new = y - alpha * grad
        a_new = np.maximum(a_new, 0.0)  # project to non-negative
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.full(n, S / n)
        else:
            a_new = a_new * (S / s)  # project to sum = S
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Backtracking line search
        if obj_new > c_y.max():
            alpha *= 0.5
            if alpha < 1e-10:
                alpha = 1e-10
                # Reset
                y = a.copy()
        else:
            alpha = min(alpha * 1.05, 0.5)  # grow step size slowly
        
        # Nesterov update
        a_prev = a.copy()
        a = a_new
        beta_t = (1 + t) / (3 + t)
        y = a + beta_t * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-10:
            y = y * (S / sy)
    
    return best_a, best_obj

def golden_section_coordinate_descent(a, S, time_budget, max_sweeps=100):

    n = len(a)
    a = a.copy().astype(np.float64)
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = fft_convolve_self(a).max()
    
    golden = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-12:

                lo, hi = 0.0, S / n * 2
            else:
                lo, hi = 0.0, orig * 3.0
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)

            def eval_at(val):
                a[i] = val
                s = a.sum()
                if s < 1e-10:
                    return np.inf

                    c2 = c1

                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)

            # Best point is the midpoint of the final interval
            optimal = (lo + hi) / 2.0

def soft_max_objective_grad(theta, n, T, S):
    """Compute soft-max objective and gradient w.r.t. log-parameters."""
    a = S * np.exp(theta - theta.max())
    a = a / a.sum()
    a = a * S  # ensure sum = S
    
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Soft-max
    shifted = T * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / T
    
    # Weights
    w = np.exp(shifted)
    w = w / w.sum()
    
    # Gradient w.r.t. a
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    grad_a = 2.0 * full_conv[n - 1: end_idx][:n]
    
    # Chain rule for log-parameters
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta
