    fa = np.fft.rfft(a, size)

    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

    return 2.0 * n * np.max(conv) / (S * S)

    for it in range(n_iter):
        conv = fft_convolve(a, a)
        max_c = np.max(conv)
        # Reduce peaks: a ← a - alpha * (conv - flat) projected onto a's support
        alpha = 1.0 / max(n, 10)
        a = a - alpha * np.convolve(a, a)[:n]  # approximate gradient step
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

def nesterov_smoothed_descent(a_init, n, time_limit, tau_start, tau_end, lr, rng):
    """Nesterov-accelerated gradient descent on smoothed max convolution."""
    a = a_init.copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum = 1
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.inf
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        # Exponential tau decay
        tau = tau_start * (tau_end / tau_start) ** (elapsed ** 0.6)
        
        # Nesterov extrapolation
        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        beta = (t_prev - 1.0) / t_mom
        y = a + beta * (a - a_prev)
        y = project_simplex(y)
        
        value, grad, conv = smoothed_max_and_grad(y, tau)
        
        # Gradient step
        lr_eff = lr / (1.0 + 0.1 * step)
        candidate = y - lr_eff * grad
        candidate = project_simplex(candidate)
        
        a_prev = a
        a = candidate
        step += 1
        
        # Check actual value periodically
        if step % 20 == 0:
            actual = np.max(fft_convolve(a, a))
            if actual < best_val:
                best_val = actual
                best_a = a.copy()
        
        # Adaptive tau: if stuck, increase tau temporarily
        if step % 500 == 0:
            actual = np.max(fft_convolve(a, a))
            if actual >= best_val * 1.001:
                # Stuck, might need to explore
                pass  # The Nesterov momentum should handle this
    
    return best_a, best_val, 2.0 * n * best_val

def lp_direction(a, M, time_limit_single):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)_k <= M for all k, g >= 0.
    Use scipy.optimize.linprog.
    """
    from scipy.optimize import linprog
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) → minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: conv(a, g)_k <= M for k = 0, ..., 2n-2
    # conv(a,g)_k = sum_i a_i * g_{k-i} (where indices are valid)
    # For each k, this is a linear constraint in g.
    # Number of constraints: 2n-1
    # But for large n, this is too many. Let's use only the top constraints.
    
    # Compute current conv(a, a) to find active constraints
    conv_aa = fft_convolve(a, a)
    # The active constraints are those near the maximum
    top_k = min(n, 20)  # Use top 20 constraints
    active_indices = np.argsort(conv_aa)[-top_k:]
    
    # Build constraint matrix A such that A @ g <= b
    # For constraint k: sum_i a_i * g_{k-i} <= M
    # The coefficient of g_j in constraint k is a_{k-j} (if 0 <= k-j < n)
    A = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[idx, j] = a[i]
    b = M * np.ones(len(active_indices))
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'time_limit': max(1.0, time_limit_single), 'presolve': True})
        if result.success:
            g = result.x
            g = np.clip(g, 0, None)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g
    except Exception:
        pass
    return None

def coordinate_polish(a, time_limit, rng):
    """Aggressive coordinate descent polish."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    current_score = 2.0 * n * np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 2000
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        r = rng.random()
        
        if r < 0.35:
            # Pair transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            frac = rng.uniform(0.02, 0.2)
            cand = a.copy()
            amt = cand[i] * frac
            cand[i] -= amt
            cand[j] += amt
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand /= cs
            
        elif r < 0.55:
            # Single element scale
            j = rng.integers(n)
            factor = np.exp(rng.normal(0, 0.08))
            cand = a.copy()
            cand[j] *= factor
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand /= cs
            
        elif r < 0.75:
            # Block modification
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            cand[start:start+width] *= np.exp(rng.normal(0, 0.05, size=width))
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand /= cs
            
        else:
            # Peak-targeted reduction
            conv = fft_convolve(a, a)
            peak = np.max(conv)
            peak_idx = int(np.argmax(conv))
            # Reduce elements contributing to the peak
            # conv[peak_idx] = sum_i a_i * a_{peak_idx - i}
            cand = a.copy()
            n_reduce = max(3, n // 32)
            # Find which pairs contribute most
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs = contribs + np.zeros(n)  # just to initialize
            # Simplified: reduce top contributors
            top_idx = np.argsort(a)[-n_reduce:]
            cand[top_idx] *= (1.0 - rng.uniform(0.02, 0.08, size=n_reduce))
            # Redistribute
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                total_removed = np.sum(a[top_idx] - cand[top_idx])
                n_add = min(10, len(others))
                add_idx = rng.choice(others, size=n_add, replace=False)
                cand[add_idx] += total_removed / n_add
            cand = np.clip(cand, 0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand /= cs
        
        new_score = 2.0 * n * np.max(fft_convolve(cand, cand))
        if new_score < current_score:
            a = cand
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            no_improve += 1
    
    return best_a, best_score
