```python
    n = len(a)

    S = np.sqrt(2.0 * n)  # Target sum
```

```python
import cvxpy as cp

def lp_direction(f, M):
    """
    Solve: maximize sum(b) subject to b >= 0 and (f*b)[k] <= M for all k.
    Returns the optimal b.
    """
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # Compute f * b symbolically
    # (f*b)[k] = sum_i f[i] * b[k-i] for valid i
    # This is a linear function of b
    
    constraints = []
    for k in range(2 * n - 1):
        # (f*b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * b[k-i]
        coeffs = np.zeros(n)
        for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
            j = k - i
            if 0 <= j < n:
                coeffs[j] += f[i]
        constraints.append(coeffs @ b <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if b.value is None:
        return None
```

```python
"""
LP-based peak reduction with spectral initialization and gradient polishing.
Uses linear programming to find directions that increase the sum while maintaining
the peak constraint, combined with spectral methods for initialization and
gradient-based methods for final polishing.
"""

import numpy as np
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def build_convolution_matrix(f):
    """Build matrix T such that T @ b = convolve(f, b) for any b of same length."""
    n = len(f)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            T[k, j] = f[i]
    return T


def lp_direction(f, M, solver=cp.CLARABEL):
    """
    Solve: maximize sum(b) subject to b >= 0 and (f*b)[k] <= M for all k.
    Returns (b_opt, sum_b_opt).
    """
    n = len(f)
    T = build_convolution_matrix(f)
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve()
        except:
            return None, 0.0
    
    if b.value is None:
        return None, 0.0
    return b.value, float(cp.sum(b).value)


def lp_iteration(f, max_iters=500, verbose=True):
    """
    LP-based iteration: repeatedly solve LP and update f.
    """
    n = len(f)
    S_target = np.sqrt(2.0 * n)
    
    # Normalize f
    S = np.sum(f)
    f = f * S_target / S
    
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > 2000:  # Safety
            break
        
        c = fft_convolve(f, f)
        M = np.max(c)
        
        b_opt, sum_b = lp_direction(f, M)
        
        if b_opt is None or sum_b <= S_target * 1.001:
            # No improvement possible with LP
            if verbose:
                print(f"    LP iter {it}: sum_b={sum_b:.4f} <= S_target={S_target:.4f}, stopping")
            break
        
        # Update: f_new = f + t * b_opt
        # Choose t to maximize improvement
        # Score change ≈ 2t * (conv(f,b)[k_max]/M - sum(b)/S)
        # We want conv(f,b)[k_max] to be as small as possible
        # For now, use small t
        t = 0.1 * S_target / sum_b
        
        f_new = f + t * b_opt
        f_new = np.maximum(f_new, 0.0)
        
        # Normalize
        S_new = np.sum(f_new)
        if S_new < 1e-10:
            break
        f_new = f_new * S_target / S_new
        
        new_score = local_score(f_new)
        
        if new_score < current_score:
            f = f_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                if verbose:
                    print(f"    LP iter {it}: score={new_score:.6f}, sum_b={sum_b:.2f}, t={t:.4f}")
        else:
            # Try smaller t
            t = 0.01 * S_target / sum_b
            f_new = f + t * b_opt
            f_new = np.maximum(f_new, 0.0)
            S_new = np.sum(f_new)
            if S_new > 1e-10:
                f_new = f_new * S_target / S_new
                new_score = local_score(f_new)

                        best_f = f_new.copy()
                    if verbose:

            print(f"    LP iter {it}: cur={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score
```

**Alternative: Min-max formulation**

The score is 2n * max(c) / S^2. Let's fix S = sqrt(2n) (WLOG due to scale invariance). Then we want to minimize max(c).

This is a min-max problem: minimize t subject to c[k] <= t for all k, a >= 0, sum(a) = sqrt(2n).

This is non-convex because c[k] = sum_i a[i]*a[k-i] is bilinear in a.

However, we can use the following trick: fix a linearization of the peak. At each iteration:
- Let k* = argmax(c)
- The constraint c[k*] <= t becomes: sum_i a[i]*a[k*-i] <= t
- This is still bilinear, but we can linearize: a[i]*a[k*-i] <= (a[i] + a[k*-i])^2 / 4 (by AM-GM)

Or better: use the fact that for fixed a, the gradient of c[k*] with respect to a is:
dc[k*]/da[j] = 2 * (a[k*-j] + a[j-k*]) (with appropriate bounds)

This gives us a subgradient, and we can do projected subgradient descent.

**Projected subgradient descent:**
- Objective: min max_k c[k] (with sum(a) = S, a >= 0)
- Subgradient at k*: g[j] = 2*(a[k*-j] + a[j-k*]) (for the active constraint k*)
- Step: a_new = a - lr * g
- Project: a_new = projection(a_new, {a' >= 0, sum(a') = S})

The projection onto {a' >= 0, sum(a') = S} can be done by:
1. Start with a_new
2. Subtract a constant from all elements
3. Clip negative elements to 0
4. Repeat until sum = S
(This is the "water-filling" algorithm)

This is simple and effective. Let me implement it.

Actually, for the max of multiple c[k], the subgradient is a convex combination of the subgradients of the active constraints. If only one constraint is active (k* = argmax), the subgradient is just g[j] = 2*(a[k*-j] + a[j-k*]).

If multiple constraints are active (multiple k with c[k] = max(c)), we average their subgradients.

Let me combine all these ideas into a final algorithm:

```python
def projected_subgradient(a_init, time_budget, rng):
    """Projected subgradient descent on min max(c) with sum constraint."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init * S / np.sum(a_init)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    lr = 0.1
    step = 0
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        k_star = np.argmax(c)
        peak = c[k_star]
        
        # Subgradient
        g = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j < n:
                val += a[k_star - j]
            if 0 <= j - k_star < n:
                val += a[j - k_star]
            g[j] = 2.0 * val
        
        # Step
        a_new = a - lr * g
        
        # Project: a_new >= 0, sum(a_new) = S
        a_new = project_nonneg_sum(a_new, S)
        
        # Check score
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay learning rate
        step += 1
        lr = 0.1 / np.sqrt(1 + step * 0.01)
        
        # Occasionally try a larger step or restart
        if step % 1000 == 0:
            # Check if stuck
            c_new = fft_convolve(a, a)
            if np.max(c_new) > peak * 1.01:
                # Reset to best
                a = best_a.copy()
                lr = 0.05
    
    return best_a, best_score


def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    n = len(a)
    x = a.copy()
    for _ in range(100):
        x = np.maximum(x, 0)
        diff = S - np.sum(x)
        if abs(diff) < 1e-10:
            break
        # Distribute the difference
        # Add to positive elements proportionally, or just add uniformly
        pos_mask = x > 0
        if np.sum(pos_mask) == 0:
            x = np.ones(n) * S / n
            break
        x[pos_mask] += diff / np.sum(pos_mask)
    return x
```

Wait, the projection might not be exact with this simple approach. Let me use a proper algorithm:

The projection onto {x >= 0, sum(x) = S} is:
x* = max(a - tau, 0) where tau is chosen so that sum(max(a - tau, 0)) = S.

This can be found by bisection on tau.

```python
def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S} using bisection on tau."""
    n = len(a)
    if S <= 0:
        return np.zeros(n)
    if np.all(a >= 0) and abs(np.sum(a) - S) < 1e-10:
        return a.copy()
    
    # Find tau such that sum(max(a - tau, 0)) = S
    lo = a.min() - 1
    hi = a.max() + 1
    
    # Actually, tau should be in [min(a) - 1, max(a)]
    # If tau < min(a), all elements are positive: sum = sum(a) - n*tau
    # If tau > max(a), all elements are zero: sum = 0
    # We want sum = S
    
    for _ in range(100):
        tau = (lo + hi) / 2
        x = np.maximum(a - tau, 0)
        s = x.sum()
        if s > S:
            lo = tau  # Need larger tau to reduce sum
        else:
            hi = tau
        if abs(s - S) < 1e-12:
            break
    
    return np.maximum(a - tau, 0)
```

This is O(n * log(precision)) which is very fast.

"""
Multi-strategy optimization for minimizing autocorrelation peak score.
Combines spectral domain flattening, LP-based direction finding,
projected subgradient descent, and coordinate descent polishing.
"""

import numpy as np
import time
import cvxpy as cp


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(a)
    if S <= 0:
        return np.zeros(n)
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    tau = (lo + hi) / 2.0
    for _ in range(200):
        x = np.maximum(a - tau, 0.0)
        s = x.sum()
        if s > S:
            lo = tau
        else:
            hi = tau
        tau = (lo + hi) / 2.0
        if abs(s - S) < 1e-12:
            break
    return np.maximum(a - tau, 0.0)


def spectral_flatten_iterations(a, n_iter, alpha=0.5):
    """
    Iterative spectral flattening.
    alpha: how much to reduce dynamic range (0=full flatten, 1=no change)
    """
    n = len(a)
    S = np.sum(a)
    pad_size = 1
    while pad_size < 2 * n:
        pad_size <<= 1
    
    current = a.copy()
    best = a.copy()
    best_score = local_score(a)
    
    for it in range(n_iter):
        A = np.fft.fft(current, pad_size)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        log_mag = np.log(mag_safe)
        log_target = np.mean(log_mag)
        new_log_mag = log_target + alpha * (log_mag - log_target)
        new_mag = np.exp(new_log_mag)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new, pad_size)[:n].real
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new = a_new * S / S_new
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        current = a_new
    
    return best, best_score


def build_conv_matrix(f):
    """Build Toeplitz matrix for convolution with f."""
    n = len(f)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            T[k, j] = f[i]
    return T


def solve_lp(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    n = len(f)
    T = build_conv_matrix(f)
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve()
        except Exception:
            return None, 0.0
    if b.value is None:
        return None, 0.0
    return np.asarray(b.value).flatten(), float(np.sum(b.value))


def lp_optimize(a_init, time_budget, rng):
    """LP-based optimization."""
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = a_init * S_target / np.sum(a_init)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_budget:
            break
        c = fft_convolve(a, a)
        M = np.max(c)
        b_opt, sum_b = solve_lp(a, M)
        if b_opt is None:
            break
        if sum_b <= S_target * 1.001:
            break
        # Try different step sizes
        improved = False
        for t_factor in [0.5, 0.2, 0.1, 0.05]:
            t = t_factor * S_target / sum_b
            a_new = a + t * b_opt
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-10:
                continue
            a_new = a_new * S_target / S_new
            score = local_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        if not improved:
            # No step size worked, try smaller
            t = 0.01 * S_target / sum_b
            a_new = a + t * b_opt
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new > 1e-10:
                a_new = a_new * S_target / S_new
                a = a_new
        if it % 100 == 0:
            print(f"    LP iter {it}: score={local_score(a):.6f}, best={best_score:.6f}, sum_b={sum_b:.2f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def subgradient_optimize(a_init, time_budget, rng):
    """Projected subgradient descent."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init * S / np.sum(a_init)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    step = 0
    lr = 0.05
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        c = fft_convolve(a, a)
        peak_val = np.max(c)
        
        # Use top-k peaks for a smoother subgradient
        k_peaks = 5
        top_k = np.argsort(c)[-k_peaks:][::-1]
        
        g = np.zeros(n)
        for k_star in top_k:
            for j in range(n):
                val = 0.0
                if 0 <= k_star - j < n:
                    val += a[k_star - j]
                if 0 <= j - k_star < n:
                    val += a[j - k_star]
                g[j] += val
        g *= 2.0 / k_peaks
        
        # Projected gradient step
        a_new = a - lr * g
        a_new = project_nonneg_sum(a_new, S)
        
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        step += 1
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
        elif step % 500 == 0:
            lr = max(0.001, 0.05 / np.sqrt(step / 500))
        
        # Restart if stuck
        if no_improve > 1000:
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a * S / a.sum()
            no_improve = 0
            lr = 0.05
        
        if step % 1000 == 0:
            print(f"    SG step {step}: cur={local_score(a):.6f}, best={best_score:.6f}, lr={lr:.4f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a_init, time_budget, rng):
    """Coordinate descent polish."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init * S / np.sum(a_init)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_budget:
                break
            if a[j] < 1e-8:
                continue
            
            # Try scaling this coordinate
            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                old = a[j]
                a[j] = old * factor
                score = local_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old
                    break
        
        if not improved and rng.random() < 0.1:
            # Random perturbation
            j = int(rng.integers(0, n))
            a[j] = max(0.0, a[j] * np.exp(rng.normal(0, 0.1)))
            a = a * S / a.sum()
    
    return best_a, best_score


def make_initial(n, rng, method='chirp'):
    """Create initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if method == 'chirp':
        a = 1.0 + 0.3 * np.cos(np.pi * t * t / n)
        a += 0.05 * rng.standard_normal(n)
    elif method == 'broadband':
        a = np.ones(n)
        for k in range(1, 12):
            freq = k * (1 + 0.7 * rng.random())
            phase = rng.uniform(0, 2 * np.pi)
            a += (0.1 / np.sqrt(k)) * np.cos(2 * np.pi * freq * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
    elif method == 'random':
        a = 1.0 + 0.5 * rng.standard_normal(n)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)
