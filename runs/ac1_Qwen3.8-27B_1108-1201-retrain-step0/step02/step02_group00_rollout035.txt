```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses Nesterov-accelerated active-set subgradient + FISTA with log-sum-exp.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def active_set_subgradient(a, c, top_k):
    """
    Compute weighted subgradient using top-K active lags.
    g[m] = 2 * sum_k w[k] * a[k-m] (with a[i]=0 for i out of bounds)
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    # Build the weighted subgradient using FFT
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def nesterov_subgrad(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512, cooldown=500):
    """
    Nesterov-accelerated active-set subgradient method.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive top_k: start large, decrease
        frac = min(1.0, it / cooldown)
        top_k = max(1, int(top_k_max * (1 - frac) + top_k_init * frac))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        grad = active_set_subgradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = 2.0 * S / (n * gnorm * (1.0 + 0.01 * it))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Restart if stuck
        if no_improve > 200 and rng.random() < 0.1:
            y = project_simplex(best_local + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            no_improve = 0
    
    return best_local, best_score

def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with adaptive log-sum-exp."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-8)
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Occasional restart
        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

def polish_exact(a_init, n, S, max_time, rng):

def bessel_init(n, rng):
    """Initialize with a Bessel-like shape (high at edges, low in center)."""
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1)
    # Shape: high at edges, low in center
    # Use 1 - cos(π*t) which is 0 at edges and 2 at center... no, that's the opposite.
    # We want high at edges: use cos(π*t) which is 1 at edges and -1 at center
    # Or: 1 + cos(π*t) which is 2 at edges and 0 at center
    shape = 1.0 + 0.8 * np.cos(np.pi * t)
    shape = np.maximum(shape, 0.01)  # Ensure positive
    return shape + rng.random(n) * 0.01
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy:
1. Normalize to sum = sqrt(2n), reducing objective to max of self-convolution.
2. Multi-phase optimization:
   Phase 1: Nesterov-accelerated subgradient with adaptive active set.
   Phase 2: FISTA with log-sum-exp (adaptive beta).
   Phase 3: Exact subgradient polishing with backtracking.
3. Multiple initializations: given sequence, Bessel-like, structured.
4. Time management with deadline and restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_active_gradient(a, c, top_k):
    """
    Compute weighted subgradient using top-K active lags via FFT.
    g[m] = 2 * sum_k w[k] * a[(k-m) mod bounds]
    """
    n = len(a)
    conv_len = 2 * n - 1
    top_k = min(top_k, conv_len)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(conv_len)
    for i in range(top_k):
        w_full[int(top_indices[i])] = weights[i]

    fsize = 1 << (conv_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def nesterov_subgrad(a_init, n, S, max_time, rng,
                     top_k_start=512, top_k_end=1, cooldown=300,
                     restart_interval=500, noise_scale=0.005):
    """
    Nesterov-accelerated active-set subgradient method with adaptive K.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = float(c.max()) if (c := fft_convolve_self(a)).any() else np.inf
    best_score = a.copy().sum() ** 2 / n  # rough initial
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    last_restart = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive top-k: interpolate from top_k_start to top_k_end
        frac = min(1.0, it / float(cooldown))
        top_k = max(1, int(round(top_k_start * (1 - frac) + top_k_end * frac)))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()

        grad = compute_active_gradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: Lipschitz-based with decay
        step = 2.0 * S / (n * gnorm * (1.0 + 0.005 * it))
        step = min(step, 2.0 * S / (n * gnorm))

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart to break out of oscillation
        if it - last_restart >= restart_interval:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score * 1.001:
                y = project_simplex(
                    best_local + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
                t = 1.0
                last_restart = it

    return best_local, best_score


def fista_lse(a_init, n, S, max_time, rng,
              beta_init=10.0, beta_max=800.0, noise_scale=0.003):
    """FISTA with adaptive log-sum-exp smooth approximation."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    last_restart = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()

        grad = compute_lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta * (1.0 + 0.0005 * it))
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Occasional noisy restart
        if it - last_restart >= 4000 and rng.random() < 0.15:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score * 1.002:
                y = project_simplex(
                    best_local + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
                t = 1.0
                last_restart = it
                beta = beta_init

    return best_local, best_score


def polish_backtracking(a_init, n, S, max_time, rng,
                        top_k=5, max_fails=3):
    """
    Polishing with exact subgradient and backtracking line search.
    Ensures monotone decrease of the objective.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    consecutive_fails = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()

        grad = compute_active_gradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step = 1.0 * S / (n * gnorm)
        improved = False
        for attempt in range(max_fails + 1):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur - 1e-15:
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            consecutive_fails += 1
            # Add small noise to escape
            a = project_simplex(
                a + rng.normal(0, 1e-6 * S / np.sqrt(n), n), S)
            if consecutive_fails > 50:
                break
        else:
            consecutive_fails = 0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_inits(n, rng):
    """Generate diverse initializations for the search."""
    inits = []

    # Try to use the given global sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0.0, 1000.0)
        if given.size == n and np.sum(given) > 0:
            inits.append(("given", given.copy()))
        elif given.size >= 128:
            # Interpolate to target length
            x_old = np.linspace(0, 1, len(given))
            x_new = np.linspace(0, 1, n)
            from scipy.interpolate import interp1d
            f = interp1d(x_old, given, kind='cubic', fill_value='extrapolate')
            inits.append(("given_interp", np.abs(f(x_new))))
    except Exception:
        pass

    # Bessel-like: high at edges, low in center
    t = np.linspace(0, 1, n, endpoint=False)
    shape_edges = 1.0 + 0.6 * np.cos(np.pi * t)
    shape_edges = np.maximum(shape_edges, 0.01)
    inits.append(("edges", shape_edges.copy()))

    # Opposite: high in center
    shape_center = 1.0 + 0.6 * np.cos(2 * np.pi * (t - 0.5))
    shape_center = np.maximum(shape_center, 0.01)
    inits.append(("center", shape_center.copy()))

    # Uniform with small noise
    inits.append(("uniform", np.ones(n) + rng.normal(0, 0.01, n)))

    # Random
    inits.append(("random", rng.random(n) * 0.5 + 0.1))

    # Multi-frequency sinusoidal
    sine_shape = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 7 * t)
    sine_shape = np.maximum(sine_shape, 0.01)
    inits.append(("sine", sine_shape.copy()))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))

    global_best = np.inf
    global_a = None

    print("=== Starting search: n=%d, target < 1.5027 ===" % n)

    # Generate initializations
    inits = generate_inits(n, rng)
    print("Generated %d initializations" % len(inits))

    # Track best from each phase
    all_results = []

    # --- Phase 1: Nesterov subgradient (broad, fast) ---
    remaining = deadline - time.monotonic()
    phase1_frac = 0.35
    phase1_time = remaining * phase1_frac
    n_inits = len(inits)
    per_init = phase1_time / n_inits

    print("\n--- Phase 1: Nesterov subgrad (%d inits, ~%.0fs each) ---" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        t0 = time.monotonic()
        a_res, score = nesterov_subgrad(
            arr, n, S, per_init, rng,
            top_k_start=1024, top_k_end=1, cooldown=200,
            restart_interval=400, noise_scale=0.005)
        a_norm = safe_normalize(a_res, n)
        c = fft_convolve_self(a_norm)
        obj = 2.0 * n * c.max() / (S ** 2)
        all_results.append((obj, a_norm.copy(), "p1_" + name))
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  P1 [%s]: %.6f" % (name, obj))

    all_results.sort(key=lambda x: x[0])

    # --- Phase 2: FISTA with log-sum-exp (refine top candidates) ---
    remaining = deadline - time.monotonic()
    phase2_frac = 0.35
    phase2_time = remaining * phase2_frac
    n_top = min(3, len(all_results))
    per_init2 = phase2_time / max(1, n_top)

    print("\n--- Phase 2: FISTA LSE (top %d, ~%.0fs each) ---" % (n_top, per_init2))

    all_results2 = []
    for obj, a_val, name in all_results[:n_top]:
        if time.monotonic() > deadline - 30:
            break
        t0 = time.monotonic()
        a_res, score = fista_lse(
            a_val, n, S, per_init2, rng,
            beta_init=20.0, beta_max=600.0, noise_scale=0.003)
        a_norm = safe_normalize(a_res, n)
        c = fft_convolve_self(a_norm)
        obj2 = 2.0 * n * c.max() / (S ** 2)
        all_results2.append((obj2, a_norm.copy(), "p2_" + name))
        if obj2 < global_best:
            global_best = obj2
            global_a = a_norm.copy()
            print("  P2 [%s]: %.6f" % (name, obj2))

    all_results2.sort(key=lambda x: x[0])

    # --- Phase 3: Polishing with backtracking (final squeeze) ---
    remaining = deadline - time.monotonic()
    phase3_frac = 0.30
    phase3_time = remaining * phase3_frac
    n_top3 = min(2, len(all_results2))
    per_init3 = phase3_time / max(1, n_top3)

    print("\n--- Phase 3: Polishing backtracking (top %d, ~%.0fs each) ---" % (n_top3, per_init3))

    all_results3 = []
    for obj, a_val, name in all_results2[:n_top3]:
        if time.monotonic() > deadline - 15:
            break
        t0 = time.monotonic()
        a_res, score = polish_backtracking(
            a_val, n, S, per_init3, rng, top_k=5, max_fails=5)
        a_norm = safe_normalize(a_res, n)
        c = fft_convolve_self(a_norm)
        obj3 = 2.0 * n * c.max() / (S ** 2)
        all_results3.append((obj3, a_norm.copy(), "p3_" + name))
        if obj3 < global_best:
            global_best = obj3
            global_a = a_norm.copy()
            print("  P3 [%s]: %.6f" % (name, obj3))

    # --- Phase 4: Final high-beta FISTA on the overall best ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n--- Phase 4: Final high-beta FISTA (%.0fs) ---" % remaining)
            a_res, score = fista_lse(
                global_a, n, S, remaining * 0.6, rng,
                beta_init=200.0, beta_max=2000.0, noise_scale=0.001)
            a_norm = safe_normalize(a_res, n)
            c = fft_convolve_self(a_norm)
            obj4 = 2.0 * n * c.max() / (S ** 2)
            if obj4 < global_best:
                global_best = obj4
                global_a = a_norm.copy()
                print("  P4: %.6f" % obj4)

            # Final polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a_res2, _ = polish_backtracking(
                    global_a, n, S, remaining2, rng, top_k=3, max_fails=8)
                a_norm2 = safe_normalize(a_res2, n)
                c2 = fft_convolve_self(a_norm2)
                obj5 = 2.0 * n * c2.max() / (S ** 2)
                if obj5 < global_best:
                    global_best = obj5
                    global_a = a_norm2.copy()
                    print("  P4-polish: %.6f" % obj5)

    # --- Final verification ---
    if global_a is not None:
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\n=== Final verification: %.6f ===" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * 0.5
    else:
        print("\n=== No valid solution found, using fallback ===")
        global_a = np.ones(256, dtype=np.float64) * 0.5

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * 0.5

    print("Returning sequence of length %d, best objective = %.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
```
