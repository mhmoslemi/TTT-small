```python
def lp_smooth_step(f, M, max_sum=None):
    """
    Solve LP: maximize sum(g) subject to f*g <= M for all k, g >= 0.
    Optionally: sum(g) <= max_sum.
    """
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: for each k in 0..2n-2, sum_i f[i]*g[k-i] <= M
    # Valid i for each k: max(0, k-(n-1)) <= i <= min(n-1, k)
    A = np.zeros((2*n-1, n))
    b = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[k, i] = f[i]
    
    # Solve
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    return None
```

```python
def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def smooth_max(c, tau):
    """Log-sum-exp approximation to max."""
    c_max = np.max(c)
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))

def smooth_max_grad(c, tau):
    """Value and gradient of smooth max w.r.t. c."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    value = c_max + tau * np.log(np.sum(exp_shifted))
    grad_c = w  # d/dc of smooth_max
    return value, grad_c
```

```python
"""
LP-smoothing + smooth-max gradient descent for minimizing the peak-to-sum ratio
of the self-convolution of non-negative sequences. Uses iterative LP-based
direction finding combined with annealed log-sum-exp gradient descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast evaluation of the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def smooth_max_value_and_weights(c, tau):
    """Compute smooth max and its weights (softmax over c/tau)."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, w

def smooth_self_conv_gradient(a, tau):
    """
    Compute smooth max of a*a and gradient w.r.t. a.
    Returns (value, grad, c, w).
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = smooth_max_value_and_weights(c, tau)
    
    # grad[j] = 2 * sum_m a[m] * w[j+m] = 2 * (a * w_rev)[2n-2-j]
    w_rev = w[::-1]
    corr = fft_convolve(a, w_rev)
    grad = 2.0 * corr[2*n-2 : n-2 : -1]
    
    return value, grad, c, w

def lp_smooth_direction(f, M):
    """
    Solve LP: maximize sum(g) subject to f*g <= M for all k, g >= 0.
    Returns the optimal g or None if infeasible.
    """
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build constraint matrix A: A[k, i] = f[i] if i is valid for position k
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        A[k, i_start:i_end] = f[i_start:i_end]
    
    b = np.full(2*n-1, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                     options={'presolve': True})
    
    if result.success and result.x is not None:
        return result.x
    return None

def lp_smooth_iteration(f, t=0.1):
    """
    One LP smoothing iteration:
    1. Compute M = max(f*f)
    2. Solve LP to get g
    3. Rescale g to match sum(f)
    4. Update: f_new = (1-t)*f + t*g
    Returns (f_new, improvement, M).
    """
    n = len(f)
    c = fft_convolve(f, f)
    M = np.max(c)
    S = np.sum(f)
    
    g = lp_smooth_direction(f, M)
    if g is None:
        return f, 0.0, M
    
    # Rescale g to have same sum as f
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return f, 0.0, M
    g = g * S / g_sum
    
    # Update
    f_new = (1 - t) * f + t * g
    f_new = np.clip(f_new, 0.0, 1000.0)
    
    # Compute improvement
    c_new = fft_convolve(f_new, f_new)
    M_new = np.max(c_new)
    improvement = M - M_new
    
    return f_new, improvement, M

def lp_smooth_optimize(f_init, time_limit, rng):
    """
    Run LP smoothing iterations with adaptive step size.
    """
    f = f_init.astype(np.float64).copy()
    current_score = local_score(f)
    best = f.copy()
    best_score = current_score
    n = len(f)
    t0 = time.monotonic()
    
    # Adaptive step size
    t = 0.1
    no_improve_count = 0
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        f_new, improvement, M = lp_smooth_iteration(f, t)
        
        new_score = local_score(f_new)
        
        if new_score < current_score:
            current_score = new_score
            f = f_new
            if current_score < best_score:
                best_score = current_score
                best = f.copy()
                no_improve_count = 0
            # Increase step size slightly if improving
            t = min(0.3, t * 1.05)
        else:
            no_improve_count += 1
            # Decrease step size
            t = max(0.01, t * 0.7)
            
            # If stuck for a while, try a different approach
            if no_improve_count > 50:
                # Try larger step with fresh LP
                t = 0.2
                no_improve_count = 0
        
        iteration += 1
        if iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, t={t:.3f}, time={elapsed:.0f}s")
    
    return best, best_score

def smooth_max_gradient_descent(f_init, time_limit, rng):
    """
    Gradient descent on the smooth max of the self-convolution.
    Uses log-sum-exp with annealed temperature.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    current_score = local_score(f)
    best = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    # Adam parameters
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0
    
    # Temperature schedule
    tau_start = 5.0 / n
    tau_end = 0.01 / n
    
    no_improve = 0
    patience = 500
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Anneal temperature
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        # Compute gradient
        value, grad, c, w = smooth_self_conv_gradient(f, tau)
        
        # We want to minimize max(c) / S^2 (with normalization S fixed)
        # d(score)/d(f[j]) = 2n * d(max(c))/d(f[j]) / S^2 - 2n * max(c) * 2*S / S^4 * df_sum/df[j]
        # = 2n * grad[j] / S^2 - 4n * max(c) / S^3
        # (since dS/df[j] = 1)
        
        # Actually, let's minimize max(c) subject to sum(f) = S, f >= 0
        # Using Lagrange multiplier: minimize max(c) - lambda * (sum(f) - S)
        # Gradient of max(c) w.r.t. f is grad (computed above)
        # So effective gradient: grad - lambda * ones
        # At optimum: grad - lambda * ones = 0, so lambda = mean(grad)
        
        grad_effective = grad - np.mean(grad)
        
        # Adam update
        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad_effective
        v_buf = beta2 * v_buf + (1 - beta2) * grad_effective ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)
        
        lr = 1.0 / np.sqrt(step + 1)
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = np.clip(f, 0.0, 1000.0)
        
        # Check score periodically
        if step % 100 == 0:
            score = local_score(f)
            if score < best_score:
                best_score = score
                best = f.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            if no_improve > patience:
                # Restart from best with perturbation
                f = best + 0.01 * rng.standard_normal(n) * np.mean(best)
                f = np.clip(f, 0.0, None)
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                step = 0
                no_improve = 0
        
        if step % 1000 == 0:
            score = local_score(f)
            print(f"      GD step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}")
    
    final_score = local_score(f)
    if final_score < best_score:
        best_score = final_score
        best = f.copy()
    
    return best, best_score

def coordinate_polish(f_init, time_limit, rng):
    """
    Fast coordinate descent: for each coordinate, try a few multiplicative
    factors and keep the best.
    """
    f = f_init.astype(np.float64).copy()
```

```python
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def smooth_max_value_and_weights(c, tau):
    """Compute smooth max and softmax weights."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, w

    """
```

                start = int(rng.integers(0, n - block))
                f = best.copy()
                f[start:start + block] *= np.exp(rng.normal(0, 0.05, block))
            f = np.clip(f, 0.0, None)
            current_score = local_score(f)
            t = 0.1
            no_improve = 0

        iteration += 1
        if verbose and iteration % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LP it={iteration}, cur={current_score:.7f}, best={best_score:.7f}, t={t:.3f}, {elapsed:.0f}s")

    return best, best_score


def smooth_max_gradient_descent(f_init, time_limit, rng, verbose=True):
    """Gradient descent on smooth max of self-convolution."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    best = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    tau_start = 10.0 / n
    tau_end = 0.05 / n
    no_improve = 0
    patience = 400

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        value, grad, c, w = smooth_self_conv_gradient(f, tau)
        grad_eff = grad - np.mean(grad)

        step += 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad_eff
        v_buf = beta2 * v_buf + (1 - beta2) * grad_eff ** 2
        m_hat = m_buf / (1.0 - beta1 ** step)
        v_hat = v_buf / (1.0 - beta2 ** step)

        lr = 0.5 / np.sqrt(step + 1)
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = np.clip(f, 0.0, 1000.0)

        if step % 100 == 0:
            score = local_score(f)
            if score < best_score:
                best_score = score
                best = f.copy()
                no_improve = 0
            else:
                no_improve += 100

            if no_improve > patience:
                f = best + 0.003 * np.mean(best) * rng.standard_normal(n)
                f = np.clip(f, 0.0, None)
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                step = 0
                no_improve = 0

        if verbose and step % 2000 == 0:
            score = local_score(f)
            print(f"        GD step={step}, score={score:.7f}, best={best_score:.7f}, tau={tau:.2e}")

    final_score = local_score(f)
    if final_score < best_score:
        best_score = final_score
        best = f.copy()
    return best, best_score


def coordinate_polish(f_init, time_limit, rng, verbose=True):
    """Coordinate descent with multiplicative factors."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    current_score = local_score(f)
    best = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    factors = [0.75, 0.88, 0.95, 1.05, 1.12, 1.25, 1.4]
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-10:
                continue
            base = f[j]
            for fac in factors:
                f[j] = base * fac
                ns = local_score(f)
                if ns < current_score:
                    current_score = ns
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = f.copy()
                else:
                    f[j] = base
                    break

        if not improved:
            j = int(rng.integers(0, n))
            if f[j] > 1e-10:
                f[j] *= np.exp(rng.normal(0, 0.08))

        iteration += 1
        if verbose and iteration % 20 == 0:
            print(f"        CD iter={iteration}, cur={current_score:.7f}, best={best_score:.7f}, {time.monotonic()-t0:.0f}s")

    return best, best_score
