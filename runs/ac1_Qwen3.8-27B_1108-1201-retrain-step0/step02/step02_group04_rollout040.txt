def coordinate_descent(u_init, time_limit, rng, verbose=False):
    """Refine using coordinate descent: try changing one element at a time."""
    n = len(u_init)
    u = u_init.astype(np.float64).copy()
    u = np.clip(u, 0, None)
    S = np.sum(u)
    if S < 1e-10:
        return u, np.inf
    u = u / S
    
    best_score = score_from_u(u)
    best_u = u.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try random coordinate updates
        for _ in range(1000):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(n)
            # Try several values for u[i]
            for delta in [-u[i]*0.5, -u[i]*0.2, u[i]*0.2, u[i]*0.5, u[i]+0.01/n, u[i]-0.01/n]:
                u_new = u.copy()
                u_new[i] = max(0, u[i] + delta)
                S_new = np.sum(u_new)
                if S_new < 1e-10:
                    continue
                u_new = u_new / S_new
                s_new = score_from_u(u_new)
                if s_new < best_score - 1e-12:
                    u = u_new
                    best_score = s_new
                    best_u = u_new.copy()
                    improved = True
                    break
        
        if not improved:
            # Try adding noise
            noise = rng.standard_normal(n) * 0.01 / n
            u_new = np.clip(u + noise, 0, None)
            S_new = np.sum(u_new)
            if S_new > 1e-10:
                u_new = u_new / S_new
                s_new = score_from_u(u_new)
                if s_new < best_score * 1.01:
                    u = u_new
    
    return best_u, best_score
