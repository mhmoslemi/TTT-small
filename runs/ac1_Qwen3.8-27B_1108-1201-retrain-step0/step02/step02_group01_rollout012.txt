import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def a_from_theta(theta, S):
    """Convert log-parameters to sequence: a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_softmax_and_weights(c, T):
    """Compute soft-max value and weights efficiently."""
    c_max = c.max()
    shifted = T * (c - c_max)
    # log-sum-exp
    max_shifted = shifted.max()  # should be 0
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = c_max + log_sum / T
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()
    return obj, c_max, weights


def compute_grad_from_weights(w, a):
    """Compute grad[m] = 2 * sum_j w[m+j] * a[j] for m=0..n-1."""
    n = len(a)
    a_rev = a[::-1]
    full_conv = fft_convolve(w, a_rev)
    end_idx = min(2 * n - 1, len(full_conv))
    grad = 2.0 * full_conv[n - 1: end_idx][:n]
    return grad


def run_adam_anneal(theta_init, n, S, time_budget):
    """Adam with temperature annealing on log-parameters."""
    theta = theta_init.copy().astype(np.float64)
    n_par = len(theta)
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    a = a_from_theta(theta, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = (time.monotonic() - start) / time_budget
        
        # Temperature: exponential from 3 to 50
        T = 3.0 * np.exp(3.5 * progress)
        
        a = a_from_theta(theta, S)
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_obj:
            best_obj = cur_max
            best_a = a.copy()
        
        # Compute soft-max objective gradient
        _, _, w = compute_softmax_and_weights(c, T)
        grad_a = compute_grad_from_weights(w, a)
        
        # Chain rule through softmax
        dot = np.dot(grad_a, a)
        grad = grad_a * a - (a / S) * dot
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate: start at 0.1, decay to 0.01
        lr = 0.1 * (0.1 ** progress)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj


def run_subgradient_nesterov(a_init, n, S, time_budget, top_k=16):
    """Subgradient method with Nesterov acceleration for the max function."""
    a = a_init.copy().astype(np.float64)
    a = a * (S / a.sum())
    
    # Nesterov state
    y = a.copy()  # interpolation point
    a_prev = a.copy()
    
    start = time.monotonic()
    c = fft_convolve_self(y)
    best_obj = c.max()
    best_a = a.copy()
    
    t = 0
    while time.monotonic() - start < time_budget:
        t += 1
        progress = (time.monotonic() - start) / time_budget
        
        # Compute subgradient at y
        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best_obj:
            best_obj = cur_max
            best_a = a.copy()
        
        # Top-k weights
        top_idx = np.argpartition(c_y, -top_k)[-top_k:]
        w = np.zeros(len(c_y))
        w[top_idx] = 1.0 / top_k
        
        grad = compute_grad_from_weights(w, y)
        
        # Nesterov step
        alpha = 2.0 / (t + 1.0)
        
        # Learning rate: decrease over time
        # Use Polyak-like step: eta = (f(a) - f*) / ||g||^2
        # We don't know f*, so use a heuristic
        lr = 0.5 / (t * (1.0 + t * 0.01))
        
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        # Update Nesterov interpolation
        y = (1.0 - alpha) * a_new + alpha * y
        y = np.maximum(y, 0.0)
        y = y * (S / y.sum())
        
        a_prev = a
        a = a_new
    
    return best_a, best_obj

def perturb_and_reopt(a, n, S, time_budget, rng, n_trials=5):
    """Perturb the sequence and re-optimize from the perturbed point."""

        # Perturb: add multiplicative noise

        a_pert = a * noise
        a_pert = np.maximum(a_pert, 0.0)

def get_initialization(name, n, rng, S):
    """Generate various initializations."""
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / (n / 8)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine":
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 5 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "anti_peak":
        i = np.arange(n)
        a = 1.0 - 0.5 * np.cos(2 * np.pi * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "bimodal":
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n * 0.3) / (n / 12)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.7) / (n / 12)) ** 2)
        return a * (S / a.sum())
    elif name == "sawtooth":
        i = np.arange(n)
        period = n // 4
        a = np.maximum(0, 1.0 - 2.0 * np.abs((i % period) / period - 0.5))
        a = a + 0.1
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)
