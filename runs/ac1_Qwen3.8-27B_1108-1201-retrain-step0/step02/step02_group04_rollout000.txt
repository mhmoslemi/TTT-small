            a_test = a_cur.copy()
            a_test[rng.integers(0, n)] += 0.01 * rng.standard_normal()
            a_test = np.clip(a_test, 0.0, 1000.0)
            s_test = local_score_fast(a_test)
            if s_test < s_cur - 1e-12:
                a_cur = a_test
                s_cur = s_test
        if s_cur < best_local_score:
            best_local_score = s_cur
            best_local = a_cur.copy()

    return best_local, best_local_score
