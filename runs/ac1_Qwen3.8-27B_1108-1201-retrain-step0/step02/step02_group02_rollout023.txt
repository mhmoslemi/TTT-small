Multi-phase optimization for minimizing autocorrelation peak of non-negative step functions.
Uses softmax gradient descent with efficient FFT-based gradients, autocorrelation flattening
local search, and multi-length exploration. Targets score below 1.5027.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def score_from_p(p):
    """Score for probability vector p (sum=1): 2n * max(p*p)."""
    n = len(p)
    c = fft_convolve(p, p)
    return 2.0 * n * np.max(c)


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    if s == 0:
        return np.ones(len(theta)) / len(theta)
    return e / s


def topk_grad_p(p, tau):
    """
    Compute soft-max value and gradient of max(p*p) w.r.t. p.
    Uses O(n log n) FFT-based computation.
    """
    n = len(p)
    c = fft_convolve(p, p)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)

    # Gradient: grad_m = (p * w_rev)_{n-1+m} + (p * w)_m
    w_rev = w[::-1]
    pw = fft_convolve(p, w)
    pwr = fft_convolve(p, w_rev)
    grad = pwr[n - 1:2 * n - 1] + pw[:n]

    return value, grad, c, w


def exact_peak_grad_p(p):
    """Exact subgradient of max(p*p) at the peak lag."""
    n = len(p)
    c = fft_convolve(p, p)
    k_star = int(np.argmax(c))

    # grad_m = p_{m+k*} + p_{m-k*} (with bounds)
    grad = np.zeros(n)
    for m in range(n):
        val = 0.0
        if m + k_star < n:
            val += p[m + k_star]
        if m - k_star >= 0:
            val += p[m - k_star]
        grad[m] = val

    # The factor of 2 comes from the fact that (p*p)_k has p_m appearing twice when k=0
    # But our formula already handles this: grad_m = sum of both terms
    # Actually for the exact max, grad = 2 * grad_m when k_star = 0
    # Let me re-derive: f = (p*p)_{k*} = sum_j p_j p_{j+k*}
    # df/dp_m = p_{m+k*} (if valid) + p_{m-k*} (if valid)
    # This is what we computed above (without the factor of 2)
    # But when k* = 0: df/dp_m = 2*p_m, and our formula gives p_m + p_m = 2*p_m ✓
    # When k* > 0: df/dp_m = p_{m+k*} + p_{m-k*} (one or both may be zero) ✓

    return c[k_star], grad, k_star


def softmax_to_theta_grad(g_p, p):
    """Convert gradient w.r.t. p to gradient w.r.t. theta (softmax parameterization)."""
    return p * (g_p - np.sum(p * g_p))

                best_theta = theta.copy()

    p_final = softmax_normalize(best_theta)
    a_final = p_final * S_target
    final_score = compute_score(a_final)
    if final_score < best_score:
        best_score = final_score

    return a_final, best_score


def flatten_local_search(a_init, time_limit, rng, verbose=False):
    """
    Autocorrelation flattening local search.
    Identifies peak lags and redistributes mass to flatten the autocorrelation.
    """
    n = len(a_init)
    S = np.sum(a_init)
    current = a_init.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.99997
    step = 0

    if verbose:
        print(f"      FL initial: {best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fft_convolve(current, current)
        max_c = np.max(c)
        mean_c = np.mean(c[:n])  # Consider only lags 0 to n-1

        # Strategy selection
        r = rng.random()

        if r < 0.35:
            # Reduce peak contributors, boost non-peak contributors
            peak_idx = int(np.argmax(c))
            # Compute per-element contribution to ALL lags above threshold
            threshold = mean_c * 1.1
            hot_lags = np.where(c[:n] > threshold)[0]

            pressure = np.zeros(n)
            for k in hot_lags:
                for i in range(n):
                    if 0 <= k - i < n:
                        pressure[i] += current[i] * current[k - i]
                    if 0 <= i - k < n:
                        pressure[i] += current[i] * current[i - k]

            # Reduce high-pressure, increase low-pressure
            n_bottom = max(4, n // 10)
            n_top = max(3, n // 15)
            bottom_idx = np.argsort(pressure)[:n_bottom]
            top_idx = np.argsort(pressure)[::-1][:n_top]

            total_top = np.sum(current[top_idx])
            reduction = rng.uniform(0.01, 0.04)
            reduce_amt = total_top * reduction
            current[top_idx] *= (1.0 - reduction)
            # Distribute to bottom
            bottom_weights = np.ones(n_bottom) / n_bottom
            current[bottom_idx] += reduce_amt * bottom_weights

        elif r < 0.55:
            # Pair swap: move mass from high-autocorrelation element to low
            c_contrib = np.zeros(n)
            for i in range(n):
                c_contrib[i] = 0.0
                for k in range(n):
                    if 0 <= k - i < n:
                        c_contrib[i] += c[k] * current[i] * current[k - i] / (S * S)
                    if 0 <= i - k < n:
                        c_contrib[i] += c[k] * current[i] * current[i - k] / (S * S)

            # Pick from top contributors to reduce, random or low to increase
            top_sorted = np.argsort(c_contrib)[::-1]
            j_red = top_sorted[int(rng.integers(0, max(1, n // 8)))]
            j_inc = int(rng.integers(0, n))
            if j_inc != j_red and current[j_red] > 1e-6:
                amt = current[j_red] * rng.uniform(0.01, 0.05)
                current[j_red] -= amt
                current[j_inc] += amt
                current[j_red] = max(0.0, current[j_red])

        elif r < 0.75:
            # Block scaling
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.03))
            current[start:start + width] *= factor

        elif r < 0.9:
            # Lag-targeted: find the hottest lag and directly reduce its contributors
            peak_idx = int(np.argmax(c))
            # The pair (i, peak_idx - i) or (peak_idx - i, i) contributes to c[peak_idx]
            # Reduce the largest contributor pair
            contribs = {}
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            if contribs:
                sorted_items = sorted(contribs.items(), key=lambda x: x[1], reverse=True)
                k_top = len(sorted_items)
                for idx in range(min(3, k_top)):
                    i, val = sorted_items[idx]
                    j = peak_idx - i
                    reduction = rng.uniform(0.02, 0.05)
                    dec = current[i] * reduction
                    current[i] -= dec
                    # Add to a random low element
                    j_inc = int(rng.integers(0, n))
                    current[j_inc] += dec
                    current[i] = max(0.0, current[i])
                    break

        else:
            # Small global perturbation
            current *= np.exp(rng.normal(0, 0.003, size=n))

        current = np.clip(current, 0.0, 1000.0)

        # Renormalize to maintain sum (optional - try both)
        new_score = compute_score(current)

        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
                if verbose and step % 2000 == 0:
                    print(f"      FL step={step}, best={best_score:.6f}")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score
        else:
            # Revert
            current = best.copy()
            current_score = best_score

        temp *= cooling

    return best, best_score


def coordinate_polish(a_init, time_limit, rng):
    """
    Coordinate descent: for each coordinate, try multiple values to minimize score.
    """
    n = len(a_init)
    current = a_init.copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            base = current[j]
            if base < 1e-8:
                continue

            best_local = base
            best_local_score = current_score

            # Try geometric ratios
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                trial = base * frac
                if trial > 1000:
                    continue
                current[j] = trial
                s = compute_score(current)
                if s < best_local_score:
                    best_local_score = s
                    best_local = trial
                current[j] = trial

            if best_local != base:
                current[j] = best_local
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            else:
                current[j] = base

        if not improved:
            # Try a random perturbation
            j = int(rng.integers(0, n))
            current[j] = max(0.0, current[j] * np.exp(rng.normal(0, 0.1)))
            s = compute_score(current)
            if s < current_score * 1.001:
                current_score = s
            iteration += 1

        if iteration % 10 == 0:
            pass  # No print to save time

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: gradient descent + flattening + coordinate polish
    across multiple sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    start_seq = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            s = compute_score(start_seq)
            print(f"Provided: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    except Exception:
        pass

    # Define n values to try and time allocation
    n_configs = [
        (256, 0.20),
        (320, 0.15),
        (384, 0.25),
        (512, 0.20),
        (448, 0.10),
    ]

    # If we have a starting sequence, give more time to its length
    if start_seq is not None:
        n_start = len(start_seq)
        # Add or boost the starting length
        found = False
        for i, (n, frac) in enumerate(n_configs):
            if n == n_start:
                n_configs[i] = (n, min(0.4, frac + 0.15))
                found = True
                break
        if not found:
            n_configs.append((n_start, 0.25))

        # Renormalize fractions
        total_frac = sum(f for _, f in n_configs)
        n_configs = [(n, f / total_frac) for n, f in n_configs]

    print(f"n_configs: {[(n, f) for n, f in n_configs]}")

    for n, frac in n_configs:
        if time.monotonic() > deadline - 30:
            break

        t_alloc = frac * time_budget
        t_rem = deadline - time.monotonic()
        t_alloc = min(t_alloc, t_rem - 10)
        if t_alloc < 10:
            continue

        print(f"\n{'='*50}")
        print(f"Optimizing n={n}, time={t_alloc:.0f}s")
        print(f"{'='*50}")

        # Phase 1: Gradient optimization (70% of allocated time)
        t_grad = t_alloc * 0.65
        t_flatten = t_alloc * 0.25
        t_coord = t_alloc * 0.10

        # Multiple starts
        n_starts = 3 if n <= 320 else 2
        best_local_score = np.inf
        best_local_seq = None

        for s_idx in range(n_starts):
            if time.monotonic() > deadline - 30:
                break

            t_start = (time.monotonic() - t_global) / time_budget if time_budget > 0 else 0
            t_remaining_for_this = min(t_grad / max(1, n_starts - s_idx), deadline - time.monotonic() - 20)
            if t_remaining_for_this < 5:
                break

            # Create initial probability vector
            if s_idx == 0 and start_seq is not None:
                # Interpolate to target n
                if len(start_seq) != n:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a_interp = np.interp(x_new, x_old, start_seq)
                else:
                    a_interp = start_seq.copy()
                p_init = np.clip(a_interp, 0.01, None)
                p_init = p_init / np.sum(p_init)
            else:
                methods = ['random', 'structured', 'bimodal', 'periodic']
                p_init = make_init(n, rng, methods[s_idx % len(methods)])

            a_res, score_res = gradient_optimize(n, p_init, t_remaining_for_this, rng, verbose=True)

            if score_res < best_local_score:
                best_local_score = score_res
                best_local_seq = a_res.copy()
                print(f"        Start {s_idx}: score={score_res:.6f}")

            # Flatten local search (if time remains)
            if time.monotonic() < deadline - 20 and best_local_seq is not None:
                t_flat = min(t_flatten / 3, deadline - time.monotonic() - 15)
                if t_flat > 5:
                    a_flat, score_flat = flatten_local_search(best_local_seq, t_flat, rng, verbose=True)
                    if score_flat < best_local_score:
                        best_local_score = score_flat
                        best_local_seq = a_flat.copy()
                        print(f"        After flatten: score={score_flat:.6f}")

        # Coordinate polish (if time remains)
        if best_local_seq is not None and time.monotonic() < deadline - 15:
            t_cd = min(t_coord, deadline - time.monotonic() - 10)
            if t_cd > 5:
                a_cd, score_cd = coordinate_polish(best_local_seq, t_cd, rng)
                if score_cd < best_local_score:
                    best_local_score = score_cd
                    best_local_seq = a_cd.copy()
                    print(f"        After CD: score={score_cd:.6f}")

        # Update global best
        if best_local_seq is not None and best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f"  *** New global best: {best_score:.6f} (n={n}) ***")

    # Final polish with remaining time
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_final = deadline - time.monotonic() - 10
        if t_final > 10:
            print(f"\nFinal polish, t={t_final:.0f}s")
            n_f = len(best_seq)
            # Alternate between flatten and coordinate
            t_each = t_final / 2
            a_f, s_f = flatten_local_search(best_seq, t_each, rng, verbose=True)
            if s_f < best_score:
                best_score = s_f
                best_seq = a_f.copy()

            remaining = deadline - time.monotonic() - 5
            if remaining > 5:
                a_c, s_c = coordinate_polish(best_seq, remaining, rng)
                if s_c < best_score:
                    best_score = s_c
                    best_seq = a_c.copy()

    # Verify and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"=== FINAL: score={final_eval:.6f}, n={len(final_list)} ===")
        print(f"{'='*50}")

        if final_eval < np.inf:
            return final_list

    # Fallback
    if best_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in best_seq]
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
