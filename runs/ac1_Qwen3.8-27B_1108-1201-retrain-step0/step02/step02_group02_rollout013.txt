    n = len(a)
    m = len(b)

    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    """
    Solve LP: maximize sum(b) subject to (a*b)[k] <= M for all k, b >= 0.
    Returns the optimal b, or None if infeasible.
    """

    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    """Build the convolution constraint matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # a[j] contributes to (a*b)[k] at position k=j, with coefficient a[j]*b[0]
        # Actually, (a*b)[k] = sum_j a[j]*b[k-j]
        # So A[k, m] = a[k-m] if 0 <= k-m < n
        # For fixed j (=m), A[k, j] = a[k-j] if 0 <= k-j < n
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        for k in range(j, min(j+n, 2*n-1)):
            A[k, j] = a[k-j]
    return A

    A = build_conv_matrix(a)

    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:

   So the change is: dv * (a[k-j] + a[j-k]) + dv^2 * I(k=2j)
   
   (With the understanding that a[m] = 0 if m is out of range.)

So for the peak at k*, the new peak value (if it's still at k*) is:
c_new[k*] = c[k*] + dv * (a[k*-j] + a[j-k*]) + dv^2 * I(k*=2j)

And the new sum is: S + dv.

The new score (if peak is still at k*) is:
2n * (c[k*] + dv * g + dv^2 * I) / (S + dv)^2

where g = a[k*-j] + a[j-k*].

To minimize this, take derivative and set to 0:
d/dv [2n * (c + dv*g + dv^2*I) / (S+dv)^2] = 0

This gives a quadratic equation in dv. But I also need to check that the peak hasn't moved to another index.

- Try a[j] * 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5

Actually, each score evaluation requires an FFT convolution, which is O(n log n) ≈ 384 * 9 ≈ 3500 operations. So 7 * 384 * 3500 ≈ 9.4M operations per sweep. At ~100M ops/sec in Python (with numpy), that's about 0.1s per sweep. Very fast!

**Peak shaving:**
Identify the top contributors to the peak and reduce them, while increasing the non-contributors.

For the peak at index k*, the contribution of a[j] to c[k*] is a[j] * a[k*-j] (if both are in range). The top contributors are the j's with the largest a[j] * a[k*-j].

Strategy: reduce the top-k contributors by a small amount, and redistribute to the bottom-k.

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]
