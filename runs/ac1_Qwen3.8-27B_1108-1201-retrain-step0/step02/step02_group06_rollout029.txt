```python
import numpy as np
import time

def fft_convolve(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:len(a)+len(b)-1]

def score(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S*S)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def optimize_subgrad(a_init, time_limit, rng, verbose=True):
    """Nesterov-accelerated subgradient method for min max(conv) s.t. a>=0, sum(a)=1"""
    a = a_init.copy()
    a = np.clip(a, 0, None)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    a_prev = a.copy()
    n = len(a)
    t = 1.0
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    c = 1.0  # step size parameter
    patience = 0
    max_patience = 50000
    
    while time.monotonic() - t0 < time_limit:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = a + beta * (a - a_prev)
        y = project_simplex(y)
        
        conv = fft_convolve(y, y)
        max_val = np.max(conv)
        
        # Find active lags (within 1% of max)
        threshold = max_val * 0.999
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            seg = y[k - i_max : k - i_min + 1][::-1]
            grad[i_min:i_max+1] += 2.0 * seg
        grad /= len(active)
        
        # Step
        lr = c / (t_new * np.sqrt(np.sum(grad**2) + 1e-10))
        candidate = y - lr * grad
        candidate = project_simplex(candidate)
        
        a_prev = a.copy()
        a = candidate
        t = t_new
        step += 1
        
        # Evaluate
        if step % 10 == 0:
            conv_eval = fft_convolve(a, a)
            mv = np.max(conv_eval)
            if mv < best_conv_max:
                best_conv_max = mv
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            # Adaptive step size
            if patience > 1000:
                c *= 0.95
            elif patience == 0:
                c = min(c * 1.01, 10.0)
        
        if step % 5000 == 0 and verbose:
            print(f"        subgrad step={step}, best_max_conv={best_conv_max:.8f}, "
                  f"score={2*n*best_conv_max:.6f}, t={time.monotonic()-t0:.1f}s")
        
        if patience > max_patience:
            break
    
    return best_a, 2.0 * n * best_conv_max
```

For the subgradient method with Nesterov acceleration, the standard result is that the step size should be `c / sqrt(t)` where c is a constant related to the Lipschitz constant of the subgradient. The optimal c depends on the problem.

Actually, for minimizing `f(x) = max_k g_k(x)` where each `g_k` has Lipschitz constant `L_k`, the overall Lipschitz constant is `max_k L_k = max_k ||grad g_k||`.

For our problem, `g_k(a) = conv(a,a)[k]` and `||grad g_k|| = 2 * ||a[k-:]+...]||_2 ≤ 2 * ||a||_2`.

Since `sum(a) = 1` and `a >= 0`, `||a||_2 <= 1`. So `L ≤ 2`.

The standard Nesterov accelerated subgradient method with step size `L / t` gives a convergence rate of `O(L^2 / T)` after T steps.

But in practice, we want to use a larger step size initially and decrease it. Let me use `lr = alpha / t` where alpha starts at some value and is adjusted.

Actually, I think a simpler and more effective approach would be to use the **proximal gradient** method with a carefully chosen step size, combined with backtracking line search.

```python
def optimize_hybrid(a_init, time_limit, rng, verbose=True):
    a = a_init.copy()
    a = np.clip(a, 0, None)
    S = np.sum(a)
    a = a / S
    n = len(a)
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    t0 = time.monotonic()
    step = 0
    
    # Temperature schedule
    tau_max = 5.0 / n
    tau_min = 0.01 / n
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_max * (tau_min / tau_max) ** (frac ** 0.5)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = a + beta * (a - a_prev)
        y = project_simplex(y)
        
        conv = fft_convolve(y, y)
        max_c = np.max(conv)
        
        # Smooth max gradient
        shifted = (conv - max_c) / tau
        # Clip to avoid overflow
        shifted = np.clip(shifted, -100, 100)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient = 2 * cross_corr(y, w)
        # cross_corr[y, w][j] = sum_k w[k] * y[k-j]
        size = 2 * n
        fy = np.fft.rfft(y, size)
        fw = np.fft.rfft(w, size)
        corr = np.fft.irfft(np.conj(fy) * fw, size)[:n]
        grad = 2.0 * corr
        
        # Normalize gradient and step
        grad_norm = np.sqrt(np.sum(grad**2)) + 1e-10
        lr = 2.0 / (t_new * grad_norm)
        
        candidate = y - lr * grad
        candidate = project_simplex(candidate)
        
        a_prev = a.copy()
        a = candidate
        t = t_new
        step += 1
        
        if step % 20 == 0:
            conv_eval = fft_convolve(a, a)
            mv = np.max(conv_eval)
            if mv < best_conv_max:
                best_conv_max = mv
                best_a = a.copy()
    
    return best_a, 2.0 * n * best_conv_max
```
